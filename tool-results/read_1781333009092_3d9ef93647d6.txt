    93→   142→/**
    94→   143→ * Fetch font binary data using XMLHttpRequest instead of fetch().
    95→   144→ *
    96→   145→ * CRITICAL: In Capacitor with CapacitorHttp enabled, the native HTTP plugin
    97→   146→ * intercepts ALL fetch() calls and can corrupt binary data (woff2 fonts).
    98→   147→ * XMLHttpRequest is NOT intercepted by CapacitorHttp, so it returns
    99→   148→ * correct binary data. This is the key fix for mushaf mode in Android APK.
   100→   149→ */
   101→   150→function fetchFontViaXHR(url: string): Promise<ArrayBuffer> {
   102→   151→  return new Promise((resolve, reject) => {
   103→   152→    const xhr = new XMLHttpRequest();
   104→   153→    xhr.open('GET', url, true);
   105→   154→    xhr.responseType = 'arraybuffer';
   106→   155→    xhr.onload = () => {
   107→   156→      if (xhr.status === 200) {
   108→   157→        resolve(xhr.response as ArrayBuffer);
   109→   158→      } else {
   110→   159→        reject(new Error(`XHR HTTP ${xhr.status}`));
   111→   160→      }
   112→   161→    };
   113→   162→    xhr.onerror = () => reject(new Error('XHR network error'));
   114→   163→    xhr.ontimeout = () => reject(new Error('XHR timeout'));
   115→   164→    xhr.timeout = 15000; // 15 second timeout
   116→   165→    xhr.send();
   117→   166→  });
   118→   167→}

   125→   174→  if (_isCapacitor) {

   136→   185→/**
   137→   186→ * Verify a font is truly usable on Canvas by drawing a PUA character
   138→   187→ * and checking if the pixels are non-blank.
   139→   188→ *
   140→   189→ * IMPORTANT: In Android WebView (Capacitor), Canvas pixel verification is unreliable
   141→   190→ * because WebView processes fonts differently. We use document.fonts.check()
   142→   191→ * as a fallback verification method in that case.
   143→   192→ */
   144→   193→function verifyFontOnCanvas(fontName: string): boolean {
   145→   194→  try {
   146→   195→    // In Capacitor/Android WebView, skip pixel-level Canvas verification
   147→   196→    // because it's unreliable — trust document.fonts API instead
   148→   197→    if (_isCapacitor) {
   149→   198→      const fontsCheck = document.fonts.check(`40px "${fontName}"`);
   150→   199→      console.log(`[Mushaf] Capacitor: document.fonts.check for "${fontName}": ${fontsCheck}`);
   151→   200→      return fontsCheck;
   152→   201→    }
   153→   202→
   154→   203→    const testSize = 40;
   155→   204→    const canvas = document.createElement('canvas');
   156→   205→    canvas.width = testSize * 2;
   157→   206→    canvas.height = testSize * 2;
   158→   207→    const ctx = canvas.getContext('2d');
   159→   208→    if (!ctx) return false;
   160→   209→
   161→   210→    ctx.fillStyle = '#FFFFFF';
   162→   211→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   163→   212→
   164→   213→    ctx.font = `${testSize}px "${fontName}"`;
   165→   214→    ctx.fillStyle = '#000000';
   166→   215→    ctx.textAlign = 'center';
   167→   216→    ctx.textBaseline = 'middle';
   168→   217→    ctx.fillText('\uF100', canvas.width / 2, canvas.height / 2);
   169→   218→
   170→   219→    const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
   171→   220→    const data = imageData.data;
   172→   221→    for (let i = 0; i < data.length; i += 4) {
   173→   222→      if (data[i] < 250 || data[i + 1] < 250 || data[i + 2] < 250) {
   174→   223→        return true;
   175→   224→      }
   176→   225→    }
   177→   226→
   178→   227→    return false;
   179→   228→  } catch {
   180→   229→    return false;
   181→   230→  }
   182→   231→}

   184→   233→/**
   185→   234→ * Create a hidden DOM element that uses the font. This forces the browser
   186→   235→ * to download and parse the font binary, making it available for Canvas.
   187→   236→ * Some browsers (especially Android WebView) won't make fonts available
   188→   237→ * to Canvas unless they're used in the DOM first.
   189→   238→ */
   190→   239→function createFontPreloadElement(fontName: string): void {
   191→   240→  const id = `qcf-preload-${fontName}`;
   192→   241→  if (document.getElementById(id)) return;
   193→   242→
   194→   243→  const span = document.createElement('span');
   195→   244→  span.id = id;
   196→   245→  span.setAttribute('aria-hidden', 'true');
   197→   246→  span.style.cssText = `
   198→   247→    position: absolute;
   199→   248→    left: -9999px;
   200→   249→    top: -9999px;
   201→   250→    visibility: hidden;
   202→   251→    font-family: "${fontName}";
   203→   252→    font-size: 40px;
   204→   253→    pointer-events: none;
   205→   254→  `;
   206→   255→  // Use multiple PUA characters to ensure the browser loads the font
   207→   256→  span.textContent = '\uF100\uF101\uF102\uF103\uF104\uF105';
   208→   257→  document.body.appendChild(span);
   209→   258→}

   211→   260→/**
   212→   261→ * Load a QCF4 font using the most reliable method available.
   213→   262→ *
   214→   263→ * Strategy order (most to least reliable):
   215→   264→ * 1. XHR fetch as ArrayBuffer → FontFace from binary data
   216→   265→ *    (Bypasses CapacitorHttp which corrupts binary data)
   217→   266→ * 2. FontFace API with URL (browser handles download)
   218→   267→ * 3. CSS @font-face injection with DOM preload element
   219→   268→ *
   220→   269→ * In Capacitor: We skip pixel-level Canvas verification and trust
   221→   270→ * document.fonts API instead.
   222→   271→ */
   223→   272→async function ensureFontLoaded(fontName: string): Promise<void> {
   224→   273→  if (loadedFonts.has(fontName)) return;
   225→   274→
   226→   275→  // Deduplicate concurrent loads
   227→   276→  if (loadingFonts.has(fontName)) {
   228→   277→    const result = await loadingFonts.get(fontName);
   229→   278→    if (result) return;
   230→   279→    loadingFonts.delete(fontName);
   231→   280→  }
   232→   281→
   233→   282→  const loadPromise = _doLoadFont(fontName);
   234→   283→  loadingFonts.set(fontName, loadPromise);
   235→   284→
   236→   285→  try {
   237→   286→    const success = await loadPromise;
   238→   287→    if (success) {
   239→   288→      loadedFonts.add(fontName);
   240→   289→    }
   241→   290→  } finally {
   242→   291→    loadingFonts.delete(fontName);
   243→   292→  }
   244→   293→}

   246→   295→/**
   247→   296→ * Internal font loading implementation.
   248→   297→ * Returns true if the font was loaded and verified.
   249→   298→ */
   250→   299→async function _doLoadFont(fontName: string): Promise<boolean> {
   251→   300→  const fontUrl = getFontUrl(fontName);
   252→   301→  const waitMs = _isCapacitor ? 1500 : 200;
   253→   302→
   254→   303→  for (let attempt = 1; attempt <= (_isCapacitor ? 2 : 3); attempt++) {
   255→   304→    console.log(`[Mushaf] Loading font "${fontName}" (attempt ${attempt})...`);
   256→   305→
   257→   306→    // --- Strategy 1: XHR/Fetch ArrayBuffer → FontFace (most reliable) ---
   258→   307→    try {
   259→   308→      console.log(`[Mushaf] Strategy 1: ArrayBuffer for "${fontName}" from ${fontUrl}`);
   260→   309→      const buffer = await fetchFontBinary(fontUrl);
   261→   310→      console.log(`[Mushaf] Font "${fontName}" downloaded: ${buffer.byteLength} bytes`);
   262→   311→
   263→   312→      // Validate: woff2 files should be at least 10KB
   264→   313→      if (buffer.byteLength < 10000) {
   265→   314→        console.warn(`[Mushaf] Font "${fontName}" too small (${buffer.byteLength} bytes), likely corrupted`);
   266→   315→        throw new Error(`Font data too small: ${buffer.byteLength} bytes`);
   267→   316→      }
   268→   317→
   269→   318→      const fontFace = new FontFace(fontName, buffer, { display: 'block' });
   270→   319→      const loaded = await fontFace.load();
   271→   320→      document.fonts.add(loaded);
   272→   321→
   273→   322→      await document.fonts.ready;
   274→   323→      createFontPreloadElement(fontName);
   275→   324→      await new Promise<void>((r) => setTimeout(r, waitMs));
   276→   325→
   277→   326→      const verified = verifyFontOnCanvas(fontName);
   278→   327→      console.log(`[Mushaf] Strategy 1 verify for "${fontName}": ${verified}`);
   279→   328→      if (verified) {
   280→   329→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via ArrayBuffer`);
   281→   330→        return true;
   282→   331→      }
   283→   332→
   284→   333→      // In Capacitor: trust document.fonts and proceed even if Canvas verify fails
   285→   334→      if (_isCapacitor) {
   286→   335→        const fontsCheck = document.fonts.check(`40px "${fontName}"`);
   287→   336→        if (fontsCheck) {
   288→   337→          console.log(`[Mushaf] Capacitor: Trusting document.fonts.check for "${fontName}"`);
   289→   338→          return true;
   290→   339→        }
   291→   340→        // Even if fonts.check fails, if we added the font, proceed anyway
   292→   341→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" despite verification failure`);
   293→   342→        return true;
   294→   343→      }
   295→   344→    } catch (e) {
   296→   345→      console.warn(`[Mushaf] ArrayBuffer strategy failed for "${fontName}":`, e);
   297→   346→    }
   298→   347→
   299→   348→    // --- Strategy 2: FontFace with URL (browser handles download) ---
   300→   349→    try {
   301→   350→      console.log(`[Mushaf] Strategy 2: FontFace URL for "${fontName}"`);
   302→   351→      const fontFace = new FontFace(fontName, `url('${fontUrl}')`, { display: 'block' });
   303→   352→      const loaded = await fontFace.load();
   304→   353→      document.fonts.add(loaded);
   305→   354→
   306→   355→      await document.fonts.ready;
   307→   356→      createFontPreloadElement(fontName);
   308→   357→      await new Promise<void>((r) => setTimeout(r, waitMs));
   309→   358→
   310→   359→      const verified = verifyFontOnCanvas(fontName);
   311→   360→      console.log(`[Mushaf] Strategy 2 verify for "${fontName}": ${verified}`);
   312→   361→      if (verified) {
   313→   362→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via FontFace URL`);
   314→   363→        return true;
   315→   364→      }
   316→   365→
   317→   366→      if (_isCapacitor) {
   318→   367→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" (strategy 2) despite verification failure`);
   319→   368→        return true;
   320→   369→      }
   321→   370→    } catch (e) {
   322→   371→      console.warn(`[Mushaf] FontFace URL strategy failed for "${fontName}":`, e);
   323→   372→    }
   324→   373→
   325→   374→    // --- Strategy 3: CSS @font-face + DOM preload ---
   326→   375→    try {
   327→   376→      console.log(`[Mushaf] Strategy 3: CSS @font-face for "${fontName}"`);
   328→   377→      const styleId = fontName === BSML_FONT ? 'qcf-basml' : `qcf-${fontName}`;
   329→   378→      if (!document.getElementById(styleId)) {
   330→   379→        const style = document.createElement('style');
   331→   380→        style.id = styleId;
   332→   381→        style.textContent = `
   333→   382→          @font-face {
   334→   383→            font-family: '${fontName}';
   335→   384→            src: url('${fontUrl}') format('woff2');
   336→   385→            font-display: block;
   337→   386→          }`;
   338→   387→        document.head.appendChild(style);
   339→   388→      }
   340→   389→
   341→   390→      createFontPreloadElement(fontName);
   342→   391→
   343→   392→      try {
   344→   393→        await document.fonts.load(`40px "${fontName}"`, '\uF100');
   345→   394→      } catch {
   346→   395→        /* some browsers don't support the text param */
   347→   396→      }
   348→   397→
   349→   398→      await document.fonts.ready;
   350→   399→      const cssWaitMs = _isCapacitor ? 1500 : 500;
   351→   400→      await new Promise<void>((r) => setTimeout(r, cssWaitMs));
   352→   401→
   353→   402→      const verified = verifyFontOnCanvas(fontName);
   354→   403→      console.log(`[Mushaf] Strategy 3 verify for "${fontName}": ${verified}`);
   355→   404→      if (verified) {
   356→   405→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via CSS @font-face`);
   357→   406→        return true;
   358→   407→      }
   359→   408→
   360→   409→      if (_isCapacitor) {
   361→   410→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" (strategy 3) despite verification failure`);
   362→   411→        return true;
   363→   412→      }
   364→   413→    } catch (e) {
   365→   414→      console.warn(`[Mushaf] CSS @font-face strategy failed for "${fontName}":`, e);
   366→   415→    }
   367→   416→
   368→   417→    if (attempt < (_isCapacitor ? 2 : 3)) {
   369→   418→      const delay = _isCapacitor ? 1000 : 500 * attempt;
   370→   419→      console.warn(`[Mushaf] All strategies failed for "${fontName}" (attempt ${attempt}), retrying in ${delay}ms...`);
   371→   420→      await new Promise<void>((r) => setTimeout(r, delay));
   372→   421→    }
   373→   422→  }
   374→   423→
   375→   424→  // In Capacitor: proceed anyway — better to show something than blank
   376→   425→  if (_isCapacitor) {
   377→   426→    console.warn(`[Mushaf] Capacitor: Proceeding despite font "${fontName}" loading failure — will retry render later`);
   378→   427→    return true;
   379→   428→  }
   380→   429→
   381→   430→  console.error(`[Mushaf] FAILED to load font "${fontName}" after all attempts.`);
   382→   431→  return false;
   383→   432→}

   398→   447→/**
   399→   448→ * Render a mushaf page onto a canvas element.
   400→   449→ *
   401→   450→ * For Capacitor/Android WebView, includes special handling:
   402→   451→ * - Uses XHR for font binary data (bypasses CapacitorHttp)
   403→   452→ * - Skips strict Canvas pixel verification
   404→   453→ * - Schedules a re-render after a delay for WebView font processing
   405→   454→ * - Adds an overall timeout to prevent hanging
   406→   455→ */
   407→   456→export async function renderPage(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   408→   457→  console.log(`[Mushaf] renderPage(${pageNum}) starting... Capacitor=${_isCapacitor}`);
   409→   458→
   410→   459→  // Add overall timeout for Capacitor to prevent hanging
   411→   460→  if (_isCapacitor) {
   412→   461→    const timeoutMs = 30000;
   413→   462→    return Promise.race([
   414→   463→      _renderPageInternal(pageNum, targetCanvas),
   415→   464→      new Promise<RenderPageResult>((resolve) =>
   416→   465→        setTimeout(() => {
   417→   466→          console.error(`[Mushaf] renderPage(${pageNum}) timed out after ${timeoutMs}ms`);
   418→   467→          _renderPageWithCurrentFonts(pageNum, targetCanvas).then(resolve);
   419→   468→        }, timeoutMs)
   420→   469→      )
   421→   470→    ]);
   422→   471→  }
   423→   472→
   424→   473→  return _renderPageInternal(pageNum, targetCanvas);
   425→   474→}

   427→   476→/**
   428→   477→ * Attempt to render a page using whatever fonts are currently available,
   429→   478→ * even if font loading failed. Better to show partial text than blank.
   430→   479→ */
   431→   480→async function _renderPageWithCurrentFonts(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   432→   481→  try {
   433→   482→    const data = await loadPageData(pageNum);
   434→   483→    if (!data) return { canvas: null, layout: null };
   435→   484→
   436→   485→    const pageFont = data.font || getPageFont(pageNum, null);
   437→   486→    const canvas = targetCanvas || document.createElement('canvas');
   438→   487→    canvas.width = CANVAS_W;
   439→   488→    canvas.height = CANVAS_H;
   440→   489→    const ctx = canvas.getContext('2d');
   441→   490→    if (!ctx) return { canvas: null, layout: data };
   442→   491→
   443→   492→    const colors = getColors();
   444→   493→    ctx.fillStyle = colors.bg;
   445→   494→    ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   446→   495→    renderPageContent(ctx, data, pageFont, colors);
   447→   496→    drawPageFrame(ctx, colors);
   448→   497→    drawPageNumber(ctx, pageNum, colors);
   449→   498→    return { canvas, layout: data };
   450→   499→  } catch {
   451→   500→    return { canvas: null, layout: null };
   452→   501→  }
   453→   502→}

   455→   504→async function _renderPageInternal(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   456→   505→  const data = await loadPageData(pageNum);
   457→   506→  if (!data) {
   458→   507→    console.error(`[Mushaf] Failed to load page data for page ${pageNum}`);
   459→   508→    return { canvas: null, layout: null };
   460→   509→  }
   461→   510→  console.log(`[Mushaf] Page ${pageNum} data loaded: ${data.lines?.length || 0} lines`);
   462→   511→
   463→   512→  const pageFont = data.font || getPageFont(pageNum, null);
   464→   513→
   465→   514→  // Collect ALL unique font names used on this page
   466→   515→  const pageFonts = new Set<string>();
   467→   516→  pageFonts.add(pageFont);
   468→   517→  pageFonts.add(BSML_FONT);
   469→   518→  for (const line of data.lines ?? []) {
   470→   519→    for (const w of line.words ?? []) {
   471→   520→      if (w.font) pageFonts.add(w.font);
   472→   521→    }
   473→   522→  }
   474→   523→
   475→   524→  console.log(`[Mushaf] Page ${pageNum} needs fonts: ${[...pageFonts].join(', ')}`);
   476→   525→
   477→   526→  // Load all fonts in parallel with individual timeouts
   478→   527→  const fontPromises = [...pageFonts].map((fn) =>
   479→   528→    Promise.race([
   480→   529→      ensureFontLoaded(fn),
   481→   530→      new Promise<void>((resolve) => setTimeout(resolve, _isCapacitor ? 8000 : 5000))
   482→   531→    ])
   483→   532→  );
   484→   533→  await Promise.all(fontPromises);
   485→   534→
   486→   535→  // Ensure Reem Kufi is loaded for bismillah rendering
   487→   536→  try {
   488→   537→    await document.fonts.load('500 40px "Reem Kufi"');
   489→   538→  } catch {
   490→   539→    /* ignore */
   491→   540→  }
   492→   541→
   493→   542→  // Wait for all fonts to be ready
   494→   543→  try {
   495→   544→    await document.fonts.ready;
   496→   545→  } catch {
   497→   546→    /* ignore */
   498→   547→  }
   499→   548→
   500→   549→  // In Capacitor: extra wait for WebView font processing
   501→   550→  if (_isCapacitor) {
   502→   551→    await new Promise<void>((r) => setTimeout(r, 800));
   503→   552→    await document.fonts.ready;
   504→   553→  } else {
   505→   554→    // Non-Capacitor: verify font on Canvas
   506→   555→    if (!loadedFonts.has(pageFont) || !verifyFontOnCanvas(pageFont)) {
   507→   556→      console.warn(`[Mushaf] Font "${pageFont}" not verified on Canvas, attempting final reload...`);
   508→   557→      loadedFonts.delete(pageFont);
   509→   558→      await ensureFontLoaded(pageFont);
   510→   559→    }
   511→   560→  }
   512→   561→
   513→   562→  const canvas = targetCanvas || document.createElement('canvas');
   514→   563→  canvas.width = CANVAS_W;
   515→   564→  canvas.height = CANVAS_H;
   516→   565→  const ctx = canvas.getContext('2d');
   517→   566→
   518→   567→  const colors = getColors();
   519→   568→
   520→   569→  ctx!.fillStyle = colors.bg;
   521→   570→  ctx!.fillRect(0, 0, CANVAS_W, CANVAS_H);
   522→   571→
   523→   572→  renderPageContent(ctx!, data, pageFont, colors);
   524→   573→
   525→   574→  drawPageFrame(ctx!, colors);
   526→   575→
   527→   576→  drawPageNumber(ctx!, pageNum, colors);
   528→   577→
   529→   578→  console.log(`[Mushaf] Page ${pageNum} rendered successfully`);
   530→   579→
   531→   580→  // In Capacitor: schedule MULTIPLE re-renders because Android WebView
   532→   581→  // sometimes needs several render passes for fonts to take effect on Canvas.
   533→   582→  if (_isCapacitor) {
   534→   583→    const renderData = { data, pageFont, colors, pageNum };
   535→   584→    [2500, 5000, 8000].forEach((delay) => {
   536→   585→      setTimeout(() => {
   537→   586→        console.log(`[Mushaf] Capacitor: Re-rendering page ${renderData.pageNum} after ${delay}ms delay`);
   538→   587→        try {
   539→   588→          const c = targetCanvas || document.querySelector('.mushaf-page-canvas') as HTMLCanvasElement;
   540→   589→          if (c && c.width === CANVAS_W) {
   541→   590→            const ctx2 = c.getContext('2d');
   542→   591→            if (ctx2) {
   543→   592→              ctx2.fillStyle = renderData.colors.bg;
   544→   593→              ctx2.fillRect(0, 0, CANVAS_W, CANVAS_H);
   545→   594→              renderPageContent(ctx2, renderData.data, renderData.pageFont, renderData.colors);
   546→   595→              drawPageFrame(ctx2, renderData.colors);
   547→   596→              drawPageNumber(ctx2, renderData.pageNum, renderData.colors);
   548→   597→              console.log(`[Mushaf] Capacitor: Re-render of page ${renderData.pageNum} complete at ${delay}ms`);
   549→   598→            }
   550→   599→          }
   551→   600→        } catch (e) {
   552→   601→          console.warn('[Mushaf] Capacitor re-render failed:', e);
   553→   602→        }
   554→   603→      }, delay);
   555→   604→    });
   556→   605→  }
   557→   606→
   558→   607→  return { canvas, layout: data };
   559→   608→}

   561→   610→function measureLine(ctx: CanvasRenderingContext2D, words: PageWord[], pageFont: string, fontSize: number): number[] {

   566→   615→  });

   569→   618→export function getLineY(lineIndex: number, lineCount: number, imgHeight: number): number {
   570→   619→  if (lineIndex <= 0) return 0;
   571→   620→  if (lineIndex >= lineCount) return imgHeight;
   572→   621→  const usableHeight = CANVAS_H - TOP_OFFSET - BOTTOM_OFFSET - PAD_V;
   573→   622→  const stdLineHeight = usableHeight / STD_LINES;
   574→   623→  const isShortPage = lineCount < STD_LINES;
   575→   624→  const lineSpacing = isShortPage ? (usableHeight - stdLineHeight) / Math.max(1, lineCount - 1) : stdLineHeight;
   576→   625→  const y = TOP_OFFSET + lineIndex * lineSpacing + (isShortPage ? 0 : stdLineHeight / 2);
   577→   626→  return (y / CANVAS_H) * imgHeight;
   578→   627→}

   580→   629→/** Pre-compute per-word tajweed coloring data for the entire page. */
   581→   630→function computePageTajweed(data: PageLayoutData): { wordIdx: number; lineIdx: number; color: string | null }[] | null {
   582→   631→  if (!state.tajweedEnabled) return null;
   583→   632→
   584→   633→  const verseKeyWords = new Map<string, { text: string; lineIdx: number; wordIdx: number }[]>();
   585→   634→  for (let li = 0; li < data.lines.length; li++) {
   586→   635→    const line = data.lines[li];
   587→   636→    if (!line?.words) continue;
   588→   637→    for (let wi = 0; wi < line.words.length; wi++) {
   589→   638→      const w = line.words[wi];
   590→   639→      if (w.verse_key && w.text && w.type === 'word') {
   591→   640→        if (!verseKeyWords.has(w.verse_key)) verseKeyWords.set(w.verse_key, []);
   592→   641→        verseKeyWords.get(w.verse_key)!.push({ text: w.text, lineIdx: li, wordIdx: wi });
   593→   642→      }
   594→   643→    }
   595→   644→  }
   596→   645→
   597→   646→  const result: { wordIdx: number; lineIdx: number; color: string | null }[] = [];
   598→   647→
   599→   648→  for (const [vk, words] of verseKeyWords) {
   600→   649→    const parts = vk.split(':');
   601→   650→    const surah = parseInt(parts[0], 10);
   602→   651→    const ayah = parseInt(parts[1], 10);
   603→   652→
   604→   653→    const annotations: TajweedAnnotation[] = getAyahAnnotations(surah, ayah);
   605→   654→    if (annotations.length === 0) {
   606→   655→      for (const w of words) result.push({ wordIdx: w.wordIdx, lineIdx: w.lineIdx, color: null });
   607→   656→      continue;
   608→   657→    }
   609→   658→
   610→   659→    const ayahText = words.map((w) => w.text).join(' ');
   611→   660→    const colorMap = buildColorMap(annotations);
   612→   661→    if (!colorMap || colorMap.size === 0) {
   613→   662→      for (const w of words) result.push({ wordIdx: w.wordIdx, lineIdx: w.lineIdx, color: null });
   614→   663→      continue;
   615→   664→    }
   616→   665→
   617→   666→    let outputPos = 0;
   618→   667→    for (let wi = 0; wi < words.length; wi++) {
   619→   668→      const wordText = words[wi].text;
   620→   669→      let wordColor: string | null = null;
   621→   670→      for (let ci = 0; ci < wordText.length; ci++) {
   622→   671→        const c = colorMap.get(outputPos + ci);
   623→   672→        if (c) {
   624→   673→          wordColor = c;
   625→   674→          break;
   626→   675→        }
   627→   676→      }
   628→   677→      result.push({ wordIdx: words[wi].wordIdx, lineIdx: words[wi].lineIdx, color: wordColor });
   629→   678→      outputPos += wordText.length + 1;
   630→   679→    }
   631→   680→  }
   632→   681→
   633→   682→  return result;
   634→   683→}

   636→   685→function renderPageContent(
   637→   686→  ctx: CanvasRenderingContext2D,
   638→   687→  data: PageLayoutData,
   639→   688→  pageFont: string,
   640→   689→  colors: PageColors
   641→   690→): void {
   642→   691→  const lines = data.lines;
   643→   692→  if (!lines || lines.length === 0) return;
   644→   693→
   645→   694→  const lineCount = lines.length;
   646→   695→  const usableHeight = CANVAS_H - TOP_OFFSET - BOTTOM_OFFSET - PAD_V;
   647→   696→  const availableW = CANVAS_W - PAD_H * 2;
   648→   697→
   649→   698→  const stdLineHeight = usableHeight / STD_LINES;
   650→   699→  const baseFontSize = Math.max(26, Math.min(55, stdLineHeight * 0.85));
   651→   700→  const pageFontSize = baseFontSize;
   652→   701→
   653→   702→  const isShortPage = lineCount < 15;
   654→   703→  const lineSpacing = isShortPage ? (usableHeight - stdLineHeight) / Math.max(1, lineCount - 1) : stdLineHeight;
   655→   704→
   656→   705→  const lineWidths: (LineWidths | null)[] = [];
   657→   706→  for (let i = 0; i < lineCount; i++) {
   658→   707→    const line = lines[i];
   659→   708→    if (!line?.words || line.words.length === 0) {
   660→   709→      lineWidths.push(null);
   661→   710→      continue;
   662→   711→    }
   663→   712→    const widths = measureLine(ctx, line.words, pageFont, pageFontSize);
   664→   713→    const totalW = widths.reduce((a: number, b: number) => a + b, 0);
   665→   714→    const gap = line.words.length > 1 ? (availableW - totalW) / (line.words.length - 1) : 0;
   666→   715→    lineWidths.push({ widths, gap: Math.max(0, gap) });
   667→   716→  }
   668→   717→
   669→   718→  const tajweedColors = computePageTajweed(data);
   670→   719→  const tajweedLookup = new Map<string, string | null>();
   671→   720→  if (tajweedColors) {
   672→   721→    for (const entry of tajweedColors) {
   673→   722→      tajweedLookup.set(`${entry.lineIdx}:${entry.wordIdx}`, entry.color);
   674→   723→    }
   675→   724→  }
   676→   725→
   677→   726→  const bsmlColor = isNightMode() ? '#d4af37' : '#8b6914';
   678→   727→  const headerColor = isNightMode() ? '#c4a87c' : '#7a5c3a';
   679→   728→  const bsmlFont = 'Reem Kufi';
   680→   729→  const bsmlText = 'بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ';
   681→   730→
   682→   731→  ctx.textBaseline = 'middle';
   683→   732→
   684→   733→  for (let i = 0; i < lineCount; i++) {
   685→   734→    const line = lines[i];
   686→   735→    if (!line?.words || line.words.length === 0) continue;
   687→   736→
   688→   737→    const y = TOP_OFFSET + i * lineSpacing + (isShortPage ? 0 : stdLineHeight / 2);
   689→   738→    const words = line.words;
   690→   739→    const lw = lineWidths[i];
   691→   740→    if (!lw) continue;
   692→   741→    const { widths, gap } = lw;
   693→   742→
   694→   743→    const lineType = words[0]?.type;
   695→   744→    const isBismillahLine = lineType === 'bismillah';
   696→   745→    const isHeaderLine = lineType === 'surah_header';
   697→   746→
   698→   747→    if (isBismillahLine) {
   699→   748→      const bsmlFontSize = Math.round(pageFontSize * 0.95);
   700→   749→      ctx.font = `500 ${bsmlFontSize}px "${bsmlFont}", "Scheherazade New", serif`;
   701→   750→      ctx.textAlign = 'center';
   702→   751→      ctx.fillStyle = bsmlColor;
   703→   752→      ctx.fillText(bsmlText, CANVAS_W / 2, y);
   704→   753→      continue;
   705→   754→    }
   706→   755→
   707→   756→    let x = CANVAS_W - PAD_H;
   708→   757→
   709→   758→    for (let j = 0; j < words.length; j++) {
   710→   759→      const w = words[j];
   711→   760→      const fn = w.font || pageFont;
   712→   761→      ctx.font = `${pageFontSize}px "${fn}"`;
   713→   762→      ctx.textAlign = 'right';
   714→   763→
   715→   764→      if (isHeaderLine) {
   716→   765→        ctx.fillStyle = headerColor;
   717→   766→      } else {
   718→   767→        const wordColor = tajweedLookup ? tajweedLookup.get(`${i}:${j}`) : null;
   719→   768→        ctx.fillStyle = wordColor || colors.txt;
   720→   769→      }
   721→   770→
   722→   771→      ctx.fillText(w.char, x, y);
   723→   772→      x -= widths[j] + gap;
   724→   773→    }
   725→   774→  }
   726→   775→}
