   110→   110→export async function loadPageData(pageNum: number): Promise<PageLayoutData | null> {
   111→   111→  const key = `qcf4-${pageNum}`;
   112→   112→  if (layoutCache.has(key)) return layoutCache.get(key)!;
   113→   113→  const padded = String(pageNum).padStart(3, '0');
   114→   114→  try {
   115→   115→    const res = await fetch(`${PAGE_BASE}${padded}.json`);
   116→   116→    if (!res.ok) return null;
   117→   117→    const data = (await res.json()) as PageLayoutData;
   118→   118→    // Evict oldest entry if cache is full (LRU policy)
   119→   119→    if (layoutCache.size >= LAYOUT_CACHE_MAX) {
   120→   120→      const firstKey = layoutCache.keys().next().value;
   121→   121→      if (firstKey !== undefined) layoutCache.delete(firstKey);
   122→   122→    }
   123→   123→    layoutCache.set(key, data);
   124→   124→    return data;
   125→   125→  } catch (e: unknown) {
   126→   126→    console.warn('[Mushaf] Failed to load page data:', e);
   127→   127→    return null;
   128→   128→  }
   129→   129→}

   142→   142→/**
   143→   143→ * Fetch font binary data using XMLHttpRequest instead of fetch().
   144→   144→ *
   145→   145→ * CRITICAL: In Capacitor with CapacitorHttp enabled, the native HTTP plugin
   146→   146→ * intercepts ALL fetch() calls and can corrupt binary data (woff2 fonts).
   147→   147→ * XMLHttpRequest is NOT intercepted by CapacitorHttp, so it returns
   148→   148→ * correct binary data. This is the key fix for mushaf mode in Android APK.
   149→   149→ */
   150→   150→function fetchFontViaXHR(url: string): Promise<ArrayBuffer> {
   151→   151→  return new Promise((resolve, reject) => {
   152→   152→    const xhr = new XMLHttpRequest();
   153→   153→    xhr.open('GET', url, true);
   154→   154→    xhr.responseType = 'arraybuffer';
   155→   155→    xhr.onload = () => {
   156→   156→      if (xhr.status === 200) {
   157→   157→        resolve(xhr.response as ArrayBuffer);
   158→   158→      } else {
   159→   159→        reject(new Error(`XHR HTTP ${xhr.status}`));
   160→   160→      }
   161→   161→    };
   162→   162→    xhr.onerror = () => reject(new Error('XHR network error'));
   163→   163→    xhr.ontimeout = () => reject(new Error('XHR timeout'));
   164→   164→    xhr.timeout = 15000; // 15 second timeout
   165→   165→    xhr.send();
   166→   166→  });
   167→   167→}

   169→   169→/**
   170→   170→ * Fetch font binary data. In Capacitor, uses XHR to avoid CapacitorHttp
   171→   171→ * interception. In browsers, uses standard fetch().
   172→   172→ */
   173→   173→async function fetchFontBinary(url: string): Promise<ArrayBuffer> {
   174→   174→  if (_isCapacitor) {
   175→   175→    // Use XHR in Capacitor to avoid CapacitorHttp interception
   176→   176→    console.log(`[Mushaf] Using XHR for font binary (bypassing CapacitorHttp): ${url}`);
   177→   177→    return fetchFontViaXHR(url);
   178→   178→  }
   179→   179→  // Use standard fetch in browser
   180→   180→  const res = await fetch(url);
   181→   181→  if (!res.ok) throw new Error(`HTTP ${res.status}`);
   182→   182→  return res.arrayBuffer();
   183→   183→}

   185→   185→/**
   186→   186→ * Verify a font is truly usable on Canvas by drawing a PUA character
   187→   187→ * and checking if the pixels are non-blank.
   188→   188→ *
   189→   189→ * IMPORTANT: In Android WebView (Capacitor), Canvas pixel verification is unreliable
   190→   190→ * because WebView processes fonts differently. We use document.fonts.check()
   191→   191→ * as a fallback verification method in that case.
   192→   192→ */
   193→   193→function verifyFontOnCanvas(fontName: string): boolean {
   194→   194→  try {
   195→   195→    // In Capacitor/Android WebView, skip pixel-level Canvas verification
   196→   196→    // because it's unreliable — trust document.fonts API instead
   197→   197→    if (_isCapacitor) {
   198→   198→      const fontsCheck = document.fonts.check(`40px "${fontName}"`);
   199→   199→      console.log(`[Mushaf] Capacitor: document.fonts.check for "${fontName}": ${fontsCheck}`);
   200→   200→      return fontsCheck;
   201→   201→    }
   202→   202→
   203→   203→    const testSize = 40;
   204→   204→    const canvas = document.createElement('canvas');
   205→   205→    canvas.width = testSize * 2;
   206→   206→    canvas.height = testSize * 2;
   207→   207→    const ctx = canvas.getContext('2d');
   208→   208→    if (!ctx) return false;
   209→   209→
   210→   210→    ctx.fillStyle = '#FFFFFF';
   211→   211→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   212→   212→
   213→   213→    ctx.font = `${testSize}px "${fontName}"`;
   214→   214→    ctx.fillStyle = '#000000';
   215→   215→    ctx.textAlign = 'center';
   216→   216→    ctx.textBaseline = 'middle';
   217→   217→    ctx.fillText('\uF100', canvas.width / 2, canvas.height / 2);
   218→   218→
   219→   219→    const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
   220→   220→    const data = imageData.data;
   221→   221→    for (let i = 0; i < data.length; i += 4) {
   222→   222→      if (data[i] < 250 || data[i + 1] < 250 || data[i + 2] < 250) {
   223→   223→        return true;
   224→   224→      }
   225→   225→    }
   226→   226→
   227→   227→    return false;
   228→   228→  } catch {
   229→   229→    return false;
   230→   230→  }
   231→   231→}
   232→   232→
   233→   233→/**
   234→   234→ * Create a hidden DOM element that uses the font. This forces the browser
   235→   235→ * to download and parse the font binary, making it available for Canvas.
   236→   236→ * Some browsers (especially Android WebView) won't make fonts available
   237→   237→ * to Canvas unless they're used in the DOM first.
   238→   238→ */
   239→   239→function createFontPreloadElement(fontName: string): void {
   240→   240→  const id = `qcf-preload-${fontName}`;
   241→   241→  if (document.getElementById(id)) return;
   242→   242→
   243→   243→  const span = document.createElement('span');
   244→   244→  span.id = id;
   245→   245→  span.setAttribute('aria-hidden', 'true');
   246→   246→  span.style.cssText = `
   247→   247→    position: absolute;
   248→   248→    left: -9999px;
   249→   249→    top: -9999px;
   250→   250→    visibility: hidden;
   251→   251→    font-family: "${fontName}";
   252→   252→    font-size: 40px;
   253→   253→    pointer-events: none;
   254→   254→  `;
   255→   255→  // Use multiple PUA characters to ensure the browser loads the font
   256→   256→  span.textContent = '\uF100\uF101\uF102\uF103\uF104\uF105';
   257→   257→  document.body.appendChild(span);
   258→   258→}

   260→   260→/**
   261→   261→ * Load a QCF4 font using the most reliable method available.
   262→   262→ *
   263→   263→ * Strategy order (most to least reliable):
   264→   264→ * 1. XHR fetch as ArrayBuffer → FontFace from binary data
   265→   265→ *    (Bypasses CapacitorHttp which corrupts binary data)
   266→   266→ * 2. FontFace API with URL (browser handles download)
   267→   267→ * 3. CSS @font-face injection with DOM preload element
   268→   268→ *
   269→   269→ * In Capacitor: We skip pixel-level Canvas verification and trust
   270→   270→ * document.fonts API instead.
   271→   271→ */
   272→   272→async function ensureFontLoaded(fontName: string): Promise<void> {
   273→   273→  if (loadedFonts.has(fontName)) return;
   274→   274→
   275→   275→  // Deduplicate concurrent loads
   276→   276→  if (loadingFonts.has(fontName)) {
   277→   277→    const result = await loadingFonts.get(fontName);
   278→   278→    if (result) return;
   279→   279→    loadingFonts.delete(fontName);
   280→   280→  }
   281→   281→
   282→   282→  const loadPromise = _doLoadFont(fontName);
   283→   283→  loadingFonts.set(fontName, loadPromise);
   284→   284→
   285→   285→  try {
   286→   286→    const success = await loadPromise;
   287→   287→    if (success) {
   288→   288→      loadedFonts.add(fontName);
   289→   289→    }
   290→   290→  } finally {
   291→   291→    loadingFonts.delete(fontName);
   292→   292→  }
   293→   293→}

   295→   295→/**
   296→   296→ * Internal font loading implementation.
   297→   297→ * Returns true if the font was loaded and verified.
   298→   298→ */
   299→   299→async function _doLoadFont(fontName: string): Promise<boolean> {
   300→   300→  const fontUrl = getFontUrl(fontName);
   301→   301→  const waitMs = _isCapacitor ? 1500 : 200;
   302→   302→
   303→   303→  for (let attempt = 1; attempt <= (_isCapacitor ? 2 : 3); attempt++) {
   304→   304→    console.log(`[Mushaf] Loading font "${fontName}" (attempt ${attempt})...`);
   305→   305→
   306→   306→    // --- Strategy 1: XHR/Fetch ArrayBuffer → FontFace (most reliable) ---
   307→   307→    try {
   308→   308→      console.log(`[Mushaf] Strategy 1: ArrayBuffer for "${fontName}" from ${fontUrl}`);
   309→   309→      const buffer = await fetchFontBinary(fontUrl);
   310→   310→      console.log(`[Mushaf] Font "${fontName}" downloaded: ${buffer.byteLength} bytes`);
   311→   311→
   312→   312→      // Validate: woff2 files should be at least 10KB
   313→   313→      if (buffer.byteLength < 10000) {
   314→   314→        console.warn(`[Mushaf] Font "${fontName}" too small (${buffer.byteLength} bytes), likely corrupted`);
   315→   315→        throw new Error(`Font data too small: ${buffer.byteLength} bytes`);
   316→   316→      }
   317→   317→
   318→   318→      const fontFace = new FontFace(fontName, buffer, { display: 'block' });
   319→   319→      const loaded = await fontFace.load();
   320→   320→      document.fonts.add(loaded);
   321→   321→
   322→   322→      await document.fonts.ready;
   323→   323→      createFontPreloadElement(fontName);
   324→   324→      await new Promise<void>((r) => setTimeout(r, waitMs));
   325→   325→
   326→   326→      const verified = verifyFontOnCanvas(fontName);
   327→   327→      console.log(`[Mushaf] Strategy 1 verify for "${fontName}": ${verified}`);
   328→   328→      if (verified) {
   329→   329→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via ArrayBuffer`);
   330→   330→        return true;
   331→   331→      }
   332→   332→
   333→   333→      // In Capacitor: trust document.fonts and proceed even if Canvas verify fails
   334→   334→      if (_isCapacitor) {
   335→   335→        const fontsCheck = document.fonts.check(`40px "${fontName}"`);
   336→   336→        if (fontsCheck) {
   337→   337→          console.log(`[Mushaf] Capacitor: Trusting document.fonts.check for "${fontName}"`);
   338→   338→          return true;
   339→   339→        }
   340→   340→        // Even if fonts.check fails, if we added the font, proceed anyway
   341→   341→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" despite verification failure`);
   342→   342→        return true;
   343→   343→      }
   344→   344→    } catch (e) {
   345→   345→      console.warn(`[Mushaf] ArrayBuffer strategy failed for "${fontName}":`, e);
   346→   346→    }
   347→   347→
   348→   348→    // --- Strategy 2: FontFace with URL (browser handles download) ---
   349→   349→    try {
   350→   350→      console.log(`[Mushaf] Strategy 2: FontFace URL for "${fontName}"`);
   351→   351→      const fontFace = new FontFace(fontName, `url('${fontUrl}')`, { display: 'block' });
   352→   352→      const loaded = await fontFace.load();
   353→   353→      document.fonts.add(loaded);
   354→   354→
   355→   355→      await document.fonts.ready;
   356→   356→      createFontPreloadElement(fontName);
   357→   357→      await new Promise<void>((r) => setTimeout(r, waitMs));
   358→   358→
   359→   359→      const verified = verifyFontOnCanvas(fontName);
   360→   360→      console.log(`[Mushaf] Strategy 2 verify for "${fontName}": ${verified}`);
   361→   361→      if (verified) {
   362→   362→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via FontFace URL`);
   363→   363→        return true;
   364→   364→      }
   365→   365→
   366→   366→      if (_isCapacitor) {
   367→   367→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" (strategy 2) despite verification failure`);
   368→   368→        return true;
   369→   369→      }
   370→   370→    } catch (e) {
   371→   371→      console.warn(`[Mushaf] FontFace URL strategy failed for "${fontName}":`, e);
   372→   372→    }
   373→   373→
   374→   374→    // --- Strategy 3: CSS @font-face + DOM preload ---
   375→   375→    try {
   376→   376→      console.log(`[Mushaf] Strategy 3: CSS @font-face for "${fontName}"`);
   377→   377→      const styleId = fontName === BSML_FONT ? 'qcf-basml' : `qcf-${fontName}`;
   378→   378→      if (!document.getElementById(styleId)) {
   379→   379→        const style = document.createElement('style');
   380→   380→        style.id = styleId;
   381→   381→        style.textContent = `
   382→   382→          @font-face {
   383→   383→            font-family: '${fontName}';
   384→   384→            src: url('${fontUrl}') format('woff2');
   385→   385→            font-display: block;
   386→   386→          }`;
   387→   387→        document.head.appendChild(style);
   388→   388→      }
   389→   389→
   390→   390→      createFontPreloadElement(fontName);
   391→   391→
   392→   392→      try {
   393→   393→        await document.fonts.load(`40px "${fontName}"`, '\uF100');
   394→   394→      } catch {
   395→   395→        /* some browsers don't support the text param */
   396→   396→      }
   397→   397→
   398→   398→      await document.fonts.ready;
   399→   399→      const cssWaitMs = _isCapacitor ? 1500 : 500;
   400→   400→      await new Promise<void>((r) => setTimeout(r, cssWaitMs));
   401→   401→
   402→   402→      const verified = verifyFontOnCanvas(fontName);
   403→   403→      console.log(`[Mushaf] Strategy 3 verify for "${fontName}": ${verified}`);
   404→   404→      if (verified) {
   405→   405→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via CSS @font-face`);
   406→   406→        return true;
   407→   407→      }
   408→   408→
   409→   409→      if (_isCapacitor) {
   410→   410→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" (strategy 3) despite verification failure`);
   411→   411→        return true;
   412→   412→      }
   413→   413→    } catch (e) {
   414→   414→      console.warn(`[Mushaf] CSS @font-face strategy failed for "${fontName}":`, e);
   415→   415→    }
   416→   416→
   417→   417→    if (attempt < (_isCapacitor ? 2 : 3)) {
   418→   418→      const delay = _isCapacitor ? 1000 : 500 * attempt;
   419→   419→      console.warn(`[Mushaf] All strategies failed for "${fontName}" (attempt ${attempt}), retrying in ${delay}ms...`);
   420→   420→      await new Promise<void>((r) => setTimeout(r, delay));
   421→   421→    }
   422→   422→  }
   423→   423→
   424→   424→  // In Capacitor: proceed anyway — better to show something than blank
   425→   425→  if (_isCapacitor) {
   426→   426→    console.warn(`[Mushaf] Capacitor: Proceeding despite font "${fontName}" loading failure — will retry render later`);
   427→   427→    return true;
   428→   428→  }
   429→   429→
   430→   430→  console.error(`[Mushaf] FAILED to load font "${fontName}" after all attempts.`);
   431→   431→  return false;
   432→   432→}

   445→   445→/* ===================== RENDERING ===================== */
   446→   446→
   447→   447→/**
   448→   448→ * Render a mushaf page onto a canvas element.
   449→   449→ *
   450→   450→ * For Capacitor/Android WebView, includes special handling:
   451→   451→ * - Uses XHR for font binary data (bypasses CapacitorHttp)
   452→   452→ * - Skips strict Canvas pixel verification
   453→   453→ * - Schedules a re-render after a delay for WebView font processing
   454→   454→ * - Adds an overall timeout to prevent hanging
   455→   455→ */
   456→   456→export async function renderPage(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   457→   457→  console.log(`[Mushaf] renderPage(${pageNum}) starting... Capacitor=${_isCapacitor}`);
   458→   458→
   459→   459→  // Add overall timeout for Capacitor to prevent hanging
   460→   460→  if (_isCapacitor) {
   461→   461→    const timeoutMs = 30000;
   462→   462→    return Promise.race([
   463→   463→      _renderPageInternal(pageNum, targetCanvas),
   464→   464→      new Promise<RenderPageResult>((resolve) =>
   465→   465→        setTimeout(() => {
   466→   466→          console.error(`[Mushaf] renderPage(${pageNum}) timed out after ${timeoutMs}ms`);
   467→   467→          _renderPageWithCurrentFonts(pageNum, targetCanvas).then(resolve);
   468→   468→        }, timeoutMs)
   469→   469→      )
   470→   470→    ]);
   471→   471→  }
   472→   472→
   473→   473→  return _renderPageInternal(pageNum, targetCanvas);
   474→   474→}

   495→   495→    renderPageContent(ctx, data, pageFont, colors);
   496→   496→    drawPageFrame(ctx, colors);
   497→   497→    drawPageNumber(ctx, pageNum, colors);

   500→   500→    return { canvas: null, layout: null };
   501→   501→  }

   504→   504→async function _renderPageInternal(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   505→   505→  const data = await loadPageData(pageNum);
   506→   506→  if (!data) {
   507→   507→    console.error(`[Mushaf] Failed to load page data for page ${pageNum}`);
   508→   508→    return { canvas: null, layout: null };
   509→   509→  }
   510→   510→  console.log(`[Mushaf] Page ${pageNum} data loaded: ${data.lines?.length || 0} lines`);
   511→   511→
   512→   512→  const pageFont = data.font || getPageFont(pageNum, null);
   513→   513→
   514→   514→  // Collect ALL unique font names used on this page
   515→   515→  const pageFonts = new Set<string>();
   516→   516→  pageFonts.add(pageFont);
   517→   517→  pageFonts.add(BSML_FONT);
   518→   518→  for (const line of data.lines ?? []) {
   519→   519→    for (const w of line.words ?? []) {
   520→   520→      if (w.font) pageFonts.add(w.font);
   521→   521→    }
   522→   522→  }
   523→   523→
   524→   524→  console.log(`[Mushaf] Page ${pageNum} needs fonts: ${[...pageFonts].join(', ')}`);
   525→   525→
   526→   526→  // Load all fonts in parallel with individual timeouts
   527→   527→  const fontPromises = [...pageFonts].map((fn) =>
   528→   528→    Promise.race([
   529→   529→      ensureFontLoaded(fn),
   530→   530→      new Promise<void>((resolve) => setTimeout(resolve, _isCapacitor ? 8000 : 5000))
   531→   531→    ])
   532→   532→  );
   533→   533→  await Promise.all(fontPromises);
   534→   534→
   535→   535→  // Ensure Reem Kufi is loaded for bismillah rendering
   536→   536→  try {
   537→   537→    await document.fonts.load('500 40px "Reem Kufi"');
   538→   538→  } catch {
   539→   539→    /* ignore */
   540→   540→  }
   541→   541→
   542→   542→  // Wait for all fonts to be ready
   543→   543→  try {
   544→   544→    await document.fonts.ready;
   545→   545→  } catch {
   546→   546→    /* ignore */
   547→   547→  }
   548→   548→
   549→   549→  // In Capacitor: extra wait for WebView font processing
   550→   550→  if (_isCapacitor) {
   551→   551→    await new Promise<void>((r) => setTimeout(r, 800));
   552→   552→    await document.fonts.ready;
   553→   553→  } else {
   554→   554→    // Non-Capacitor: verify font on Canvas
   555→   555→    if (!loadedFonts.has(pageFont) || !verifyFontOnCanvas(pageFont)) {
   556→   556→      console.warn(`[Mushaf] Font "${pageFont}" not verified on Canvas, attempting final reload...`);
   557→   557→      loadedFonts.delete(pageFont);
   558→   558→      await ensureFontLoaded(pageFont);
   559→   559→    }
   560→   560→  }
   561→   561→
   562→   562→  const canvas = targetCanvas || document.createElement('canvas');
   563→   563→  canvas.width = CANVAS_W;
   564→   564→  canvas.height = CANVAS_H;
   565→   565→  const ctx = canvas.getContext('2d');
   566→   566→
   567→   567→  const colors = getColors();
   568→   568→
   569→   569→  ctx!.fillStyle = colors.bg;
   570→   570→  ctx!.fillRect(0, 0, CANVAS_W, CANVAS_H);
   571→   571→
   572→   572→  renderPageContent(ctx!, data, pageFont, colors);
   573→   573→
   574→   574→  drawPageFrame(ctx!, colors);
   575→   575→
   576→   576→  drawPageNumber(ctx!, pageNum, colors);
   577→   577→
   578→   578→  console.log(`[Mushaf] Page ${pageNum} rendered successfully`);
   579→   579→
   580→   580→  // In Capacitor: schedule MULTIPLE re-renders because Android WebView
   581→   581→  // sometimes needs several render passes for fonts to take effect on Canvas.
   582→   582→  if (_isCapacitor) {
   583→   583→    const renderData = { data, pageFont, colors, pageNum };
   584→   584→    [2500, 5000, 8000].forEach((delay) => {
   585→   585→      setTimeout(() => {
   586→   586→        console.log(`[Mushaf] Capacitor: Re-rendering page ${renderData.pageNum} after ${delay}ms delay`);
   587→   587→        try {
   588→   588→          const c = targetCanvas || document.querySelector('.mushaf-page-canvas') as HTMLCanvasElement;
   589→   589→          if (c && c.width === CANVAS_W) {
   590→   590→            const ctx2 = c.getContext('2d');
   591→   591→            if (ctx2) {
   592→   592→              ctx2.fillStyle = renderData.colors.bg;
   593→   593→              ctx2.fillRect(0, 0, CANVAS_W, CANVAS_H);
   594→   594→              renderPageContent(ctx2, renderData.data, renderData.pageFont, renderData.colors);
   595→   595→              drawPageFrame(ctx2, renderData.colors);
   596→   596→              drawPageNumber(ctx2, renderData.pageNum, renderData.colors);
   597→   597→              console.log(`[Mushaf] Capacitor: Re-render of page ${renderData.pageNum} complete at ${delay}ms`);
   598→   598→            }
   599→   599→          }
   600→   600→        } catch (e) {
   601→   601→          console.warn('[Mushaf] Capacitor re-render failed:', e);
   602→   602→        }
   603→   603→      }, delay);
   604→   604→    });
   605→   605→  }
   606→   606→
   607→   607→  return { canvas, layout: data };
   608→   608→}

   610→   610→function measureLine(ctx: CanvasRenderingContext2D, words: PageWord[], pageFont: string, fontSize: number): number[] {
   611→   611→  return words.map((w) => {
   612→   612→    const fn = w.font || pageFont;
   613→   613→    ctx.font = `${fontSize}px "${fn}"`;
   614→   614→    return ctx.measureText(w.char).width;
   615→   615→  });

   618→   618→export function getLineY(lineIndex: number, lineCount: number, imgHeight: number): number {
   619→   619→  if (lineIndex <= 0) return 0;
   620→   620→  if (lineIndex >= lineCount) return imgHeight;
   621→   621→  const usableHeight = CANVAS_H - TOP_OFFSET - BOTTOM_OFFSET - PAD_V;
   622→   622→  const stdLineHeight = usableHeight / STD_LINES;
   623→   623→  const isShortPage = lineCount < STD_LINES;
   624→   624→  const lineSpacing = isShortPage ? (usableHeight - stdLineHeight) / Math.max(1, lineCount - 1) : stdLineHeight;
   625→   625→  const y = TOP_OFFSET + lineIndex * lineSpacing + (isShortPage ? 0 : stdLineHeight / 2);
   626→   626→  return (y / CANVAS_H) * imgHeight;
   627→   627→}

   629→   629→/** Pre-compute per-word tajweed coloring data for the entire page. */
   630→   630→function computePageTajweed(data: PageLayoutData): { wordIdx: number; lineIdx: number; color: string | null }[] | null {
   631→   631→  if (!state.tajweedEnabled) return null;
   632→   632→
   633→   633→  const verseKeyWords = new Map<string, { text: string; lineIdx: number; wordIdx: number }[]>();
   634→   634→  for (let li = 0; li < data.lines.length; li++) {
   635→   635→    const line = data.lines[li];
   636→   636→    if (!line?.words) continue;
   637→   637→    for (let wi = 0; wi < line.words.length; wi++) {
   638→   638→      const w = line.words[wi];
   639→   639→      if (w.verse_key && w.text && w.type === 'word') {
   640→   640→        if (!verseKeyWords.has(w.verse_key)) verseKeyWords.set(w.verse_key, []);
   641→   641→        verseKeyWords.get(w.verse_key)!.push({ text: w.text, lineIdx: li, wordIdx: wi });
   642→   642→      }
   643→   643→    }
   644→   644→  }
   645→   645→
   646→   646→  const result: { wordIdx: number; lineIdx: number; color: string | null }[] = [];
   647→   647→
   648→   648→  for (const [vk, words] of verseKeyWords) {
   649→   649→    const parts = vk.split(':');
   650→   650→    const surah = parseInt(parts[0], 10);
   651→   651→    const ayah = parseInt(parts[1], 10);
   652→   652→
   653→   653→    const annotations: TajweedAnnotation[] = getAyahAnnotations(surah, ayah);
   654→   654→    if (annotations.length === 0) {
   655→   655→      for (const w of words) result.push({ wordIdx: w.wordIdx, lineIdx: w.lineIdx, color: null });
   656→   656→      continue;
   657→   657→    }
   658→   658→
   659→   659→    const ayahText = words.map((w) => w.text).join(' ');
   660→   660→    const colorMap = buildColorMap(annotations);
   661→   661→    if (!colorMap || colorMap.size === 0) {
   662→   662→      for (const w of words) result.push({ wordIdx: w.wordIdx, lineIdx: w.lineIdx, color: null });
   663→   663→      continue;
   664→   664→    }
   665→   665→
   666→   666→    let outputPos = 0;
   667→   667→    for (let wi = 0; wi < words.length; wi++) {
   668→   668→      const wordText = words[wi].text;
   669→   669→      let wordColor: string | null = null;
   670→   670→      for (let ci = 0; ci < wordText.length; ci++) {
   671→   671→        const c = colorMap.get(outputPos + ci);
   672→   672→        if (c) {
   673→   673→          wordColor = c;
   674→   674→          break;
   675→   675→        }
   676→   676→      }
   677→   677→      result.push({ wordIdx: words[wi].wordIdx, lineIdx: words[wi].lineIdx, color: wordColor });
   678→   678→      outputPos += wordText.length + 1;
   679→   679→    }
   680→   680→  }
   681→   681→
   682→   682→  return result;
   683→   683→}

   685→   685→function renderPageContent(
   686→   686→  ctx: CanvasRenderingContext2D,
   687→   687→  data: PageLayoutData,
   688→   688→  pageFont: string,
   689→   689→  colors: PageColors
   690→   690→): void {
   691→   691→  const lines = data.lines;
   692→   692→  if (!lines || lines.length === 0) return;
   693→   693→
   694→   694→  const lineCount = lines.length;
   695→   695→  const usableHeight = CANVAS_H - TOP_OFFSET - BOTTOM_OFFSET - PAD_V;
   696→   696→  const availableW = CANVAS_W - PAD_H * 2;
   697→   697→
   698→   698→  const stdLineHeight = usableHeight / STD_LINES;
   699→   699→  const baseFontSize = Math.max(26, Math.min(55, stdLineHeight * 0.85));
   700→   700→  const pageFontSize = baseFontSize;
   701→   701→
   702→   702→  const isShortPage = lineCount < 15;
   703→   703→  const lineSpacing = isShortPage ? (usableHeight - stdLineHeight) / Math.max(1, lineCount - 1) : stdLineHeight;
   704→   704→
   705→   705→  const lineWidths: (LineWidths | null)[] = [];
   706→   706→  for (let i = 0; i < lineCount; i++) {
   707→   707→    const line = lines[i];
   708→   708→    if (!line?.words || line.words.length === 0) {
   709→   709→      lineWidths.push(null);
   710→   710→      continue;
   711→   711→    }
   712→   712→    const widths = measureLine(ctx, line.words, pageFont, pageFontSize);
   713→   713→    const totalW = widths.reduce((a: number, b: number) => a + b, 0);
   714→   714→    const gap = line.words.length > 1 ? (availableW - totalW) / (line.words.length - 1) : 0;
   715→   715→    lineWidths.push({ widths, gap: Math.max(0, gap) });
   716→   716→  }
   717→   717→
   718→   718→  const tajweedColors = computePageTajweed(data);
   719→   719→  const tajweedLookup = new Map<string, string | null>();
   720→   720→  if (tajweedColors) {
   721→   721→    for (const entry of tajweedColors) {
   722→   722→      tajweedLookup.set(`${entry.lineIdx}:${entry.wordIdx}`, entry.color);
   723→   723→    }
   724→   724→  }
   725→   725→
   726→   726→  const bsmlColor = isNightMode() ? '#d4af37' : '#8b6914';
   727→   727→  const headerColor = isNightMode() ? '#c4a87c' : '#7a5c3a';
   728→   728→  const bsmlFont = 'Reem Kufi';
   729→   729→  const bsmlText = 'بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ';
   730→   730→
   731→   731→  ctx.textBaseline = 'middle';
   732→   732→
   733→   733→  for (let i = 0; i < lineCount; i++) {
   734→   734→    const line = lines[i];
   735→   735→    if (!line?.words || line.words.length === 0) continue;
   736→   736→
   737→   737→    const y = TOP_OFFSET + i * lineSpacing + (isShortPage ? 0 : stdLineHeight / 2);
   738→   738→    const words = line.words;
   739→   739→    const lw = lineWidths[i];
   740→   740→    if (!lw) continue;
   741→   741→    const { widths, gap } = lw;
   742→   742→
   743→   743→    const lineType = words[0]?.type;
   744→   744→    const isBismillahLine = lineType === 'bismillah';
   745→   745→    const isHeaderLine = lineType === 'surah_header';
   746→   746→
   747→   747→    if (isBismillahLine) {
   748→   748→      const bsmlFontSize = Math.round(pageFontSize * 0.95);
   749→   749→      ctx.font = `500 ${bsmlFontSize}px "${bsmlFont}", "Scheherazade New", serif`;
   750→   750→      ctx.textAlign = 'center';
   751→   751→      ctx.fillStyle = bsmlColor;
   752→   752→      ctx.fillText(bsmlText, CANVAS_W / 2, y);
   753→   753→      continue;
   754→   754→    }
   755→   755→
   756→   756→    let x = CANVAS_W - PAD_H;
   757→   757→
   758→   758→    for (let j = 0; j < words.length; j++) {
   759→   759→      const w = words[j];
   760→   760→      const fn = w.font || pageFont;
   761→   761→      ctx.font = `${pageFontSize}px "${fn}"`;
   762→   762→      ctx.textAlign = 'right';
   763→   763→
   764→   764→      if (isHeaderLine) {
   765→   765→        ctx.fillStyle = headerColor;
   766→   766→      } else {
   767→   767→        const wordColor = tajweedLookup ? tajweedLookup.get(`${i}:${j}`) : null;
   768→   768→        ctx.fillStyle = wordColor || colors.txt;
   769→   769→      }
   770→   770→
   771→   771→      ctx.fillText(w.char, x, y);
   772→   772→      x -= widths[j] + gap;
   773→   773→    }
   774→   774→  }
   775→   775→}
