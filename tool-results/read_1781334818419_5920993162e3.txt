   120→   120→export async function loadPageData(pageNum: number): Promise<PageLayoutData | null> {
   121→   121→  const key = `qcf4-${pageNum}`;
   122→   122→  if (layoutCache.has(key)) return layoutCache.get(key)!;
   123→   123→  const padded = String(pageNum).padStart(3, '0');
   124→   124→  try {
   125→   125→    const res = await fetch(`${PAGE_BASE}${padded}.json`);
   126→   126→    if (!res.ok) return null;
   127→   127→    const data = (await res.json()) as PageLayoutData;
   128→   128→    // Evict oldest entry if cache is full (LRU policy)
   129→   129→    if (layoutCache.size >= LAYOUT_CACHE_MAX) {
   130→   130→      const firstKey = layoutCache.keys().next().value;
   131→   131→      if (firstKey !== undefined) layoutCache.delete(firstKey);
   132→   132→    }
   133→   133→    layoutCache.set(key, data);
   134→   134→    return data;
   135→   135→  } catch (e: unknown) {
   136→   136→    console.warn('[Mushaf] Failed to load page data:', e);
   137→   137→    return null;
   138→   138→  }
   139→   139→}

   145→   145→function getFontUrl(fontName: string): string {
   146→   146→  if (fontName === BSML_FONT) {
   147→   147→    return `${FONT_BASE}${BSML_FONT}.woff2`;

   152→   152→/**
   153→   153→ * Fetch font binary data using XMLHttpRequest instead of fetch().
   154→   154→ *
   155→   155→ * CRITICAL: In Capacitor with CapacitorHttp enabled, the native HTTP plugin
   156→   156→ * intercepts ALL fetch() calls and can corrupt binary data (woff2 fonts).
   157→   157→ * XMLHttpRequest is NOT intercepted by CapacitorHttp, so it returns
   158→   158→ * correct binary data. This is the key fix for mushaf mode in Android APK.
   159→   159→ */
   160→   160→function fetchFontViaXHR(url: string): Promise<ArrayBuffer> {
   161→   161→  return new Promise((resolve, reject) => {
   162→   162→    const xhr = new XMLHttpRequest();
   163→   163→    xhr.open('GET', url, true);
   164→   164→    xhr.responseType = 'arraybuffer';
   165→   165→    xhr.onload = () => {
   166→   166→      if (xhr.status === 200) {
   167→   167→        resolve(xhr.response as ArrayBuffer);
   168→   168→      } else {
   169→   169→        reject(new Error(`XHR HTTP ${xhr.status}`));
   170→   170→      }
   171→   171→    };
   172→   172→    xhr.onerror = () => reject(new Error('XHR network error'));
   173→   173→    xhr.ontimeout = () => reject(new Error('XHR timeout'));
   174→   174→    xhr.timeout = 15000; // 15 second timeout
   175→   175→    xhr.send();
   176→   176→  });
   177→   177→}

   179→   179→/**
   180→   180→ * Fetch font binary data. In Capacitor, uses XHR to avoid CapacitorHttp
   181→   181→ * interception. In browsers, uses standard fetch().
   182→   182→ */
   183→   183→async function fetchFontBinary(url: string): Promise<ArrayBuffer> {
   184→   184→  if (isCapacitor()) {
   185→   185→    // Use XHR in Capacitor to avoid CapacitorHttp interception
   186→   186→    console.log(`[Mushaf] Using XHR for font binary (bypassing CapacitorHttp): ${url}`);
   187→   187→    return fetchFontViaXHR(url);
   188→   188→  }
   189→   189→  // Use standard fetch in browser
   190→   190→  const res = await fetch(url);
   191→   191→  if (!res.ok) throw new Error(`HTTP ${res.status}`);
   192→   192→  return res.arrayBuffer();
   193→   193→}

   195→   195→/**
   196→   196→ * Verify a font is truly usable on Canvas by drawing a PUA character
   197→   197→ * and checking if the pixels are non-blank.
   198→   198→ *
   199→   199→ * IMPORTANT: In Android WebView (Capacitor), Canvas pixel verification is unreliable
   200→   200→ * because WebView processes fonts differently. We use document.fonts.check()
   201→   201→ * as a fallback verification method in that case.
   202→   202→ */
   203→   203→function verifyFontOnCanvas(fontName: string): boolean {
   204→   204→  try {
   205→   205→    // In Capacitor/Android WebView, skip pixel-level Canvas verification
   206→   206→    // because it's unreliable — trust document.fonts API instead
   207→   207→    if (isCapacitor()) {
   208→   208→      const fontsCheck = document.fonts.check(`40px "${fontName}"`);
   209→   209→      console.log(`[Mushaf] Capacitor: document.fonts.check for "${fontName}": ${fontsCheck}`);
   210→   210→      return fontsCheck;
   211→   211→    }
   212→   212→
   213→   213→    const testSize = 40;
   214→   214→    const canvas = document.createElement('canvas');
   215→   215→    canvas.width = testSize * 2;
   216→   216→    canvas.height = testSize * 2;
   217→   217→    const ctx = canvas.getContext('2d');
   218→   218→    if (!ctx) return false;
   219→   219→
   220→   220→    ctx.fillStyle = '#FFFFFF';
   221→   221→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   222→   222→
   223→   223→    ctx.font = `${testSize}px "${fontName}"`;
   224→   224→    ctx.fillStyle = '#000000';
   225→   225→    ctx.textAlign = 'center';
   226→   226→    ctx.textBaseline = 'middle';
   227→   227→    ctx.fillText('\uF100', canvas.width / 2, canvas.height / 2);
   228→   228→
   229→   229→    const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
   230→   230→    const data = imageData.data;
   231→   231→    for (let i = 0; i < data.length; i += 4) {
   232→   232→      if (data[i] < 250 || data[i + 1] < 250 || data[i + 2] < 250) {
   233→   233→        return true;
   234→   234→      }
   235→   235→    }
   236→   236→
   237→   237→    return false;
   238→   238→  } catch {
   239→   239→    return false;
   240→   240→  }
   241→   241→}
   242→   242→
   243→   243→/**
   244→   244→ * Create a hidden DOM element that uses the font. This forces the browser
   245→   245→ * to download and parse the font binary, making it available for Canvas.
   246→   246→ * Some browsers (especially Android WebView) won't make fonts available
   247→   247→ * to Canvas unless they're used in the DOM first.
   248→   248→ */
   249→   249→function createFontPreloadElement(fontName: string): void {
   250→   250→  const id = `qcf-preload-${fontName}`;
   251→   251→  if (document.getElementById(id)) return;
   252→   252→
   253→   253→  const span = document.createElement('span');
   254→   254→  span.id = id;
   255→   255→  span.setAttribute('aria-hidden', 'true');
   256→   256→  span.style.cssText = `
   257→   257→    position: absolute;
   258→   258→    left: -9999px;
   259→   259→    top: -9999px;
   260→   260→    visibility: hidden;
   261→   261→    font-family: "${fontName}";
   262→   262→    font-size: 40px;
   263→   263→    pointer-events: none;
   264→   264→  `;
   265→   265→  // Use multiple PUA characters to ensure the browser loads the font
   266→   266→  span.textContent = '\uF100\uF101\uF102\uF103\uF104\uF105';
   267→   267→  document.body.appendChild(span);
   268→   268→}

   270→   270→/**
   271→   271→ * Load a QCF4 font using the most reliable method available.
   272→   272→ *
   273→   273→ * Strategy order (most to least reliable):
   274→   274→ * 1. XHR fetch as ArrayBuffer → FontFace from binary data
   275→   275→ *    (Bypasses CapacitorHttp which corrupts binary data)
   276→   276→ * 2. FontFace API with URL (browser handles download)
   277→   277→ * 3. CSS @font-face injection with DOM preload element
   278→   278→ *
   279→   279→ * In Capacitor: We skip pixel-level Canvas verification and trust
   280→   280→ * document.fonts API instead.
   281→   281→ */
   282→   282→async function ensureFontLoaded(fontName: string): Promise<void> {
   283→   283→  if (loadedFonts.has(fontName)) return;
   284→   284→
   285→   285→  // Deduplicate concurrent loads
   286→   286→  if (loadingFonts.has(fontName)) {
   287→   287→    const result = await loadingFonts.get(fontName);
   288→   288→    if (result) return;
   289→   289→    loadingFonts.delete(fontName);
   290→   290→  }
   291→   291→
   292→   292→  const loadPromise = _doLoadFont(fontName);
   293→   293→  loadingFonts.set(fontName, loadPromise);
   294→   294→
   295→   295→  try {
   296→   296→    const success = await loadPromise;
   297→   297→    if (success) {
   298→   298→      loadedFonts.add(fontName);
   299→   299→    }
   300→   300→  } finally {
   301→   301→    loadingFonts.delete(fontName);
   302→   302→  }
   303→   303→}

   305→   305→/**
   306→   306→ * Internal font loading implementation.
   307→   307→ * Returns true if the font was loaded and verified.
   308→   308→ */
   309→   309→async function _doLoadFont(fontName: string): Promise<boolean> {
   310→   310→  const fontUrl = getFontUrl(fontName);
   311→   311→  const capMode = isCapacitor();
   312→   312→  const waitMs = capMode ? 1200 : 200;
   313→   313→
   314→   314→  for (let attempt = 1; attempt <= (capMode ? 2 : 3); attempt++) {
   315→   315→    console.log(`[Mushaf] Loading font "${fontName}" (attempt ${attempt})...`);
   316→   316→
   317→   317→    // --- Strategy 1: XHR/Fetch ArrayBuffer → FontFace (most reliable) ---
   318→   318→    try {
   319→   319→      console.log(`[Mushaf] Strategy 1: ArrayBuffer for "${fontName}" from ${fontUrl}`);
   320→   320→      const buffer = await fetchFontBinary(fontUrl);
   321→   321→      console.log(`[Mushaf] Font "${fontName}" downloaded: ${buffer.byteLength} bytes`);
   322→   322→
   323→   323→      // Validate: woff2 files should be at least 10KB
   324→   324→      if (buffer.byteLength < 10000) {
   325→   325→        console.warn(`[Mushaf] Font "${fontName}" too small (${buffer.byteLength} bytes), likely corrupted`);
   326→   326→        throw new Error(`Font data too small: ${buffer.byteLength} bytes`);
   327→   327→      }
   328→   328→
   329→   329→      const fontFace = new FontFace(fontName, buffer, { display: 'block' });
   330→   330→      const loaded = await fontFace.load();
   331→   331→      document.fonts.add(loaded);
   332→   332→
   333→   333→      await document.fonts.ready;
   334→   334→      createFontPreloadElement(fontName);
   335→   335→      await new Promise<void>((r) => setTimeout(r, waitMs));
   336→   336→
   337→   337→      const verified = verifyFontOnCanvas(fontName);
   338→   338→      console.log(`[Mushaf] Strategy 1 verify for "${fontName}": ${verified}`);
   339→   339→      if (verified) {
   340→   340→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via ArrayBuffer`);
   341→   341→        return true;
   342→   342→      }
   343→   343→
   344→   344→      // In Capacitor: trust document.fonts and proceed even if Canvas verify fails
   345→   345→      if (capMode) {
   346→   346→        const fontsCheck = document.fonts.check(`40px "${fontName}"`);
   347→   347→        if (fontsCheck) {
   348→   348→          console.log(`[Mushaf] Capacitor: Trusting document.fonts.check for "${fontName}"`);
   349→   349→          return true;
   350→   350→        }
   351→   351→        // Even if fonts.check fails, if we added the font, proceed anyway
   352→   352→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" despite verification failure`);
   353→   353→        return true;
   354→   354→      }
   355→   355→    } catch (e) {
   356→   356→      console.warn(`[Mushaf] ArrayBuffer strategy failed for "${fontName}":`, e);
   357→   357→    }
   358→   358→
   359→   359→    // --- Strategy 2: FontFace with URL (browser handles download) ---
   360→   360→    try {
   361→   361→      console.log(`[Mushaf] Strategy 2: FontFace URL for "${fontName}"`);
   362→   362→      const fontFace = new FontFace(fontName, `url('${fontUrl}')`, { display: 'block' });
   363→   363→      const loaded = await fontFace.load();
   364→   364→      document.fonts.add(loaded);
   365→   365→
   366→   366→      await document.fonts.ready;
   367→   367→      createFontPreloadElement(fontName);
   368→   368→      await new Promise<void>((r) => setTimeout(r, waitMs));
   369→   369→
   370→   370→      const verified = verifyFontOnCanvas(fontName);
   371→   371→      console.log(`[Mushaf] Strategy 2 verify for "${fontName}": ${verified}`);
   372→   372→      if (verified) {
   373→   373→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via FontFace URL`);
   374→   374→        return true;
   375→   375→      }
   376→   376→
   377→   377→      if (capMode) {
   378→   378→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" (strategy 2) despite verification failure`);
   379→   379→        return true;
   380→   380→      }
   381→   381→    } catch (e) {
   382→   382→      console.warn(`[Mushaf] FontFace URL strategy failed for "${fontName}":`, e);
   383→   383→    }
   384→   384→
   385→   385→    // --- Strategy 3: CSS @font-face + DOM preload ---
   386→   386→    try {
   387→   387→      console.log(`[Mushaf] Strategy 3: CSS @font-face for "${fontName}"`);
   388→   388→      const styleId = fontName === BSML_FONT ? 'qcf-basml' : `qcf-${fontName}`;
   389→   389→      if (!document.getElementById(styleId)) {
   390→   390→        const style = document.createElement('style');
   391→   391→        style.id = styleId;
   392→   392→        style.textContent = `
   393→   393→          @font-face {
   394→   394→            font-family: '${fontName}';
   395→   395→            src: url('${fontUrl}') format('woff2');
   396→   396→            font-display: block;
   397→   397→          }`;
   398→   398→        document.head.appendChild(style);
   399→   399→      }
   400→   400→
   401→   401→      createFontPreloadElement(fontName);
   402→   402→
   403→   403→      try {
   404→   404→        await document.fonts.load(`40px "${fontName}"`, '\uF100');
   405→   405→      } catch {
   406→   406→        /* some browsers don't support the text param */
   407→   407→      }
   408→   408→
   409→   409→      await document.fonts.ready;
   410→   410→      const cssWaitMs = capMode ? 1500 : 500;
   411→   411→      await new Promise<void>((r) => setTimeout(r, cssWaitMs));
   412→   412→
   413→   413→      const verified = verifyFontOnCanvas(fontName);
   414→   414→      console.log(`[Mushaf] Strategy 3 verify for "${fontName}": ${verified}`);
   415→   415→      if (verified) {
   416→   416→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via CSS @font-face`);
   417→   417→        return true;
   418→   418→      }
   419→   419→
   420→   420→      if (capMode) {
   421→   421→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" (strategy 3) despite verification failure`);
   422→   422→        return true;
   423→   423→      }
   424→   424→    } catch (e) {
   425→   425→      console.warn(`[Mushaf] CSS @font-face strategy failed for "${fontName}":`, e);
   426→   426→    }
   427→   427→
   428→   428→    if (attempt < (capMode ? 2 : 3)) {
   429→   429→      const delay = capMode ? 1000 : 500 * attempt;
   430→   430→      console.warn(`[Mushaf] All strategies failed for "${fontName}" (attempt ${attempt}), retrying in ${delay}ms...`);
   431→   431→      await new Promise<void>((r) => setTimeout(r, delay));
   432→   432→    }
   433→   433→  }
   434→   434→
   435→   435→  // In Capacitor: proceed anyway — better to show something than blank
   436→   436→  if (capMode) {
   437→   437→    console.warn(`[Mushaf] Capacitor: Proceeding despite font "${fontName}" loading failure — will retry render later`);
   438→   438→    return true;
   439→   439→  }
   440→   440→
   441→   441→  console.error(`[Mushaf] FAILED to load font "${fontName}" after all attempts.`);
   442→   442→  return false;
   443→   443→}

   458→   458→/**
   459→   459→ * Render a mushaf page onto a canvas element.

   461→   461→ * For Capacitor/Android WebView, includes special handling:
   462→   462→ * - Uses XHR for font binary data (bypasses CapacitorHttp)
   463→   463→ * - Skips strict Canvas pixel verification
   464→   464→ * - Schedules a re-render after a delay for WebView font processing
   465→   465→ * - Adds an overall timeout to prevent hanging
   466→   466→ */
   467→   467→export async function renderPage(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   468→   468→  const capMode = isCapacitor();
   469→   469→  console.log(`[Mushaf] renderPage(${pageNum}) starting... Capacitor=${capMode}`);
   470→   470→
   471→   471→  // Add overall timeout for Capacitor to prevent hanging
   472→   472→  if (capMode) {
   473→   473→    const timeoutMs = 30000;
   474→   474→    return Promise.race([
   475→   475→      _renderPageInternal(pageNum, targetCanvas),
   476→   476→      new Promise<RenderPageResult>((resolve) =>
   477→   477→        setTimeout(() => {
   478→   478→          console.error(`[Mushaf] renderPage(${pageNum}) timed out after ${timeoutMs}ms`);
   479→   479→          _renderPageWithCurrentFonts(pageNum, targetCanvas).then(resolve);
   480→   480→        }, timeoutMs)
   481→   481→      )
   482→   482→    ]);
   483→   483→  }
   484→   484→
   485→   485→  return _renderPageInternal(pageNum, targetCanvas);
   486→   486→}

   489→   489→ * Attempt to render a page using whatever fonts are currently available,
   490→   490→ * even if font loading failed. Better to show partial text than blank.
   491→   491→ */
   492→   492→async function _renderPageWithCurrentFonts(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   493→   493→  try {
   494→   494→    const data = await loadPageData(pageNum);
   495→   495→    if (!data) return { canvas: null, layout: null };
   496→   496→
   497→   497→    const pageFont = data.font || getPageFont(pageNum, null);
   498→   498→    const canvas = targetCanvas || document.createElement('canvas');
   499→   499→    canvas.width = CANVAS_W;
   500→   500→    canvas.height = CANVAS_H;
   501→   501→    const ctx = canvas.getContext('2d');
   502→   502→    if (!ctx) return { canvas: null, layout: data };
   503→   503→
   504→   504→    const colors = getColors();
   505→   505→    ctx.fillStyle = colors.bg;
   506→   506→    ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   507→   507→    renderPageContent(ctx, data, pageFont, colors);
   508→   508→    drawPageFrame(ctx, colors);
   509→   509→    drawPageNumber(ctx, pageNum, colors);
   510→   510→    return { canvas, layout: data };
   511→   511→  } catch {
   512→   512→    return { canvas: null, layout: null };
   513→   513→  }

   516→   516→async function _renderPageInternal(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   517→   517→  const capMode = isCapacitor();
   518→   518→  const data = await loadPageData(pageNum);
   519→   519→  if (!data) {
   520→   520→    console.error(`[Mushaf] Failed to load page data for page ${pageNum}`);
   521→   521→    return { canvas: null, layout: null };
   522→   522→  }
   523→   523→  console.log(`[Mushaf] Page ${pageNum} data loaded: ${data.lines?.length || 0} lines`);
   524→   524→
   525→   525→  const pageFont = data.font || getPageFont(pageNum, null);
   526→   526→
   527→   527→  // Collect ALL unique font names used on this page
   528→   528→  const pageFonts = new Set<string>();
   529→   529→  pageFonts.add(pageFont);
   530→   530→  pageFonts.add(BSML_FONT);
   531→   531→  for (const line of data.lines ?? []) {
   532→   532→    for (const w of line.words ?? []) {
   533→   533→      if (w.font) pageFonts.add(w.font);
   534→   534→    }
   535→   535→  }
   536→   536→
   537→   537→  console.log(`[Mushaf] Page ${pageNum} needs fonts: ${[...pageFonts].join(', ')}`);
   538→   538→
   539→   539→  // Load all fonts in parallel with individual timeouts
   540→   540→  const fontPromises = [...pageFonts].map((fn) =>
   541→   541→    Promise.race([
   542→   542→      ensureFontLoaded(fn),
   543→   543→      new Promise<void>((resolve) => setTimeout(resolve, capMode ? 8000 : 5000))
   544→   544→    ])
   545→   545→  );
   546→   546→  await Promise.all(fontPromises);
   547→   547→
   548→   548→  // Ensure Reem Kufi is loaded for bismillah rendering
   549→   549→  try {
   550→   550→    await document.fonts.load('500 40px "Reem Kufi"');
   551→   551→  } catch {
   552→   552→    /* ignore */
   553→   553→  }
   554→   554→
   555→   555→  // Wait for all fonts to be ready
   556→   556→  try {
   557→   557→    await document.fonts.ready;
   558→   558→  } catch {
   559→   559→    /* ignore */
   560→   560→  }
   561→   561→
   562→   562→  // In Capacitor: extra wait for WebView font processing
   563→   563→  if (capMode) {
   564→   564→    await new Promise<void>((r) => setTimeout(r, 1000));
   565→   565→    await document.fonts.ready;
   566→   566→  } else {
   567→   567→    // Non-Capacitor: verify font on Canvas
   568→   568→    if (!loadedFonts.has(pageFont) || !verifyFontOnCanvas(pageFont)) {
   569→   569→      console.warn(`[Mushaf] Font "${pageFont}" not verified on Canvas, attempting final reload...`);
   570→   570→      loadedFonts.delete(pageFont);
   571→   571→      await ensureFontLoaded(pageFont);
   572→   572→    }
   573→   573→  }
   574→   574→
   575→   575→  const canvas = targetCanvas || document.createElement('canvas');
   576→   576→  canvas.width = CANVAS_W;
   577→   577→  canvas.height = CANVAS_H;
   578→   578→  const ctx = canvas.getContext('2d');
   579→   579→
   580→   580→  const colors = getColors();
   581→   581→
   582→   582→  ctx!.fillStyle = colors.bg;
   583→   583→  ctx!.fillRect(0, 0, CANVAS_W, CANVAS_H);
   584→   584→
   585→   585→  renderPageContent(ctx!, data, pageFont, colors);
   586→   586→
   587→   587→  drawPageFrame(ctx!, colors);
   588→   588→
   589→   589→  drawPageNumber(ctx!, pageNum, colors);
   590→   590→
   591→   591→  console.log(`[Mushaf] Page ${pageNum} rendered successfully`);
   592→   592→
   593→   593→  // In Capacitor: schedule a single well-timed re-render because Android WebView
   594→   594→  // sometimes needs an extra render pass for fonts to take effect on Canvas.
   595→   595→  // We do ONE re-render at 3 seconds (not 3 separate ones which cause flicker).
   596→   596→  if (capMode) {
   597→   597→    const renderData = { data, pageFont, colors, pageNum };
   598→   598→    setTimeout(() => {
   599→   599→      console.log(`[Mushaf] Capacitor: Re-rendering page ${renderData.pageNum} after 3s delay`);
   600→   600→      try {
   601→   601→        const c = targetCanvas || document.querySelector('.mushaf-page-canvas') as HTMLCanvasElement;
   602→   602→        if (c && c.width === CANVAS_W) {
   603→   603→          const ctx2 = c.getContext('2d');
   604→   604→          if (ctx2) {
   605→   605→            ctx2.fillStyle = renderData.colors.bg;
   606→   606→            ctx2.fillRect(0, 0, CANVAS_W, CANVAS_H);
   607→   607→            renderPageContent(ctx2, renderData.data, renderData.pageFont, renderData.colors);
   608→   608→            drawPageFrame(ctx2, renderData.colors);
   609→   609→            drawPageNumber(ctx2, renderData.pageNum, renderData.colors);
   610→   610→            console.log(`[Mushaf] Capacitor: Re-render of page ${renderData.pageNum} complete`);
   611→   611→          }
   612→   612→        }
   613→   613→      } catch (e) {
   614→   614→        console.warn('[Mushaf] Capacitor re-render failed:', e);
   615→   615→      }
   616→   616→    }, 3000);
   617→   617→  }
   618→   618→
   619→   619→  return { canvas, layout: data };
   620→   620→}

   622→   622→function measureLine(ctx: CanvasRenderingContext2D, words: PageWord[], pageFont: string, fontSize: number): number[] {

   630→   630→export function getLineY(lineIndex: number, lineCount: number, imgHeight: number): number {
   631→   631→  if (lineIndex <= 0) return 0;
   632→   632→  if (lineIndex >= lineCount) return imgHeight;
   633→   633→  const usableHeight = CANVAS_H - TOP_OFFSET - BOTTOM_OFFSET - PAD_V;
   634→   634→  const stdLineHeight = usableHeight / STD_LINES;
   635→   635→  const isShortPage = lineCount < STD_LINES;
   636→   636→  const lineSpacing = isShortPage ? (usableHeight - stdLineHeight) / Math.max(1, lineCount - 1) : stdLineHeight;
   637→   637→  const y = TOP_OFFSET + lineIndex * lineSpacing + (isShortPage ? 0 : stdLineHeight / 2);
   638→   638→  return (y / CANVAS_H) * imgHeight;
   639→   639→}

   641→   641→/** Pre-compute per-word tajweed coloring data for the entire page. */
   642→   642→function computePageTajweed(data: PageLayoutData): { wordIdx: number; lineIdx: number; color: string | null }[] | null {
   643→   643→  if (!state.tajweedEnabled) return null;
   644→   644→
   645→   645→  const verseKeyWords = new Map<string, { text: string; lineIdx: number; wordIdx: number }[]>();
   646→   646→  for (let li = 0; li < data.lines.length; li++) {
   647→   647→    const line = data.lines[li];
   648→   648→    if (!line?.words) continue;
   649→   649→    for (let wi = 0; wi < line.words.length; wi++) {
   650→   650→      const w = line.words[wi];
   651→   651→      if (w.verse_key && w.text && w.type === 'word') {
   652→   652→        if (!verseKeyWords.has(w.verse_key)) verseKeyWords.set(w.verse_key, []);
   653→   653→        verseKeyWords.get(w.verse_key)!.push({ text: w.text, lineIdx: li, wordIdx: wi });
   654→   654→      }
   655→   655→    }
   656→   656→  }
   657→   657→
   658→   658→  const result: { wordIdx: number; lineIdx: number; color: string | null }[] = [];
   659→   659→
   660→   660→  for (const [vk, words] of verseKeyWords) {
   661→   661→    const parts = vk.split(':');
   662→   662→    const surah = parseInt(parts[0], 10);
   663→   663→    const ayah = parseInt(parts[1], 10);
   664→   664→
   665→   665→    const annotations: TajweedAnnotation[] = getAyahAnnotations(surah, ayah);
   666→   666→    if (annotations.length === 0) {
   667→   667→      for (const w of words) result.push({ wordIdx: w.wordIdx, lineIdx: w.lineIdx, color: null });
   668→   668→      continue;
   669→   669→    }
   670→   670→
   671→   671→    const ayahText = words.map((w) => w.text).join(' ');
   672→   672→    const colorMap = buildColorMap(annotations);
   673→   673→    if (!colorMap || colorMap.size === 0) {
   674→   674→      for (const w of words) result.push({ wordIdx: w.wordIdx, lineIdx: w.lineIdx, color: null });
   675→   675→      continue;
   676→   676→    }
   677→   677→
   678→   678→    let outputPos = 0;
   679→   679→    for (let wi = 0; wi < words.length; wi++) {
   680→   680→      const wordText = words[wi].text;
   681→   681→      let wordColor: string | null = null;
   682→   682→      for (let ci = 0; ci < wordText.length; ci++) {
   683→   683→        const c = colorMap.get(outputPos + ci);
   684→   684→        if (c) {
   685→   685→          wordColor = c;
   686→   686→          break;
   687→   687→        }
   688→   688→      }
   689→   689→      result.push({ wordIdx: words[wi].wordIdx, lineIdx: words[wi].lineIdx, color: wordColor });
   690→   690→      outputPos += wordText.length + 1;
   691→   691→    }
   692→   692→  }
   693→   693→
   694→   694→  return result;
   695→   695→}

   697→   697→function renderPageContent(
   698→   698→  ctx: CanvasRenderingContext2D,
   699→   699→  data: PageLayoutData,
   700→   700→  pageFont: string,
   701→   701→  colors: PageColors
   702→   702→): void {
   703→   703→  const lines = data.lines;
   704→   704→  if (!lines || lines.length === 0) return;
   705→   705→
   706→   706→  const lineCount = lines.length;
   707→   707→  const usableHeight = CANVAS_H - TOP_OFFSET - BOTTOM_OFFSET - PAD_V;
   708→   708→  const availableW = CANVAS_W - PAD_H * 2;
   709→   709→
   710→   710→  const stdLineHeight = usableHeight / STD_LINES;
   711→   711→  const baseFontSize = Math.max(26, Math.min(55, stdLineHeight * 0.85));
   712→   712→  const pageFontSize = baseFontSize;
   713→   713→
   714→   714→  const isShortPage = lineCount < 15;
   715→   715→  const lineSpacing = isShortPage ? (usableHeight - stdLineHeight) / Math.max(1, lineCount - 1) : stdLineHeight;
   716→   716→
   717→   717→  const lineWidths: (LineWidths | null)[] = [];
   718→   718→  for (let i = 0; i < lineCount; i++) {
   719→   719→    const line = lines[i];
   720→   720→    if (!line?.words || line.words.length === 0) {
   721→   721→      lineWidths.push(null);
   722→   722→      continue;
   723→   723→    }
   724→   724→    const widths = measureLine(ctx, line.words, pageFont, pageFontSize);
   725→   725→    const totalW = widths.reduce((a: number, b: number) => a + b, 0);
   726→   726→    const gap = line.words.length > 1 ? (availableW - totalW) / (line.words.length - 1) : 0;
   727→   727→    lineWidths.push({ widths, gap: Math.max(0, gap) });
   728→   728→  }
   729→   729→
   730→   730→  const tajweedColors = computePageTajweed(data);
   731→   731→  const tajweedLookup = new Map<string, string | null>();
   732→   732→  if (tajweedColors) {
   733→   733→    for (const entry of tajweedColors) {
   734→   734→      tajweedLookup.set(`${entry.lineIdx}:${entry.wordIdx}`, entry.color);
   735→   735→    }
   736→   736→  }
   737→   737→
   738→   738→  const bsmlColor = isNightMode() ? '#d4af37' : '#8b6914';
   739→   739→  const headerColor = isNightMode() ? '#c4a87c' : '#7a5c3a';
   740→   740→  const bsmlFont = 'Reem Kufi';
   741→   741→  const bsmlText = 'بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ';
   742→   742→
   743→   743→  ctx.textBaseline = 'middle';
   744→   744→
   745→   745→  for (let i = 0; i < lineCount; i++) {
   746→   746→    const line = lines[i];
   747→   747→    if (!line?.words || line.words.length === 0) continue;
   748→   748→
   749→   749→    const y = TOP_OFFSET + i * lineSpacing + (isShortPage ? 0 : stdLineHeight / 2);
   750→   750→    const words = line.words;
   751→   751→    const lw = lineWidths[i];
   752→   752→    if (!lw) continue;
   753→   753→    const { widths, gap } = lw;
   754→   754→
   755→   755→    const lineType = words[0]?.type;
   756→   756→    const isBismillahLine = lineType === 'bismillah';
   757→   757→    const isHeaderLine = lineType === 'surah_header';
   758→   758→
   759→   759→    if (isBismillahLine) {
   760→   760→      const bsmlFontSize = Math.round(pageFontSize * 0.95);
   761→   761→      ctx.font = `500 ${bsmlFontSize}px "${bsmlFont}", "Scheherazade New", serif`;
   762→   762→      ctx.textAlign = 'center';
   763→   763→      ctx.fillStyle = bsmlColor;
   764→   764→      ctx.fillText(bsmlText, CANVAS_W / 2, y);
   765→   765→      continue;
   766→   766→    }
   767→   767→
   768→   768→    let x = CANVAS_W - PAD_H;
   769→   769→
   770→   770→    for (let j = 0; j < words.length; j++) {
   771→   771→      const w = words[j];
   772→   772→      const fn = w.font || pageFont;
   773→   773→      ctx.font = `${pageFontSize}px "${fn}"`;
   774→   774→      ctx.textAlign = 'right';
   775→   775→
   776→   776→      if (isHeaderLine) {
   777→   777→        ctx.fillStyle = headerColor;
   778→   778→      } else {
   779→   779→        const wordColor = tajweedLookup ? tajweedLookup.get(`${i}:${j}`) : null;
   780→   780→        ctx.fillStyle = wordColor || colors.txt;
   781→   781→      }
   782→   782→
   783→   783→      ctx.fillText(w.char, x, y);
   784→   784→      x -= widths[j] + gap;
   785→   785→    }
   786→   786→  }
   787→   787→}
