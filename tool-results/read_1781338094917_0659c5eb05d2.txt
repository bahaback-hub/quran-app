    41→/** Get a background image based on the current time of day. */
    42→function getAutoBackground(): (typeof NATURE_BACKGROUNDS)[number] {
    43→  const hour = new Date().getHours();
    44→  if (hour >= 4 && hour < 7) return NATURE_BACKGROUNDS[0]; // dawn
    45→  if (hour >= 7 && hour < 12) return NATURE_BACKGROUNDS[1]; // morning clouds
    46→  if (hour >= 12 && hour < 16) return NATURE_BACKGROUNDS[2]; // afternoon mountains
    47→  if (hour >= 16 && hour < 19) return NATURE_BACKGROUNDS[3]; // sunset
    48→  return NATURE_BACKGROUNDS[4]; // night sky
    49→}
    50→
    51→/** Get a specific nature background by its mood key. */
    52→function getNatureBgByMood(mood: string): (typeof NATURE_BACKGROUNDS)[number] | null {
    53→  return NATURE_BACKGROUNDS.find((bg) => bg.mood === mood) || null;
    54→}

    56→/** Get a random nature background (cycles per ayah). */
    57→function getRandomNatureBg(): (typeof NATURE_BACKGROUNDS)[number] {
    58→  const idx = Math.floor(Math.random() * NATURE_BACKGROUNDS.length);
    59→  return NATURE_BACKGROUNDS[idx];
    60→}

   495→function buildAyahHtml(text: string, surahNum: number, ayahNum: number): string {
   496→  let txt = text;
   497→  let offsetAdj = 0;
   498→  if (surahNum !== 1 && ayahNum === 1) {
   499→    const stripped = txt.replace(
   500→      /^ب[\u064B-\u065F\u0670]*س[\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*[هة][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*[نث][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*[يى][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*/u,
   501→      ''
   502→    );
   503→    offsetAdj = txt.length - stripped.length;
   504→    txt = stripped;
   505→  }
   506→  if (!_presTajweedEnabled) return escapeHtml(txt);
   507→  const annotations = getAyahAnnotations(surahNum, ayahNum);
   508→  if (annotations.length === 0) return escapeHtml(txt);
   509→  const adjusted =
   510→    offsetAdj > 0
   511→      ? annotations.map((ann: TajweedAnnotation) => ({
   512→          rule: ann.rule,
   513→          start: ann.start - offsetAdj,
   514→          end: ann.end - offsetAdj,
   515→        }))
   516→      : annotations;
   517→  const colorMap = buildColorMap(adjusted);
   518→  const words = txt.split(/\s+/).filter((w) => w.length > 0);
   519→  let outputPos = 0;
   520→  return words
   521→    .map((word: string, idx: number) => {
   522→      const wordHtml = tajweedColorWord(word, outputPos, colorMap);
   523→      outputPos += word.length + (idx < words.length - 1 ? 1 : 0); // +1 for space between words
   524→      return wordHtml;
   525→    })
   526→    .join(' ');
   527→}

   600→/** Star scene — twinkling stars with occasional shooting stars. */
   601→function renderStarsScene(canvas: HTMLCanvasElement): () => void {
   602→  const ctx = canvas.getContext('2d')!;
   603→  let stars: StarData[] = [];
   604→  let shootingStars: ShootingStar[] = [];
   605→  let time = 0;
   606→
   607→  function init(): void {
   608→    const w = canvas.width;
   609→    const h = canvas.height;
   610→    stars = [];
   611→    for (let i = 0; i < 250; i++) {
   612→      stars.push({
   613→        x: Math.random() * w,
   614→        y: Math.random() * h,
   615→        size: Math.random() * 2.5 + 0.5,
   616→        speed: Math.random() * 0.02 + 0.005,
   617→        brightness: Math.random(),
   618→        phase: Math.random() * Math.PI * 2,
   619→      });
   620→    }
   621→  }
   622→  init();
   623→
   624→  function frame(): void {
   625→    const w = canvas.width;
   626→    const h = canvas.height;
   627→    time += 0.016;
   628→
   629→    // Dark sky gradient
   630→    const grad = ctx.createLinearGradient(0, 0, 0, h);
   631→    grad.addColorStop(0, '#05050f');
   632→    grad.addColorStop(0.5, '#0a0a2e');
   633→    grad.addColorStop(1, '#101030');
   634→    ctx.fillStyle = grad;
   635→    ctx.fillRect(0, 0, w, h);
   636→
   637→    // Milky way band
   638→    const mwGrad = ctx.createLinearGradient(0, h * 0.2, 0, h * 0.6);
   639→    mwGrad.addColorStop(0, 'rgba(60, 50, 80, 0)');
   640→    mwGrad.addColorStop(0.3, 'rgba(80, 60, 120, 0.06)');
   641→    mwGrad.addColorStop(0.5, 'rgba(100, 80, 140, 0.08)');
   642→    mwGrad.addColorStop(0.7, 'rgba(80, 60, 120, 0.06)');
   643→    mwGrad.addColorStop(1, 'rgba(60, 50, 80, 0)');
   644→    ctx.fillStyle = mwGrad;
   645→    ctx.fillRect(0, 0, w, h);
   646→
   647→    // Draw twinkling stars
   648→    for (const star of stars) {
   649→      const twinkle = 0.5 + 0.5 * Math.sin(time * star.speed * 60 + star.phase);
   650→      const alpha = star.brightness * twinkle;
   651→      ctx.beginPath();
   652→      ctx.arc(star.x, star.y, star.size * (0.7 + 0.3 * twinkle), 0, Math.PI * 2);
   653→      ctx.fillStyle = `rgba(255, 255, 240, ${alpha})`;
   654→      ctx.fill();
   655→      // Glow for bigger stars
   656→      if (star.size > 1.8) {
   657→        ctx.beginPath();
   658→        ctx.arc(star.x, star.y, star.size * 3, 0, Math.PI * 2);
   659→        ctx.fillStyle = `rgba(200, 200, 255, ${alpha * 0.1})`;
   660→        ctx.fill();
   661→      }
   662→    }
   663→
   664→    // Shooting stars
   665→    if (Math.random() < 0.003) {
   666→      shootingStars.push({
   667→        x: Math.random() * w * 0.8,
   668→        y: Math.random() * h * 0.3,
   669→        vx: 4 + Math.random() * 4,
   670→        vy: 2 + Math.random() * 2,
   671→        life: 0,
   672→        maxLife: 30 + Math.random() * 30,
   673→        length: 40 + Math.random() * 60,
   674→      });
   675→    }
   676→    for (let i = shootingStars.length - 1; i >= 0; i--) {
   677→      const ss = shootingStars[i];
   678→      ss.x += ss.vx;
   679→      ss.y += ss.vy;
   680→      ss.life++;
   681→      const alpha = 1 - ss.life / ss.maxLife;
   682→      if (alpha <= 0) { shootingStars.splice(i, 1); continue; }
   683→      ctx.beginPath();
   684→      ctx.moveTo(ss.x, ss.y);
   685→      ctx.lineTo(ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6);
   686→      const ssGrad = ctx.createLinearGradient(
   687→        ss.x, ss.y,
   688→        ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6
   689→      );
   690→      ssGrad.addColorStop(0, `rgba(255, 255, 255, ${alpha})`);
   691→      ssGrad.addColorStop(1, `rgba(255, 255, 255, 0)`);
   692→      ctx.strokeStyle = ssGrad;
   693→      ctx.lineWidth = 2;
   694→      ctx.stroke();
   695→    }
   696→
   697→    _sceneAnimationId = requestAnimationFrame(frame);
   698→  }
   699→  frame();
   700→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   701→}

   703→/** Ocean waves scene — animated waves with gradient sky. */

   705→  const ctx = canvas.getContext('2d')!;
   706→  let time = 0;
   707→
   708→  const waveLayers: WaveLayer[] = [
   709→    { amplitude: 25, frequency: 0.008, speed: 0.4, yBase: 0.45, color: 'rgba(10, 60, 120, 0.4)' },
   710→    { amplitude: 20, frequency: 0.012, speed: 0.6, yBase: 0.52, color: 'rgba(15, 80, 160, 0.5)' },
   711→    { amplitude: 18, frequency: 0.015, speed: 0.8, yBase: 0.58, color: 'rgba(20, 100, 180, 0.6)' },
   712→    { amplitude: 15, frequency: 0.02, speed: 1.0, yBase: 0.64, color: 'rgba(25, 120, 200, 0.7)' },
   713→    { amplitude: 12, frequency: 0.025, speed: 1.3, yBase: 0.70, color: 'rgba(30, 140, 210, 0.8)' },
   714→    { amplitude: 8, frequency: 0.03, speed: 1.6, yBase: 0.78, color: 'rgba(35, 150, 220, 0.85)' },
   715→    { amplitude: 5, frequency: 0.04, speed: 2.0, yBase: 0.85, color: 'rgba(40, 160, 230, 0.9)' },
   716→  ];
   717→
   718→  function frame(): void {
   719→    const w = canvas.width;
   720→    const h = canvas.height;
   721→    time += 0.016;
   722→

   726→    skyGrad.addColorStop(0.3, '#1b2838');

   730→    ctx.fillStyle = skyGrad;
   731→    ctx.fillRect(0, 0, w, h);
   732→
   733→    // Moon glow
   734→    const moonX = w * 0.75;
   735→    const moonY = h * 0.15;
   736→    const moonGlow = ctx.createRadialGradient(moonX, moonY, 0, moonX, moonY, 120);
   737→    moonGlow.addColorStop(0, 'rgba(255, 255, 200, 0.3)');
   738→    moonGlow.addColorStop(0.3, 'rgba(255, 255, 200, 0.08)');

   745→    ctx.arc(moonX, moonY, 20, 0, Math.PI * 2);
   746→    ctx.fillStyle = 'rgba(255, 255, 220, 0.9)';
   747→    ctx.fill();

   749→    // Draw waves back to front

   751→      ctx.beginPath();

   757→        ctx.lineTo(x, y);
   758→      }
   759→      ctx.lineTo(w, h);
   760→      ctx.closePath();
   761→      ctx.fillStyle = wave.color;
   762→      ctx.fill();
   763→    }
   764→
   765→    // Foam/sparkles on top wave
   766→    for (let i = 0; i < 30; i++) {
   767→      const fx = (i / 30) * w + Math.sin(time * 0.5 + i) * 20;
   768→      const fy = h * 0.44 + Math.sin(fx * 0.008 + time * 0.4) * 25 - 5;
   769→      const sparkle = 0.3 + 0.7 * Math.abs(Math.sin(time * 2 + i * 0.5));
   770→      ctx.beginPath();
   771→      ctx.arc(fx, fy, 1.5, 0, Math.PI * 2);
   772→      ctx.fillStyle = `rgba(255, 255, 255, ${sparkle * 0.5})`;
   773→      ctx.fill();
   774→    }
   775→
   776→    _sceneAnimationId = requestAnimationFrame(frame);
   777→  }
   778→  frame();
   779→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   780→}

   782→/** Aurora scene — northern lights effect. */
   783→function renderAuroraScene(canvas: HTMLCanvasElement): () => void {
   784→  const ctx = canvas.getContext('2d')!;
   785→  let time = 0;
   786→
   787→  function frame(): void {
   788→    const w = canvas.width;
   789→    const h = canvas.height;
   790→    time += 0.016;
   791→
   792→    // Dark sky

   794→    skyGrad.addColorStop(0, '#020810');
   795→    skyGrad.addColorStop(0.4, '#050d1a');
   796→    skyGrad.addColorStop(0.7, '#081020');
   797→    skyGrad.addColorStop(1, '#0c1525');

   799→    ctx.fillRect(0, 0, w, h);
   800→

   802→    for (let i = 0; i < 80; i++) {
   803→      const sx = (Math.sin(i * 127.1 + 311.7) * 0.5 + 0.5) * w;
   804→      const sy = (Math.sin(i * 269.5 + 183.3) * 0.5 + 0.5) * h * 0.6;
   805→      const twinkle = 0.3 + 0.7 * Math.abs(Math.sin(time * 0.5 + i));
   806→      ctx.beginPath();
   807→      ctx.arc(sx, sy, 1, 0, Math.PI * 2);
   808→      ctx.fillStyle = `rgba(255, 255, 240, ${twinkle * 0.6})`;
   809→      ctx.fill();
   810→    }
   811→
   812→    // Aurora bands — multiple semi-transparent curved shapes
   813→    const auroraColors = [
   814→      { r: 50, g: 255, b: 100 },  // green
   815→      { r: 80, g: 200, b: 255 },  // blue
   816→      { r: 150, g: 100, b: 255 }, // purple
   817→      { r: 50, g: 255, b: 180 },  // teal
   818→    ];
   819→
   820→    for (let band = 0; band < 4; band++) {
   821→      const color = auroraColors[band];
   822→      const yCenter = h * (0.15 + band * 0.1);
   823→      const bandWidth = h * 0.15;
   824→
   825→      ctx.beginPath();
   826→      ctx.moveTo(0, yCenter);
   827→
   828→      for (let x = 0; x <= w; x += 5) {
   829→        const wave1 = Math.sin(x * 0.003 + time * 0.3 + band * 1.5) * 40;
   830→        const wave2 = Math.sin(x * 0.006 + time * 0.5 + band * 0.8) * 25;
   831→        const wave3 = Math.sin(x * 0.001 + time * 0.15 + band * 2.0) * 60;
   832→        const y = yCenter + wave1 + wave2 + wave3;
   833→        ctx.lineTo(x, y);
   834→      }
   835→
   836→      for (let x = w; x >= 0; x -= 5) {
   837→        const wave1 = Math.sin(x * 0.003 + time * 0.3 + band * 1.5) * 40;
   838→        const wave2 = Math.sin(x * 0.006 + time * 0.5 + band * 0.8) * 25;
   839→        const wave3 = Math.sin(x * 0.001 + time * 0.15 + band * 2.0) * 60;
   840→        const y = yCenter + wave1 + wave2 + wave3 + bandWidth;
   841→        ctx.lineTo(x, y);
   842→      }
   843→
   844→      ctx.closePath();
   845→
   846→      const aGrad = ctx.createLinearGradient(0, yCenter - bandWidth, 0, yCenter + bandWidth * 2);

   872→      ctx.lineTo(rx + 5, h * 0.1 + rayHeight);

   885→/** Golden particles scene — floating dust motes. */
   886→function renderParticlesScene(canvas: HTMLCanvasElement): () => void {
   887→  const ctx = canvas.getContext('2d')!;
   888→  let time = 0;
   889→  let particles: Particle[] = [];
   890→
   891→  function init(): void {
   892→    const w = canvas.width;
   893→    const h = canvas.height;
   894→    particles = [];
   895→    for (let i = 0; i < 120; i++) {
   896→      particles.push({
   897→        x: Math.random() * w,
   898→        y: Math.random() * h,
   899→        vx: (Math.random() - 0.5) * 0.3,
   900→        vy: -Math.random() * 0.5 - 0.1,
   901→        size: Math.random() * 3 + 1,
   902→        opacity: Math.random() * 0.5 + 0.2,
   903→        phase: Math.random() * Math.PI * 2,
   904→      });
   905→    }
   906→  }
   907→  init();
   908→
   909→  function frame(): void {
   910→    const w = canvas.width;
   911→    const h = canvas.height;
   912→    time += 0.016;
   913→
   914→    // Dark warm background
   915→    const bgGrad = ctx.createRadialGradient(w * 0.5, h * 0.4, 0, w * 0.5, h * 0.4, Math.max(w, h) * 0.7);
   916→    bgGrad.addColorStop(0, '#1a1510');
   917→    bgGrad.addColorStop(0.5, '#0f0d08');
   918→    bgGrad.addColorStop(1, '#080604');
   919→    ctx.fillStyle = bgGrad;
   920→    ctx.fillRect(0, 0, w, h);
   921→
   922→    // Warm light source glow
   923→    const lightGlow = ctx.createRadialGradient(w * 0.5, h * 0.3, 0, w * 0.5, h * 0.3, w * 0.4);
   924→    lightGlow.addColorStop(0, 'rgba(255, 200, 100, 0.08)');
   925→    lightGlow.addColorStop(0.5, 'rgba(255, 180, 80, 0.03)');
   926→    lightGlow.addColorStop(1, 'rgba(255, 150, 50, 0)');
   927→    ctx.fillStyle = lightGlow;
   928→    ctx.fillRect(0, 0, w, h);
   929→
   930→    // Draw and update particles
   931→    for (const p of particles) {
   932→      p.x += p.vx + Math.sin(time + p.phase) * 0.2;
   933→      p.y += p.vy;
   934→
   935→      // Wrap around
   936→      if (p.y < -10) { p.y = h + 10; p.x = Math.random() * w; }
   937→      if (p.x < -10) p.x = w + 10;
   938→      if (p.x > w + 10) p.x = -10;
   939→
   940→      const pulse = 0.6 + 0.4 * Math.sin(time * 1.5 + p.phase);
   941→      const alpha = p.opacity * pulse;
   942→
   943→      // Glow
   944→      const glowGrad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size * 4);
   945→      glowGrad.addColorStop(0, `rgba(255, 210, 100, ${alpha * 0.5})`);
   946→      glowGrad.addColorStop(1, 'rgba(255, 180, 50, 0)');
   947→      ctx.fillStyle = glowGrad;
   948→      ctx.fillRect(p.x - p.size * 4, p.y - p.size * 4, p.size * 8, p.size * 8);
   949→
   950→      // Core
   951→      ctx.beginPath();
   952→      ctx.arc(p.x, p.y, p.size * pulse, 0, Math.PI * 2);
   953→      ctx.fillStyle = `rgba(255, 220, 120, ${alpha})`;
   954→      ctx.fill();
   955→    }
   956→
   957→    _sceneAnimationId = requestAnimationFrame(frame);
   958→  }
   959→  frame();
   960→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   961→}

   963→/** Rain scene — falling rain with lightning. */
   964→function renderRainScene(canvas: HTMLCanvasElement): () => void {
   965→  const ctx = canvas.getContext('2d')!;
   966→  let time = 0;
   967→  let drops: RainDrop[] = [];
   968→  let lightningAlpha = 0;
   969→  let nextLightning = 200 + Math.random() * 400;
   970→
   971→  function init(): void {
   972→    const w = canvas.width;
   973→    const h = canvas.height;
   974→    drops = [];
   975→    for (let i = 0; i < 200; i++) {
   976→      drops.push({
   977→        x: Math.random() * w,
   978→        y: Math.random() * h,
   979→        speed: 8 + Math.random() * 8,
   980→        length: 15 + Math.random() * 20,
   981→        opacity: 0.1 + Math.random() * 0.3,
   982→      });
   983→    }
   984→  }
   985→  init();
   986→
   987→  function frame(): void {
   988→    const w = canvas.width;
   989→    const h = canvas.height;
   990→    time += 0.016;
   991→
   992→    // Stormy sky
   993→    const skyGrad = ctx.createLinearGradient(0, 0, 0, h);
   994→    skyGrad.addColorStop(0, '#0a0c12');
   995→    skyGrad.addColorStop(0.3, '#111520');
   996→    skyGrad.addColorStop(0.6, '#1a1e2e');
   997→    skyGrad.addColorStop(1, '#0d1018');
   998→    ctx.fillStyle = skyGrad;
   999→    ctx.fillRect(0, 0, w, h);
  1000→
  1001→    // Lightning flash
  1002→    if (time * 60 > nextLightning) {
  1003→      lightningAlpha = 0.6;
  1004→      nextLightning = time * 60 + 200 + Math.random() * 500;
  1005→    }
  1006→    if (lightningAlpha > 0) {
  1007→      ctx.fillStyle = `rgba(180, 190, 255, ${lightningAlpha})`;
  1008→      ctx.fillRect(0, 0, w, h);
  1009→      lightningAlpha *= 0.85;
  1010→      if (lightningAlpha < 0.01) lightningAlpha = 0;
  1011→    }
  1012→
  1013→    // Cloud layer
  1014→    for (let i = 0; i < 5; i++) {
  1015→      const cx = (i / 5) * w + Math.sin(time * 0.1 + i * 3) * 40;
  1016→      const cy = h * 0.05 + Math.sin(time * 0.05 + i) * 20;
  1017→      const cw = 200 + Math.sin(i * 7.3) * 80;
  1018→      const ch = 60 + Math.sin(i * 4.1) * 20;
  1019→      const cloudGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, cw * 0.5);
  1020→      cloudGrad.addColorStop(0, 'rgba(30, 35, 50, 0.6)');
  1021→      cloudGrad.addColorStop(1, 'rgba(20, 25, 35, 0)');
  1022→      ctx.fillStyle = cloudGrad;
  1023→      ctx.fillRect(cx - cw, cy - ch, cw * 2, ch * 2);
  1024→    }
  1025→
  1026→    // Rain drops
  1027→    ctx.lineCap = 'round';
  1028→    for (const d of drops) {
  1029→      d.y += d.speed;
  1030→      d.x -= 1; // Slight wind
  1031→
  1032→      if (d.y > h) {
  1033→        d.y = -d.length;
  1034→        d.x = Math.random() * w;
  1035→      }
  1036→      if (d.x < -20) d.x = w + 20;
  1037→
  1038→      ctx.beginPath();
  1039→      ctx.moveTo(d.x, d.y);
  1040→      ctx.lineTo(d.x - 1.5, d.y + d.length);
  1041→      ctx.strokeStyle = `rgba(150, 170, 220, ${d.opacity})`;
  1042→      ctx.lineWidth = 1;
  1043→      ctx.stroke();
  1044→    }
  1045→
  1046→    // Splash effects at bottom
  1047→    for (let i = 0; i < 15; i++) {
  1048→      const sx = (Math.sin(time * 3 + i * 17.3) * 0.5 + 0.5) * w;
  1049→      const sy = h - 10 - Math.random() * 5;
  1050→      const splashAlpha = 0.2 + 0.2 * Math.abs(Math.sin(time * 5 + i * 3.7));
  1051→      ctx.beginPath();
  1052→      ctx.arc(sx, sy, 2, 0, Math.PI * 2);
  1053→      ctx.fillStyle = `rgba(150, 170, 220, ${splashAlpha})`;
  1054→      ctx.fill();
  1055→    }
  1056→
  1057→    _sceneAnimationId = requestAnimationFrame(frame);
  1058→  }
  1059→  frame();
  1060→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
  1061→}

  1409→function handleKeyDown(e: KeyboardEvent): void {
  1410→  if (
  1411→    (e.target as HTMLElement).tagName === 'INPUT' ||
  1412→    (e.target as HTMLElement).tagName === 'TEXTAREA' ||
  1413→    (e.target as HTMLElement).tagName === 'SELECT'
  1414→  )
  1415→    return;
  1416→  // Any key press shows controls
  1417→  showControls();
  1418→  switch (e.key) {
  1419→    case 'Escape':
  1420→      // If in fullscreen, only exit fullscreen — don't close presentation
  1421→      const fsEl = document.fullscreenElement || (document as unknown as { webkitFullscreenElement?: Element }).webkitFullscreenElement;
  1422→      if (fsEl) {
  1423→        e.preventDefault();
  1424→        if (document.exitFullscreen) {
  1425→          document.exitFullscreen().catch(() => {});
  1426→        } else if ((document as unknown as { webkitExitFullscreen?: () => Promise<void> }).webkitExitFullscreen) {
  1427→          (document as unknown as { webkitExitFullscreen: () => Promise<void> }).webkitExitFullscreen().catch(() => {});
  1428→        }
  1429→      } else {
  1430→        closePresentation();
  1431→      }
  1432→      break;
  1433→    case 'ArrowRight':
  1434→    case 'ArrowDown':
  1435→      e.preventDefault();
  1436→      navigateAyah(1);
  1437→      break;
  1438→    case 'ArrowLeft':
  1439→    case 'ArrowUp':
  1440→      e.preventDefault();
  1441→      navigateAyah(-1);
  1442→      break;
  1443→    case ' ':
  1444→      e.preventDefault();
  1445→      togglePlayPause();

  1480→export function initPresentation(): void {
  1481→  console.log('[Presentation] initPresentation() called');
  1482→  injectStyles();
  1483→
  1484→  // Verify the presentation overlay exists in the DOM
  1485→  const overlay = document.getElementById('presentationOverlay');
  1486→  if (!overlay) {
  1487→    console.error('[Presentation] ERROR: presentationOverlay element not found in DOM!');
  1488→    return;
  1489→  }
  1490→  console.log('[Presentation] Overlay element found, binding event handlers...');
  1491→
  1492→  if (dom.presentationCloseBtn) dom.presentationCloseBtn.addEventListener('click', closePresentation);
  1493→  if (dom.presentationPrevBtn) dom.presentationPrevBtn.addEventListener('click', () => navigateAyah(-1));
  1494→  if (dom.presentationNextBtn) dom.presentationNextBtn.addEventListener('click', () => navigateAyah(1));
  1495→  dom.presentationOverlay?.addEventListener('click', (e: MouseEvent) => {
  1496→    // Don't close on background click when in fullscreen
  1497→    const inFs = !!(document.fullscreenElement || (document as unknown as { webkitFullscreenElement?: Element }).webkitFullscreenElement);
  1498→    if (e.target === dom.presentationOverlay && !inFs) closePresentation();
  1499→  });
  1500→
  1501→  // Play/Pause button
  1502→  if (dom.presPlayPauseBtn) {
  1503→    dom.presPlayPauseBtn.addEventListener('click', () => {
  1504→      togglePlayPause();
  1505→      updatePlayPauseBtn();
  1506→      updatePresPlayPauseBtn();
  1507→      showControls(); // Reset auto-hide timer
  1508→    });
  1509→  }
  1510→
  1511→  // Tajweed toggle button (presentation-local)
  1512→  if (dom.presTajweedBtn) {
  1513→    dom.presTajweedBtn.addEventListener('click', () => {
  1514→      togglePresTajweed();
  1515→      showControls(); // Reset auto-hide timer
  1516→    });
  1517→  }
  1518→
  1519→  // Fullscreen toggle button
  1520→  if (dom.presFullscreenBtn) {
  1521→    dom.presFullscreenBtn.addEventListener('click', () => {
  1522→      togglePresFullscreen();
  1523→      showControls(); // Reset auto-hide timer
  1524→    });
  1525→  }
  1526→
  1527→  // Mouse/touch movement shows controls
  1528→  dom.presentationOverlay?.addEventListener('mousemove', handleOverlayMouseMove);
  1529→  dom.presentationOverlay?.addEventListener('touchstart', handleOverlayMouseMove, { passive: true });
  1530→
  1531→  // Fullscreen change event
  1532→  document.addEventListener('fullscreenchange', handleFullscreenChange);
  1533→  document.addEventListener('webkitfullscreenchange', handleFullscreenChange);
  1534→
  1535→  console.log('[Presentation] initPresentation() completed successfully');
  1536→}
