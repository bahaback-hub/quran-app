    40→    40→/** Get a background image based on the current time of day. */
    41→    41→function getAutoBackground(): (typeof NATURE_BACKGROUNDS)[number] {
    42→    42→  const hour = new Date().getHours();
    43→    43→  if (hour >= 4 && hour < 7) return NATURE_BACKGROUNDS[0]; // dawn
    44→    44→  if (hour >= 7 && hour < 12) return NATURE_BACKGROUNDS[1]; // morning clouds
    45→    45→  if (hour >= 12 && hour < 16) return NATURE_BACKGROUNDS[2]; // afternoon mountains
    46→    46→  if (hour >= 16 && hour < 19) return NATURE_BACKGROUNDS[3]; // sunset
    47→    47→  return NATURE_BACKGROUNDS[4]; // night sky
    48→    48→}

   494→   494→function buildAyahHtml(text: string, surahNum: number, ayahNum: number): string {
   495→   495→  let txt = text;
   496→   496→  let offsetAdj = 0;
   497→   497→  if (surahNum !== 1 && ayahNum === 1) {
   498→   498→    const stripped = txt.replace(
   499→   499→      /^ب[\u064B-\u065F\u0670]*س[\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*[هة][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*[نث][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*[يى][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*/u,
   500→   500→      ''
   501→   501→    );
   502→   502→    offsetAdj = txt.length - stripped.length;
   503→   503→    txt = stripped;
   504→   504→  }
   505→   505→  if (!_presTajweedEnabled) return escapeHtml(txt);
   506→   506→  const annotations = getAyahAnnotations(surahNum, ayahNum);
   507→   507→  if (annotations.length === 0) return escapeHtml(txt);
   508→   508→  const adjusted =
   509→   509→    offsetAdj > 0
   510→   510→      ? annotations.map((ann: TajweedAnnotation) => ({
   511→   511→          rule: ann.rule,
   512→   512→          start: ann.start - offsetAdj,
   513→   513→          end: ann.end - offsetAdj,
   514→   514→        }))
   515→   515→      : annotations;
   516→   516→  const colorMap = buildColorMap(adjusted);
   517→   517→  const words = txt.split(/\s+/).filter((w) => w.length > 0);
   518→   518→  let outputPos = 0;
   519→   519→  return words
   520→   520→    .map((word: string, idx: number) => {
   521→   521→      const wordHtml = tajweedColorWord(word, outputPos, colorMap);
   522→   522→      outputPos += word.length + (idx < words.length - 1 ? 1 : 0); // +1 for space between words
   523→   523→      return wordHtml;
   524→   524→    })
   525→   525→    .join(' ');
   526→   526→}

   599→   599→/** Star scene — twinkling stars with occasional shooting stars. */
   600→   600→function renderStarsScene(canvas: HTMLCanvasElement): () => void {
   601→   601→  const ctx = canvas.getContext('2d')!;
   602→   602→  let stars: StarData[] = [];
   603→   603→  let shootingStars: ShootingStar[] = [];
   604→   604→  let time = 0;
   605→   605→
   606→   606→  function init(): void {
   607→   607→    const w = canvas.width;
   608→   608→    const h = canvas.height;
   609→   609→    stars = [];
   610→   610→    for (let i = 0; i < 250; i++) {
   611→   611→      stars.push({
   612→   612→        x: Math.random() * w,
   613→   613→        y: Math.random() * h,
   614→   614→        size: Math.random() * 2.5 + 0.5,
   615→   615→        speed: Math.random() * 0.02 + 0.005,
   616→   616→        brightness: Math.random(),
   617→   617→        phase: Math.random() * Math.PI * 2,
   618→   618→      });
   619→   619→    }
   620→   620→  }
   621→   621→  init();
   622→   622→
   623→   623→  function frame(): void {
   624→   624→    const w = canvas.width;
   625→   625→    const h = canvas.height;
   626→   626→    time += 0.016;
   627→   627→
   628→   628→    // Dark sky gradient
   629→   629→    const grad = ctx.createLinearGradient(0, 0, 0, h);
   630→   630→    grad.addColorStop(0, '#05050f');
   631→   631→    grad.addColorStop(0.5, '#0a0a2e');
   632→   632→    grad.addColorStop(1, '#101030');
   633→   633→    ctx.fillStyle = grad;
   634→   634→    ctx.fillRect(0, 0, w, h);
   635→   635→
   636→   636→    // Milky way band
   637→   637→    const mwGrad = ctx.createLinearGradient(0, h * 0.2, 0, h * 0.6);
   638→   638→    mwGrad.addColorStop(0, 'rgba(60, 50, 80, 0)');
   639→   639→    mwGrad.addColorStop(0.3, 'rgba(80, 60, 120, 0.06)');
   640→   640→    mwGrad.addColorStop(0.5, 'rgba(100, 80, 140, 0.08)');
   641→   641→    mwGrad.addColorStop(0.7, 'rgba(80, 60, 120, 0.06)');
   642→   642→    mwGrad.addColorStop(1, 'rgba(60, 50, 80, 0)');
   643→   643→    ctx.fillStyle = mwGrad;
   644→   644→    ctx.fillRect(0, 0, w, h);
   645→   645→
   646→   646→    // Draw twinkling stars
   647→   647→    for (const star of stars) {
   648→   648→      const twinkle = 0.5 + 0.5 * Math.sin(time * star.speed * 60 + star.phase);
   649→   649→      const alpha = star.brightness * twinkle;
   650→   650→      ctx.beginPath();
   651→   651→      ctx.arc(star.x, star.y, star.size * (0.7 + 0.3 * twinkle), 0, Math.PI * 2);
   652→   652→      ctx.fillStyle = `rgba(255, 255, 240, ${alpha})`;
   653→   653→      ctx.fill();
   654→   654→      // Glow for bigger stars
   655→   655→      if (star.size > 1.8) {
   656→   656→        ctx.beginPath();
   657→   657→        ctx.arc(star.x, star.y, star.size * 3, 0, Math.PI * 2);
   658→   658→        ctx.fillStyle = `rgba(200, 200, 255, ${alpha * 0.1})`;
   659→   659→        ctx.fill();
   660→   660→      }
   661→   661→    }
   662→   662→
   663→   663→    // Shooting stars
   664→   664→    if (Math.random() < 0.003) {
   665→   665→      shootingStars.push({
   666→   666→        x: Math.random() * w * 0.8,
   667→   667→        y: Math.random() * h * 0.3,
   668→   668→        vx: 4 + Math.random() * 4,
   669→   669→        vy: 2 + Math.random() * 2,
   670→   670→        life: 0,
   671→   671→        maxLife: 30 + Math.random() * 30,
   672→   672→        length: 40 + Math.random() * 60,
   673→   673→      });
   674→   674→    }
   675→   675→    for (let i = shootingStars.length - 1; i >= 0; i--) {
   676→   676→      const ss = shootingStars[i];
   677→   677→      ss.x += ss.vx;
   678→   678→      ss.y += ss.vy;
   679→   679→      ss.life++;
   680→   680→      const alpha = 1 - ss.life / ss.maxLife;
   681→   681→      if (alpha <= 0) { shootingStars.splice(i, 1); continue; }
   682→   682→      ctx.beginPath();
   683→   683→      ctx.moveTo(ss.x, ss.y);
   684→   684→      ctx.lineTo(ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6);
   685→   685→      const ssGrad = ctx.createLinearGradient(
   686→   686→        ss.x, ss.y,
   687→   687→        ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6
   688→   688→      );
   689→   689→      ssGrad.addColorStop(0, `rgba(255, 255, 255, ${alpha})`);
   690→   690→      ssGrad.addColorStop(1, `rgba(255, 255, 255, 0)`);
   691→   691→      ctx.strokeStyle = ssGrad;
   692→   692→      ctx.lineWidth = 2;
   693→   693→      ctx.stroke();
   694→   694→    }
   695→   695→
   696→   696→    _sceneAnimationId = requestAnimationFrame(frame);
   697→   697→  }
   698→   698→  frame();
   699→   699→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   700→   700→}

   885→   885→function renderParticlesScene(canvas: HTMLCanvasElement): () => void {
   886→   886→  const ctx = canvas.getContext('2d')!;
   887→   887→  let time = 0;
   888→   888→  let particles: Particle[] = [];
   889→   889→
   890→   890→  function init(): void {
   891→   891→    const w = canvas.width;
   892→   892→    const h = canvas.height;
   893→   893→    particles = [];
   894→   894→    for (let i = 0; i < 120; i++) {
   895→   895→      particles.push({
   896→   896→        x: Math.random() * w,
   897→   897→        y: Math.random() * h,
   898→   898→        vx: (Math.random() - 0.5) * 0.3,
   899→   899→        vy: -Math.random() * 0.5 - 0.1,
   900→   900→        size: Math.random() * 3 + 1,
   901→   901→        opacity: Math.random() * 0.5 + 0.2,
   902→   902→        phase: Math.random() * Math.PI * 2,
   903→   903→      });
   904→   904→    }
   905→   905→  }
   906→   906→  init();
   907→   907→
   908→   908→  function frame(): void {
   909→   909→    const w = canvas.width;
   910→   910→    const h = canvas.height;
   911→   911→    time += 0.016;
   912→   912→
   913→   913→    // Dark warm background
   914→   914→    const bgGrad = ctx.createRadialGradient(w * 0.5, h * 0.4, 0, w * 0.5, h * 0.4, Math.max(w, h) * 0.7);
   915→   915→    bgGrad.addColorStop(0, '#1a1510');
   916→   916→    bgGrad.addColorStop(0.5, '#0f0d08');
   917→   917→    bgGrad.addColorStop(1, '#080604');
   918→   918→    ctx.fillStyle = bgGrad;
   919→   919→    ctx.fillRect(0, 0, w, h);
   920→   920→
   921→   921→    // Warm light source glow
   922→   922→    const lightGlow = ctx.createRadialGradient(w * 0.5, h * 0.3, 0, w * 0.5, h * 0.3, w * 0.4);
   923→   923→    lightGlow.addColorStop(0, 'rgba(255, 200, 100, 0.08)');
   924→   924→    lightGlow.addColorStop(0.5, 'rgba(255, 180, 80, 0.03)');
   925→   925→    lightGlow.addColorStop(1, 'rgba(255, 150, 50, 0)');
   926→   926→    ctx.fillStyle = lightGlow;
   927→   927→    ctx.fillRect(0, 0, w, h);
   928→   928→
   929→   929→    // Draw and update particles
   930→   930→    for (const p of particles) {
   931→   931→      p.x += p.vx + Math.sin(time + p.phase) * 0.2;
   932→   932→      p.y += p.vy;
   933→   933→
   934→   934→      // Wrap around
   935→   935→      if (p.y < -10) { p.y = h + 10; p.x = Math.random() * w; }
   936→   936→      if (p.x < -10) p.x = w + 10;
   937→   937→      if (p.x > w + 10) p.x = -10;
   938→   938→
   939→   939→      const pulse = 0.6 + 0.4 * Math.sin(time * 1.5 + p.phase);
   940→   940→      const alpha = p.opacity * pulse;
   941→   941→
   942→   942→      // Glow
   943→   943→      const glowGrad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size * 4);
   944→   944→      glowGrad.addColorStop(0, `rgba(255, 210, 100, ${alpha * 0.5})`);
   945→   945→      glowGrad.addColorStop(1, 'rgba(255, 180, 50, 0)');
   946→   946→      ctx.fillStyle = glowGrad;
   947→   947→      ctx.fillRect(p.x - p.size * 4, p.y - p.size * 4, p.size * 8, p.size * 8);
   948→   948→
   949→   949→      // Core
   950→   950→      ctx.beginPath();
   951→   951→      ctx.arc(p.x, p.y, p.size * pulse, 0, Math.PI * 2);
   952→   952→      ctx.fillStyle = `rgba(255, 220, 120, ${alpha})`;
   953→   953→      ctx.fill();
   954→   954→    }
   955→   955→
   956→   956→    _sceneAnimationId = requestAnimationFrame(frame);
   957→   957→  }
   958→   958→  frame();
   959→   959→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   960→   960→}

   963→   963→function renderRainScene(canvas: HTMLCanvasElement): () => void {
   964→   964→  const ctx = canvas.getContext('2d')!;
   965→   965→  let time = 0;
   966→   966→  let drops: RainDrop[] = [];

   969→   969→
   970→   970→  function init(): void {
   971→   971→    const w = canvas.width;

   979→   979→        length: 15 + Math.random() * 20,
   980→   980→        opacity: 0.1 + Math.random() * 0.3,

   982→   982→    }
   983→   983→  }
   984→   984→  init();
   985→   985→
   986→   986→  function frame(): void {
   987→   987→    const w = canvas.width;
   988→   988→    const h = canvas.height;
   989→   989→    time += 0.016;
   990→   990→
   991→   991→    // Stormy sky
   992→   992→    const skyGrad = ctx.createLinearGradient(0, 0, 0, h);
   993→   993→    skyGrad.addColorStop(0, '#0a0c12');
   994→   994→    skyGrad.addColorStop(0.3, '#111520');
   995→   995→    skyGrad.addColorStop(0.6, '#1a1e2e');

   999→   999→
  1000→  1000→    // Lightning flash
  1001→  1001→    if (time * 60 > nextLightning) {
  1002→  1002→      lightningAlpha = 0.6;
  1003→  1003→      nextLightning = time * 60 + 200 + Math.random() * 500;
  1004→  1004→    }
  1005→  1005→    if (lightningAlpha > 0) {
  1006→  1006→      ctx.fillStyle = `rgba(180, 190, 255, ${lightningAlpha})`;
  1007→  1007→      ctx.fillRect(0, 0, w, h);
  1008→  1008→      lightningAlpha *= 0.85;
  1009→  1009→      if (lightningAlpha < 0.01) lightningAlpha = 0;
  1010→  1010→    }
  1011→  1011→
  1012→  1012→    // Cloud layer
  1013→  1013→    for (let i = 0; i < 5; i++) {
  1014→  1014→      const cx = (i / 5) * w + Math.sin(time * 0.1 + i * 3) * 40;
  1015→  1015→      const cy = h * 0.05 + Math.sin(time * 0.05 + i) * 20;
  1016→  1016→      const cw = 200 + Math.sin(i * 7.3) * 80;
  1017→  1017→      const ch = 60 + Math.sin(i * 4.1) * 20;
  1018→  1018→      const cloudGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, cw * 0.5);
  1019→  1019→      cloudGrad.addColorStop(0, 'rgba(30, 35, 50, 0.6)');
  1020→  1020→      cloudGrad.addColorStop(1, 'rgba(20, 25, 35, 0)');
  1021→  1021→      ctx.fillStyle = cloudGrad;
  1022→  1022→      ctx.fillRect(cx - cw, cy - ch, cw * 2, ch * 2);
  1023→  1023→    }
  1024→  1024→
  1025→  1025→    // Rain drops
  1026→  1026→    ctx.lineCap = 'round';
  1027→  1027→    for (const d of drops) {
  1028→  1028→      d.y += d.speed;
  1029→  1029→      d.x -= 1; // Slight wind
  1030→  1030→
  1031→  1031→      if (d.y > h) {
  1032→  1032→        d.y = -d.length;
  1033→  1033→        d.x = Math.random() * w;
  1034→  1034→      }
  1035→  1035→      if (d.x < -20) d.x = w + 20;
  1036→  1036→
  1037→  1037→      ctx.beginPath();
  1038→  1038→      ctx.moveTo(d.x, d.y);
  1039→  1039→      ctx.lineTo(d.x - 1.5, d.y + d.length);
  1040→  1040→      ctx.strokeStyle = `rgba(150, 170, 220, ${d.opacity})`;
  1041→  1041→      ctx.lineWidth = 1;
  1042→  1042→      ctx.stroke();
  1043→  1043→    }
  1044→  1044→
  1045→  1045→    // Splash effects at bottom
  1046→  1046→    for (let i = 0; i < 15; i++) {
  1047→  1047→      const sx = (Math.sin(time * 3 + i * 17.3) * 0.5 + 0.5) * w;
  1048→  1048→      const sy = h - 10 - Math.random() * 5;

  1054→  1054→    }
  1055→  1055→

  1057→  1057→  }

  1060→  1060→}

  1363→  1363→function handleKeyDown(e: KeyboardEvent): void {
  1364→  1364→  if (
  1365→  1365→    (e.target as HTMLElement).tagName === 'INPUT' ||
  1366→  1366→    (e.target as HTMLElement).tagName === 'TEXTAREA' ||
  1367→  1367→    (e.target as HTMLElement).tagName === 'SELECT'
  1368→  1368→  )
  1369→  1369→    return;
  1370→  1370→  // Any key press shows controls
  1371→  1371→  showControls();
  1372→  1372→  switch (e.key) {
  1373→  1373→    case 'Escape':
  1374→  1374→      // If in fullscreen, only exit fullscreen — don't close presentation
  1375→  1375→      const fsEl = document.fullscreenElement || (document as unknown as { webkitFullscreenElement?: Element }).webkitFullscreenElement;
  1376→  1376→      if (fsEl) {
  1377→  1377→        e.preventDefault();
  1378→  1378→        if (document.exitFullscreen) {
  1379→  1379→          document.exitFullscreen().catch(() => {});
  1380→  1380→        } else if ((document as unknown as { webkitExitFullscreen?: () => Promise<void> }).webkitExitFullscreen) {
  1381→  1381→          (document as unknown as { webkitExitFullscreen: () => Promise<void> }).webkitExitFullscreen().catch(() => {});
  1382→  1382→        }
  1383→  1383→      } else {
  1384→  1384→        closePresentation();
  1385→  1385→      }
  1386→  1386→      break;
  1387→  1387→    case 'ArrowRight':
  1388→  1388→    case 'ArrowDown':
  1389→  1389→      e.preventDefault();
  1390→  1390→      navigateAyah(1);
  1391→  1391→      break;
  1392→  1392→    case 'ArrowLeft':
  1393→  1393→    case 'ArrowUp':
  1394→  1394→      e.preventDefault();
  1395→  1395→      navigateAyah(-1);
  1396→  1396→      break;
  1397→  1397→    case ' ':
  1398→  1398→      e.preventDefault();
  1399→  1399→      togglePlayPause();
  1400→  1400→      updatePlayPauseBtn();
  1401→  1401→      updatePresPlayPauseBtn();
  1402→  1402→      break;
  1403→  1403→  }
  1404→  1404→}
