    40→/** Get a background image based on the current time of day. */
    41→function getAutoBackground(): (typeof NATURE_BACKGROUNDS)[number] {
    42→  const hour = new Date().getHours();
    43→  if (hour >= 4 && hour < 7) return NATURE_BACKGROUNDS[0]; // dawn
    44→  if (hour >= 7 && hour < 12) return NATURE_BACKGROUNDS[1]; // morning clouds
    45→  if (hour >= 12 && hour < 16) return NATURE_BACKGROUNDS[2]; // afternoon mountains
    46→  if (hour >= 16 && hour < 19) return NATURE_BACKGROUNDS[3]; // sunset
    47→  return NATURE_BACKGROUNDS[4]; // night sky
    48→}
    49→
    50→/** Get a specific nature background by its mood key. */
    51→function getNatureBgByMood(mood: string): (typeof NATURE_BACKGROUNDS)[number] | null {
    52→  return NATURE_BACKGROUNDS.find((bg) => bg.mood === mood) || null;
    53→}

    55→/** Get a random nature background (cycles per ayah). */
    56→function getRandomNatureBg(): (typeof NATURE_BACKGROUNDS)[number] {
    57→  const idx = Math.floor(Math.random() * NATURE_BACKGROUNDS.length);
    58→  return NATURE_BACKGROUNDS[idx];
    59→}

   460→function buildAyahHtml(text: string, surahNum: number, ayahNum: number): string {
   461→  let txt = text;
   462→  let offsetAdj = 0;
   463→  if (surahNum !== 1 && ayahNum === 1) {
   464→    const stripped = txt.replace(
   465→      /^ب[\u064B-\u065F\u0670]*س[\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*[هة][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*[نث][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*[يى][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*/u,
   466→      ''
   467→    );
   468→    offsetAdj = txt.length - stripped.length;
   469→    txt = stripped;
   470→  }
   471→  if (!_presTajweedEnabled) return escapeHtml(txt);
   472→  const annotations = getAyahAnnotations(surahNum, ayahNum);
   473→  if (annotations.length === 0) return escapeHtml(txt);
   474→  const adjusted =
   475→    offsetAdj > 0
   476→      ? annotations.map((ann: TajweedAnnotation) => ({
   477→          rule: ann.rule,
   478→          start: ann.start - offsetAdj,
   479→          end: ann.end - offsetAdj,
   480→        }))
   481→      : annotations;
   482→  const colorMap = buildColorMap(adjusted);
   483→  const words = txt.split(/\s+/).filter((w) => w.length > 0);
   484→  let outputPos = 0;
   485→  return words
   486→    .map((word: string, idx: number) => {
   487→      const wordHtml = tajweedColorWord(word, outputPos, colorMap);
   488→      outputPos += word.length + (idx < words.length - 1 ? 1 : 0); // +1 for space between words
   489→      return wordHtml;
   490→    })

   565→/** Star scene — twinkling stars with occasional shooting stars. */
   566→function renderStarsScene(canvas: HTMLCanvasElement): () => void {
   567→  const ctx = canvas.getContext('2d')!;
   568→  let stars: StarData[] = [];
   569→  let shootingStars: ShootingStar[] = [];
   570→  let time = 0;
   571→
   572→  function init(): void {
   573→    const w = canvas.width;
   574→    const h = canvas.height;
   575→    stars = [];
   576→    for (let i = 0; i < 250; i++) {
   577→      stars.push({
   578→        x: Math.random() * w,
   579→        y: Math.random() * h,
   580→        size: Math.random() * 2.5 + 0.5,
   581→        speed: Math.random() * 0.02 + 0.005,
   582→        brightness: Math.random(),
   583→        phase: Math.random() * Math.PI * 2,
   584→      });
   585→    }
   586→  }
   587→  init();
   588→
   589→  function frame(): void {
   590→    const w = canvas.width;
   591→    const h = canvas.height;
   592→    time += 0.016;
   593→
   594→    // Dark sky gradient
   595→    const grad = ctx.createLinearGradient(0, 0, 0, h);
   596→    grad.addColorStop(0, '#05050f');
   597→    grad.addColorStop(0.5, '#0a0a2e');
   598→    grad.addColorStop(1, '#101030');
   599→    ctx.fillStyle = grad;
   600→    ctx.fillRect(0, 0, w, h);
   601→
   602→    // Milky way band
   603→    const mwGrad = ctx.createLinearGradient(0, h * 0.2, 0, h * 0.6);
   604→    mwGrad.addColorStop(0, 'rgba(60, 50, 80, 0)');
   605→    mwGrad.addColorStop(0.3, 'rgba(80, 60, 120, 0.06)');
   606→    mwGrad.addColorStop(0.5, 'rgba(100, 80, 140, 0.08)');
   607→    mwGrad.addColorStop(0.7, 'rgba(80, 60, 120, 0.06)');
   608→    mwGrad.addColorStop(1, 'rgba(60, 50, 80, 0)');
   609→    ctx.fillStyle = mwGrad;
   610→    ctx.fillRect(0, 0, w, h);
   611→
   612→    // Draw twinkling stars
   613→    for (const star of stars) {
   614→      const twinkle = 0.5 + 0.5 * Math.sin(time * star.speed * 60 + star.phase);
   615→      const alpha = star.brightness * twinkle;
   616→      ctx.beginPath();
   617→      ctx.arc(star.x, star.y, star.size * (0.7 + 0.3 * twinkle), 0, Math.PI * 2);
   618→      ctx.fillStyle = `rgba(255, 255, 240, ${alpha})`;
   619→      ctx.fill();
   620→      // Glow for bigger stars
   621→      if (star.size > 1.8) {
   622→        ctx.beginPath();
   623→        ctx.arc(star.x, star.y, star.size * 3, 0, Math.PI * 2);
   624→        ctx.fillStyle = `rgba(200, 200, 255, ${alpha * 0.1})`;
   625→        ctx.fill();
   626→      }
   627→    }
   628→
   629→    // Shooting stars
   630→    if (Math.random() < 0.003) {
   631→      shootingStars.push({
   632→        x: Math.random() * w * 0.8,
   633→        y: Math.random() * h * 0.3,
   634→        vx: 4 + Math.random() * 4,
   635→        vy: 2 + Math.random() * 2,
   636→        life: 0,
   637→        maxLife: 30 + Math.random() * 30,
   638→        length: 40 + Math.random() * 60,
   639→      });
   640→    }
   641→    for (let i = shootingStars.length - 1; i >= 0; i--) {
   642→      const ss = shootingStars[i];
   643→      ss.x += ss.vx;
   644→      ss.y += ss.vy;
   645→      ss.life++;
   646→      const alpha = 1 - ss.life / ss.maxLife;
   647→      if (alpha <= 0) { shootingStars.splice(i, 1); continue; }
   648→      ctx.beginPath();
   649→      ctx.moveTo(ss.x, ss.y);
   650→      ctx.lineTo(ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6);
   651→      const ssGrad = ctx.createLinearGradient(
   652→        ss.x, ss.y,
   653→        ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6
   654→      );
   655→      ssGrad.addColorStop(0, `rgba(255, 255, 255, ${alpha})`);
   656→      ssGrad.addColorStop(1, `rgba(255, 255, 255, 0)`);
   657→      ctx.strokeStyle = ssGrad;
   658→      ctx.lineWidth = 2;
   659→      ctx.stroke();
   660→    }
   661→
   662→    _sceneAnimationId = requestAnimationFrame(frame);
   663→  }
   664→  frame();
   665→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   666→}

   747→/** Aurora scene — northern lights effect. */
   748→function renderAuroraScene(canvas: HTMLCanvasElement): () => void {
   749→  const ctx = canvas.getContext('2d')!;
   750→  let time = 0;
   751→
   752→  function frame(): void {
   753→    const w = canvas.width;
   754→    const h = canvas.height;
   755→    time += 0.016;

   757→    // Dark sky

   776→

   779→      { r: 50, g: 255, b: 100 },  // green
   780→      { r: 80, g: 200, b: 255 },  // blue
   781→      { r: 150, g: 100, b: 255 }, // purple
   782→      { r: 50, g: 255, b: 180 },  // teal
   783→    ];

   785→    for (let band = 0; band < 4; band++) {
   786→      const color = auroraColors[band];
   787→      const yCenter = h * (0.15 + band * 0.1);

   791→      ctx.moveTo(0, yCenter);
   792→
   793→      for (let x = 0; x <= w; x += 5) {

   796→        const wave3 = Math.sin(x * 0.001 + time * 0.15 + band * 2.0) * 60;
   797→        const y = yCenter + wave1 + wave2 + wave3;
   798→        ctx.lineTo(x, y);

   802→        const wave1 = Math.sin(x * 0.003 + time * 0.3 + band * 1.5) * 40;
   803→        const wave2 = Math.sin(x * 0.006 + time * 0.5 + band * 0.8) * 25;
   804→        const wave3 = Math.sin(x * 0.001 + time * 0.15 + band * 2.0) * 60;

   807→      }
   808→

   812→      aGrad.addColorStop(0, `rgba(${color.r}, ${color.g}, ${color.b}, 0)`);
   813→      aGrad.addColorStop(0.3, `rgba(${color.r}, ${color.g}, ${color.b}, 0.08)`);

   851→function renderParticlesScene(canvas: HTMLCanvasElement): () => void {

   865→        vy: -Math.random() * 0.5 - 0.1,
   866→        size: Math.random() * 3 + 1,

   974→      lightningAlpha *= 0.85;
   975→      if (lightningAlpha < 0.01) lightningAlpha = 0;
   976→    }
   977→
   978→    // Cloud layer
   979→    for (let i = 0; i < 5; i++) {
   980→      const cx = (i / 5) * w + Math.sin(time * 0.1 + i * 3) * 40;
   981→      const cy = h * 0.05 + Math.sin(time * 0.05 + i) * 20;
   982→      const cw = 200 + Math.sin(i * 7.3) * 80;
   983→      const ch = 60 + Math.sin(i * 4.1) * 20;
   984→      const cloudGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, cw * 0.5);
   985→      cloudGrad.addColorStop(0, 'rgba(30, 35, 50, 0.6)');
   986→      cloudGrad.addColorStop(1, 'rgba(20, 25, 35, 0)');
   987→      ctx.fillStyle = cloudGrad;
   988→      ctx.fillRect(cx - cw, cy - ch, cw * 2, ch * 2);
   989→    }
   990→
   991→    // Rain drops
   992→    ctx.lineCap = 'round';
   993→    for (const d of drops) {
   994→      d.y += d.speed;
   995→      d.x -= 1; // Slight wind
   996→
   997→      if (d.y > h) {
   998→        d.y = -d.length;
   999→        d.x = Math.random() * w;
  1000→      }
  1001→      if (d.x < -20) d.x = w + 20;
  1002→
  1003→      ctx.beginPath();
  1004→      ctx.moveTo(d.x, d.y);
  1005→      ctx.lineTo(d.x - 1.5, d.y + d.length);
  1006→      ctx.strokeStyle = `rgba(150, 170, 220, ${d.opacity})`;
  1007→      ctx.lineWidth = 1;
  1008→      ctx.stroke();
  1009→    }
  1010→
  1011→    // Splash effects at bottom
  1012→    for (let i = 0; i < 15; i++) {
  1013→      const sx = (Math.sin(time * 3 + i * 17.3) * 0.5 + 0.5) * w;
  1014→      const sy = h - 10 - Math.random() * 5;
  1015→      const splashAlpha = 0.2 + 0.2 * Math.abs(Math.sin(time * 5 + i * 3.7));
  1016→      ctx.beginPath();
  1017→      ctx.arc(sx, sy, 2, 0, Math.PI * 2);
  1018→      ctx.fillStyle = `rgba(150, 170, 220, ${splashAlpha})`;
  1019→      ctx.fill();
  1020→    }
  1021→
  1022→    _sceneAnimationId = requestAnimationFrame(frame);
  1023→  }
  1024→  frame();
  1025→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
  1026→}

  1311→function handleKeyDown(e: KeyboardEvent): void {
  1312→  if (
  1313→    (e.target as HTMLElement).tagName === 'INPUT' ||
  1314→    (e.target as HTMLElement).tagName === 'TEXTAREA' ||
  1315→    (e.target as HTMLElement).tagName === 'SELECT'
  1316→  )
  1317→    return;
  1318→  // Any key press shows controls
  1319→  showControls();
  1320→  switch (e.key) {
  1321→    case 'Escape':
  1322→      // If in fullscreen, only exit fullscreen — don't close presentation
  1323→      const fsEl = document.fullscreenElement || (document as unknown as { webkitFullscreenElement?: Element }).webkitFullscreenElement;
  1324→      if (fsEl) {
  1325→        e.preventDefault();
  1326→        if (document.exitFullscreen) {
  1327→          document.exitFullscreen().catch(() => {});
  1328→        } else if ((document as unknown as { webkitExitFullscreen?: () => Promise<void> }).webkitExitFullscreen) {
  1329→          (document as unknown as { webkitExitFullscreen: () => Promise<void> }).webkitExitFullscreen().catch(() => {});
  1330→        }
  1331→      } else {
  1332→        closePresentation();
  1333→      }
  1334→      break;
  1335→    case 'ArrowRight':
  1336→    case 'ArrowDown':
  1337→      e.preventDefault();
  1338→      navigateAyah(1);
  1339→      break;
  1340→    case 'ArrowLeft':
  1341→    case 'ArrowUp':
  1342→      e.preventDefault();
  1343→      navigateAyah(-1);
  1344→      break;
  1345→    case ' ':
  1346→      e.preventDefault();
  1347→      togglePlayPause();
  1348→      updatePlayPauseBtn();
  1349→      updatePresPlayPauseBtn();
  1350→      break;
  1351→  }
  1352→}
