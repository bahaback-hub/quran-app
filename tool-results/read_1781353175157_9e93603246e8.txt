    87→/* ===================== IndexedDB SURAH CACHE ===================== */
    88→
    89→function openSurahDB(): Promise<IDBDatabase> {
    90→  return new Promise((resolve, reject) => {
    91→    const request = indexedDB.open('QuranSurahDB', 1);
    92→    request.onupgradeneeded = (e: IDBVersionChangeEvent) => {
    93→      const db = (e.target as IDBOpenDBRequest).result;
    94→      if (!db.objectStoreNames.contains('surahs')) {
    95→        db.createObjectStore('surahs', { keyPath: 'key' });
    96→      }
    97→    };
    98→    request.onsuccess = (e: Event) => resolve((e.target as IDBOpenDBRequest).result);
    99→    request.onerror = () => reject(new Error('Failed to open QuranSurahDB'));
   100→  });
   101→}

   103→async function cacheSurahToIDB(key: string, entry: CachedSurahEntry): Promise<void> {
   104→  try {
   105→    const db = await openSurahDB();
   106→    const tx = db.transaction('surahs', 'readwrite');
   107→    tx.objectStore('surahs').put({ key, ...entry });
   108→  } catch (err) {
   109→    if (import.meta.env.DEV) console.warn('[IDB] Failed to cache surah:', err);
   110→  }
   111→}

   113→async function getCachedSurahFromIDB(key: string): Promise<CachedSurahEntry | null> {
   114→  try {
   115→    const db = await openSurahDB();
   116→    return new Promise((resolve) => {
   117→      const tx = db.transaction('surahs', 'readonly');
   118→      const req = tx.objectStore('surahs').get(key);
   119→      req.onsuccess = () => {
   120→        const result = req.result as { key: string; text: SurahTextData; audios?: (string | null)[]; timings?: number[]; audio?: { ayahs: AyahEntry[] }; translation: Record<string, unknown> | null } | undefined;
   121→        if (!result) { resolve(null); return; }
   122→        // Extract the CachedSurahEntry fields (exclude the 'key' property)
   123→        const { key: _k, ...entry } = result;
   124→        resolve(entry as unknown as CachedSurahEntry);
   125→      };
   126→      req.onerror = () => resolve(null);
   127→    });
   128→  } catch {
   129→    return null;
   130→  }
   131→}

   149→      state.surahList = data.data;
   150→      state.surahOffsets = null;

   152→      populateSurahSelect();
   153→      return;

   155→  } catch (_e) {
   156→    /* fall through to local fallback */
   157→  }

   159→    const localData = await jsonFetch('data/surah-list.json', { silent: true });

   168→    /* no local fallback */
   169→  }
   170→  if (dom.surahSelect) dom.surahSelect.innerHTML = `<option value="">${__('error_unexpected')}</option>`;
   171→  // showToast(__('failed_load_surah'), 'error'); // disabled at startup

   192→export function buildSurahOffsets(): void {
   193→  if (state.surahOffsets || !state.surahList.length) return;
   194→  state.surahOffsets = [];
   195→  let cum = 1;
   196→  const offsets: SurahOffset[] = [];
   197→  for (const s of state.surahList) {
   198→    offsets.push({ surahNum: s.number, startAbs: cum, count: s.numberOfAyahs, name: s.name });
   199→    cum += s.numberOfAyahs;
   200→  }
   201→  state.surahOffsets = offsets;
   202→}

   204→function countArabicChars(text: string): number {
   205→  return (text.match(/[\u0621-\u064A\u0660-\u0669]/g) || []).length;
   206→}

   208→/**
   209→ * Fetch real ayah timings from quran.com API for supported reciters.
   210→ * Returns fractions (0-1) matching the calculateAyahTimings() format,
   211→ * or null if unavailable (triggers character-count fallback).
   212→ */
   213→async function fetchAyahTimings(reciterId: string, surahNum: number, ayahs: AyahEntry[]): Promise<number[] | null> {
   214→  const apiId = getTimingApiId(reciterId);
   215→  if (!apiId) return null;
   216→  try {
   217→    const data = await jsonFetch(
   218→      `https://api.quran.com/api/v4/chapter_recitations/${apiId}/${surahNum}?segments=true`,
   219→      { silent: true, timeout: 8000 }
   220→    );
   221→    if (!data) return null;
   222→    const timestamps: TimestampEntry[] | undefined = data?.audio_file?.timestamps;
   223→    if (!timestamps?.length || timestamps.length !== ayahs.length) return null;
   224→    const totalDuration = timestamps[timestamps.length - 1].timestamp_to;
   225→    if (!totalDuration || totalDuration <= 0) return null;
   226→    return timestamps.map((t: TimestampEntry) => t.timestamp_from / totalDuration);
   227→  } catch {
   228→    return null;
   229→  }
   230→}
   231→
   232→function calculateAyahTimings(ayahs: AyahEntry[], surahNumber: number): number[] {
   233→  const timings: number[] = [];
   234→  const MIN_PER_AYAH = 5;
   235→  const BASMALAH_MIN = 24;
   236→  let basmalahChars = 0;
   237→  const counts = ayahs.map((a: AyahEntry, i: number) => {
   238→    const n = countArabicChars(a.text);
   239→    if (i === 0 && surahNumber !== 1 && surahNumber !== 9) {
   240→      const without = a.text.replace(/^بِسۡمِ ٱللَّهِ ٱلرَّحۡمَـٰنِ ٱلرَّحِيمِ\s*/u, '');
   241→      basmalahChars = Math.max(BASMALAH_MIN, n - countArabicChars(without));
   242→      return Math.max(MIN_PER_AYAH, countArabicChars(without));
   243→    }
   244→    return Math.max(MIN_PER_AYAH, n);
   245→  });
   246→  const total = counts.reduce((a: number, b: number) => a + b, 0) + basmalahChars;
   247→  if (!total) return ayahs.map(() => 0);
   248→  let cum = basmalahChars / total;
   249→  for (let i = 0; i < ayahs.length; i++) {
   250→    timings.push(cum);
   251→    cum += counts[i] / total;
   252→  }
   253→  return timings;
   254→}

   256→interface AbsToSurahAyahResult {
   257→  surahNum: number;
   258→  surahName: string;
   259→  ayahNumInSurah: number;
   260→}
   261→
   262→function absToSurahAyah(absNum: number): AbsToSurahAyahResult | null {
   263→  if (!state.surahOffsets) buildSurahOffsets();
   264→  if (!state.surahOffsets) return null;
   265→  for (const o of state.surahOffsets) {
   266→    if (absNum >= o.startAbs && absNum < o.startAbs + o.count) {
   267→      return { surahNum: o.surahNum, surahName: o.name, ayahNumInSurah: absNum - o.startAbs + 1 };
   268→    }
   269→  }
   270→  return null;
   271→}

   273→function getAbsNumber(surah: number, ayah: number): number | null {
   274→  if (!state.surahOffsets) buildSurahOffsets();
   275→  if (!state.surahOffsets) return null;
   276→  for (const o of state.surahOffsets) {
   277→    if (o.surahNum === surah) return o.startAbs + ayah - 1;
   278→  }
   279→  return null;
   280→}

   285→async function loadMp3quranAudio(
   286→  surahNum: number,
   287→  textData: SurahTextData,
   288→  reciterInfo: ReciterInfo,

   290→): Promise<AudioResult | null> {
   291→  try {
   292→    const audios: (string | null)[] = textData.ayahs.map(() => buildAudioUrl(reciterInfo, surahNum));

   295→      calculateAyahTimings(textData.ayahs, surahNum);
   296→    if (_loadCounter !== currentLoad) return null;
   297→    return { audios, timings };
   298→  } catch (e) {
   299→    console.warn('[Audio] mp3quran load failed:', e);
   300→    return null;
   301→  }

   324→/* ===================== LOAD & RENDER SURAH ===================== */
   325→
   326→let _loadCounter = 0;
   327→let currentSurahController: AbortController | null = null;
   328→/** Separate AbortController for background refresh — not cancelled when loading a new surah. */
   329→let _refreshController: AbortController | null = null;
   330→
   331→/**
   332→ * Load a surah (text + audio + translation), render it, finalize.
   333→ */
   334→export async function loadSurah(surahNum: number, opts: LoadSurahOptions = {}): Promise<void> {
   335→  if (!surahNum) return;
   336→  _loadCounter++;
   337→  const currentLoad = _loadCounter;
   338→  // Clear stale audio/translation before new load (batched to avoid intermediate notifications)
   339→  batch(() => {
   340→    state.loadingSurah = surahNum;
   341→    state.ayahsAudios = [];
   342→    state.ayahTimings = [];
   343→    state.translationData = null;
   344→    state.isPlaying = false;
   345→  });
   346→
   347→  prepareAudioForNewSurah();
   348→
   349→  if (state.hifdhMode) {
   350→    batch(() => {
   351→      state.hifdhMode = false;
   352→      state.repeatMode = false;
   353→      state.repeatCounter = 0;
   354→    });
   355→    dom.hifdhBtn?.classList.remove('active');
   356→    document.querySelectorAll('.ayah').forEach((el) => el.classList.remove('hifdh-mode', 'revealed'));
   357→    if (dom.repeatControls) dom.repeatControls.style.display = 'none';
   358→  }
   359→  if (state.repeatMode && !state.hifdhMode) {
   360→    state.repeatMode = false;
   361→    state.repeatCounter = 0;
   362→    dom.repeatBtn?.classList.remove('active');
   363→    if (dom.repeatControls) dom.repeatControls.style.display = 'none';
   364→  }
   365→  state.currentSurah = surahNum;
   366→
   367→  // Cancel previous in-flight request
   368→  if (currentSurahController) currentSurahController.abort();
   369→  currentSurahController = new AbortController();
   370→  const signal = currentSurahController.signal;
   371→
   372→  const cacheKey = `${surahNum}_${state.currentReciter}_${state.currentTranslation || 'notr'}`;
   373→  const reciterInfo = getReciterById(state.currentReciter) as ReciterInfo;
   374→  const isMp3quran = reciterInfo.source === 'mp3quran';
   375→  if (state.surahCache.has(cacheKey)) {
   376→    const cached = state.surahCache.get(cacheKey) as CachedSurahEntry;
   377→    if (_loadCounter !== currentLoad) return;
   378→    state.surahData = cached.text as any;
   379→    if (isMp3quran) {
   380→      state.ayahsAudios = cached.text.ayahs.map(() => buildAudioUrl(reciterInfo, surahNum) || '');
   381→      state.ayahTimings = cached.timings || calculateAyahTimings(cached.text.ayahs, surahNum);
   382→    } else {
   383→      state.ayahsAudios = Array.isArray(cached.audios)
   384→        ? (cached.audios as string[])
   385→        : cached.audio?.ayahs?.map((a: AyahEntry) => a.audio || '') || [];
   386→      state.ayahTimings = [];
   387→    }
   388→    state.translationData = cached.translation || null;
   389→    renderSurah(cached.text);
   390→    finalizeSurahLoad(opts);
   391→    state.loadingSurah = null;
   392→    return;
   393→  }
   394→
   395→  // Try IndexedDB cache for offline support before fetching from API
   396→  const idbCached = await getCachedSurahFromIDB(cacheKey);
   397→  if (idbCached) {
   398→    if (_loadCounter !== currentLoad) return;
   399→    // Also store in memory cache
   400→    immutableMapSet(state, 'surahCache', cacheKey, idbCached);
   401→    state.surahData = idbCached.text as any;
   402→    if (isMp3quran) {
   403→      state.ayahsAudios = idbCached.text.ayahs.map(() => buildAudioUrl(reciterInfo, surahNum) || '');
   404→      state.ayahTimings = idbCached.timings || calculateAyahTimings(idbCached.text.ayahs, surahNum);
   405→    } else {
   406→      state.ayahsAudios = Array.isArray(idbCached.audios)
   407→        ? (idbCached.audios as string[])
   408→        : idbCached.audio?.ayahs?.map((a: AyahEntry) => a.audio || '') || [];
   409→      state.ayahTimings = [];
   410→    }
   411→    state.translationData = idbCached.translation || null;
   412→    renderSurah(idbCached.text);
   413→    finalizeSurahLoad(opts);
   414→    state.loadingSurah = null;
   415→    // If online, still try to refresh the data in background (stale-while-revalidate)
   416→    // Use a SEPARATE AbortController so the refresh isn't cancelled when the user loads a different surah
   417→    if (navigator.onLine) {
   418→      if (_refreshController) _refreshController.abort();
   419→      _refreshController = new AbortController();
   420→      _refreshSurahFromAPI(surahNum, cacheKey, reciterInfo, isMp3quran, currentLoad, _refreshController.signal);
   421→    }
   422→    return;
   423→  }
   424→
   425→  loadingBar.show(
   426→    `${__('loading_surah')} ${state.surahList.find((s: SurahInfo) => s.number === surahNum)?.name || surahNum}...`
   427→  );
   428→  if (dom.surahContent)
   429→    dom.surahContent.innerHTML =
   430→      '<div class="skeleton-loading"><div class="skeleton-line"></div><div class="skeleton-line"></div><div class="skeleton-line"></div><div class="skeleton-line"></div></div>';
   431→
   432→  try {
   433→    const textJson = await apiFetch(`/surah/${surahNum}/quran-uthmani`, { signal, errorMsg: __('failed_load_surah') });
   434→    const textData: SurahTextData = textJson?.data;
   435→    if (!textData?.ayahs?.length) {
   436→      throw new Error(__('invalid_surah_data'));
   437→    }
   438→    if (_loadCounter !== currentLoad) return;
   439→    state.surahData = textData as any;
   440→
   441→    renderSurah(textData);
   442→    const autoPlay = opts.autoPlay;
   443→    finalizeSurahLoad({ ...opts, autoPlay: false });
   444→    recordReadingSession(surahNum, textData.ayahs.length);
   445→    loadingBar.hide();
   446→
   447→    // Load audio and translation independently (don't block render)
   448→    const audioPromise = isMp3quran
   449→      ? loadMp3quranAudio(surahNum, textData, reciterInfo, currentLoad)
   450→      : loadApiAudio(surahNum, state.currentReciter, currentLoad, signal);
   451→    const transPromise =
   452→      state.translationEnabled && state.currentTranslation
   453→        ? apiFetch(`/surah/${surahNum}/${state.currentTranslation}`, { signal, silent: true })
   454→            .then((d: { data?: Record<string, unknown> }) => d?.data || null)
   455→            .catch(() => null)
   456→        : Promise.resolve(null);
   457→
   458→    const [audioResult, transResult]: [AudioResult | null, Record<string, unknown> | null] = await Promise.all([
   459→      audioPromise,
   460→      transPromise,
   461→    ]);
   462→    if (_loadCounter !== currentLoad) return;
   463→    if (audioResult) {
   464→      state.ayahsAudios = audioResult.audios.filter((a): a is string => a !== null);
   465→      state.ayahTimings = audioResult.timings;
   466→    }
   467→    state.translationData = transResult;
   468→    if (transResult && state.surahData) {
   469→      renderSurah(state.surahData as any);
   470→      highlightCurrentAyah();
   471→    }
   472→
   473→    if (autoPlay && audioResult) {
   474→      playCurrentAyah();
   475→    }
   476→
   477→    if (state.surahCache.size >= CONFIG.CACHE_LIMIT) {
   478→      const firstKey = state.surahCache.keys().next().value;
   479→      immutableMapDelete(state, 'surahCache', firstKey);
   480→    }
   481→    const cacheEntry: CachedSurahEntry = {
   482→      text: textData,
   483→      audios: state.ayahsAudios,
   484→      timings: state.ayahTimings,
   485→      translation: state.translationData,
   486→    };
   487→    immutableMapSet(state, 'surahCache', cacheKey, cacheEntry);
   488→    // Also persist to IndexedDB for offline access
   489→    cacheSurahToIDB(cacheKey, cacheEntry);
   490→  } catch (e: unknown) {
   491→    if ((e as Error).name === 'AbortError') return;
   492→    if (state.fullQuranLoaded && state.fullQuranText) {
   493→      if (_loadCounter !== currentLoad) return;
   494→      const ayahs = state.fullQuranText.filter((a: { surah: number }) => a.surah === surahNum);
   495→      if (ayahs.length) {
   496→        state.surahData = {
   497→          number: surahNum,
   498→          name: ayahs[0].surahName,
   499→          englishName: state.surahList.find((s: SurahInfo) => s.number === surahNum)?.englishName || '',
   500→          ayahs: ayahs.map((a: { ayah: number; text: string }) => ({ numberInSurah: a.ayah, text: a.text })),
   501→        };
   502→        state.ayahsAudios = [];
   503→        renderSurah(state.surahData as any);
   504→        finalizeSurahLoad(opts);
   505→        loadingBar.hide();
   506→        // showToast(__('offline_no_audio'), ''); // disabled at startup
   507→        state.loadingSurah = null;
   508→        return;
   509→      }
   510→    }
   511→    if (dom.surahContent) dom.surahContent.innerHTML = `<p class="error-msg">⚠️ ${__('failed_load_surah')}</p>`;
   512→    // showToast(__('failed_load_surah'), 'error'); // disabled at startup
   513→    loadingBar.hide();
   514→  } finally {
   515→    state.loadingSurah = null;
   516→  }

   519→/**
   520→ * Background refresh: re-fetch surah data from API and update caches.
   521→ * Used when the surah was loaded from IDB cache but we want fresh data.
   522→ */
   523→async function _refreshSurahFromAPI(
   524→  surahNum: number,
   525→  cacheKey: string,
   526→  reciterInfo: ReciterInfo,
   527→  isMp3quran: boolean,
   528→  currentLoad: number,
   529→  signal: AbortSignal
   530→): Promise<void> {
   531→  try {
   532→    const textJson = await apiFetch(`/surah/${surahNum}/quran-uthmani`, { signal, silent: true });
   533→    const textData: SurahTextData = textJson?.data;
   534→    if (!textData?.ayahs?.length || _loadCounter !== currentLoad) return;
   535→
   536→    // Update in-memory state
   537→    state.surahData = textData as any;
   538→
   539→    // Load audio
   540→    const audioPromise = isMp3quran
   541→      ? loadMp3quranAudio(surahNum, textData, reciterInfo, currentLoad)
   542→      : loadApiAudio(surahNum, state.currentReciter, currentLoad, signal);
   543→    const transPromise =
   544→      state.translationEnabled && state.currentTranslation
   545→        ? apiFetch(`/surah/${surahNum}/${state.currentTranslation}`, { signal, silent: true })
   546→            .then((d: { data?: Record<string, unknown> }) => d?.data || null)
   547→            .catch(() => null)
   548→        : Promise.resolve(null);
   549→
   550→    const [audioResult, transResult]: [AudioResult | null, Record<string, unknown> | null] = await Promise.all([
   551→      audioPromise,
   552→      transPromise,
   553→    ]);
   554→    if (_loadCounter !== currentLoad) return;
   555→
   556→    if (audioResult) {
   557→      state.ayahsAudios = audioResult.audios.filter((a): a is string => a !== null);
   558→      state.ayahTimings = audioResult.timings;
   559→    }
   560→    state.translationData = transResult;
   561→
   562→    // Update caches
   563→    const cacheEntry: CachedSurahEntry = {
   564→      text: textData,
   565→      audios: state.ayahsAudios,
   566→      timings: state.ayahTimings,
   567→      translation: state.translationData,
   568→    };
   569→    immutableMapSet(state, 'surahCache', cacheKey, cacheEntry);
   570→    cacheSurahToIDB(cacheKey, cacheEntry);
   571→
   572→    // Re-render with fresh data
   573→    renderSurah(textData);
   574→    highlightCurrentAyah();
   575→  } catch {
   576→    // Background refresh failure is silent — the cached data is still displayed
   577→  }
   578→}

   584→function buildAyahHtml(a: AyahEntry, i: number, textData: SurahTextData): string {
   585→  const isRtlTranslation = state.currentTranslation && state.currentTranslation.startsWith('ur.');
   586→  let txt = a.text;
   587→  let offsetAdj = 0;
   588→  if (textData.number !== 1 && a.numberInSurah === 1) {
   589→    const stripped = txt.replace(
   590→      /^ب[\u064B-\u065F\u0670]*س[\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*[هة][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*[نث][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*[يى][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*/u,
   591→      ''
   592→    );
   593→    offsetAdj = txt.length - stripped.length;
   594→    txt = stripped;
   595→  }
   596→  let colorMap: Map<number, string> | null = null;
   597→  if (state.tajweedEnabled) {
   598→    const annotations: TajweedAnnotation[] = getAyahAnnotations(textData.number, a.numberInSurah);
   599→    if (annotations.length > 0) {
   600→      const adjusted =
   601→        offsetAdj > 0
   602→          ? annotations.map((ann: TajweedAnnotation) => ({
   603→              rule: ann.rule,
   604→              start: ann.start - offsetAdj,
   605→              end: ann.end - offsetAdj,
   606→            }))
   607→          : annotations;
   608→      colorMap = buildColorMap(adjusted);
   609→    }
   610→  }
   611→  let html = '';
   612→  // Juz marker: if this ayah starts a new juz, insert a divider
   613→  const juzNum = isJuzStart(textData.number, a.numberInSurah);
   614→  if (juzNum !== null) {
   615→    html += `<div class="juz-marker"><span class="juz-label">الجزء ${juzNum}</span></div>`;
   616→  }
   617→  html += `<span class="ayah" data-index="${i}" data-surah="${textData.number}" data-ayah="${a.numberInSurah}">`;
   618→  html += buildAyahWordsHtml(txt, i, colorMap);
   619→  // Sajda indicator
   620→  const sajda = isSajdaAyah(textData.number, a.numberInSurah);
   621→  if (sajda.isSajda) {
   622→    const sajdaTitle = sajda.type === 'obligatory' ? 'سجدة واجبة' : 'سجدة مستحبة';
   623→    html += ` <span class="sajda-indicator" title="${sajdaTitle}">۩</span>`;
   624→  }
   625→  html += ` <span class="ayah-number"><span class="ayah-number-inner">${a.numberInSurah}</span></span>`;
   626→  const translationData = state.translationData as { ayahs?: { text: string }[] } | null;
   627→  if (state.translationEnabled && translationData?.ayahs?.[i]) {
   628→    const transText = escapeHtml(translationData.ayahs[i].text);
   629→    html += `<span class="translation-text${isRtlTranslation ? ' rtl-lang' : ''}">${transText}</span>`;
   630→  }
   631→  html += `</span> `;
   632→  return html;
   633→}

   635→function renderAyahChunk(textData: SurahTextData, start: number, count: number): void {
   636→  if (!dom.surahContent) return;
   637→  const ayahsContainer = dom.surahContent.querySelector('.ayahs-container');
   638→  if (!ayahsContainer) return;
   639→  const end = Math.min(start + count, textData.ayahs.length);
   640→  let html = '';
   641→  for (let i = start; i < end; i++) {
   642→    html += buildAyahHtml(textData.ayahs[i], i, textData);
   643→  }
   644→  ayahsContainer.insertAdjacentHTML('beforeend', html);
   645→  _ayahsReadyCount = end;
   646→  if (_ayahsReadyCount < textData.ayahs.length) {
   647→    ensureVirtualSentinel(textData);
   648→  } else {
   649→    cleanupVirtualObserver();
   650→  }
   651→}

   653→function ensureVirtualSentinel(textData: SurahTextData): void {
   654→  cleanupVirtualObserver();
   655→  const existing = document.getElementById('virtualSentinel');
   656→  if (existing) existing.remove();
   657→  const sentinel = document.createElement('div');
   658→  sentinel.id = 'virtualSentinel';
   659→  sentinel.style.height = '1px';
   660→  dom.surahContent?.appendChild(sentinel);
   661→  _virtualObserver = new IntersectionObserver(
   662→    (entries: IntersectionObserverEntry[]) => {
   663→      if (entries[0].isIntersecting && _ayahsReadyCount < textData.ayahs.length) {
   664→        renderAyahChunk(textData, _ayahsReadyCount, VIRTUAL_CHUNK_SIZE);
   665→      }
   666→    },
   667→    { rootMargin: '200px' }
   668→  );
   669→  _virtualObserver.observe(sentinel);
   670→}
   671→
   672→function cleanupVirtualObserver(): void {
   673→  if (_virtualObserver) {
   674→    _virtualObserver.disconnect();
   675→    _virtualObserver = null;
   676→  }
   677→  const existing = document.getElementById('virtualSentinel');
   678→  if (existing) existing.remove();
   679→}

   722→function buildAyahWordsHtml(text: string, ayahIdx: number, colorMap: Map<number, string> | null): string {
   723→  const words = text.split(/\s+/).filter((w) => w.length > 0);
   724→  const useTajweed = state.tajweedEnabled && colorMap;
   725→  let outputPos = 0;
   726→  return words
   727→    .map((word: string, wIdx: number) => {
   728→      const wordHtml = useTajweed ? tajweedColorWord(word, outputPos, colorMap!) : escapeHtml(word);
   729→      outputPos += word.length + (wIdx < words.length - 1 ? 1 : 0); // +1 for space between words
   730→      return `<span class="word" data-ayah-index="${ayahIdx}" data-word-index="${wIdx}">${wordHtml}</span>`;
   731→    })
   732→    .join(' ');
   733→}

   774→export function highlightCurrentAyah(): void {
   775→  const container = dom.surahContent?.querySelector('.ayahs-container');
   776→  const ayahs = container?.children;
   777→  if (ayahs) {
   778→    for (let i = 0; i < ayahs.length; i++) ayahs[i].classList.remove('current');

   781→  if (surahData && state.currentAyahIndex >= _ayahsReadyCount) {
   782→    renderAyahChunk(surahData, _ayahsReadyCount, state.currentAyahIndex - _ayahsReadyCount + VIRTUAL_CHUNK_SIZE);
   783→  }
   784→  const cur = container?.querySelector(`.ayah[data-index="${state.currentAyahIndex}"]`) as HTMLElement | null;
   785→  if (cur) {
   786→    cur.classList.add('current');
   787→    if (state.hifdhMode) {
   788→      for (let i = 0; i < ayahs!.length; i++) ayahs![i].classList.remove('revealed');

   790→        const prev = container!.querySelector(`.ayah[data-index="${i}"]`) as HTMLElement | null;
   791→        if (prev) prev.classList.add('revealed');
   792→      }
   793→    }
   794→    cur.scrollIntoView({ behavior: 'instant', block: 'center' });
   795→  }
   796→  updatePlayerInfo();
   797→  import('./presentation.js').then((m: { syncPresentation: () => void }) => m.syncPresentation()).catch(() => {});
   798→  if (dom.tafsirCurtain && dom.tafsirCurtain.classList.contains('open')) loadTafsirForCurrentAyah();
