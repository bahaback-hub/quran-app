    46→/* ===================== INIT ===================== */
    47→
    48→/** Deep-clone adhkar settings to avoid shallow copy issues with nested objects
    49→ *  (e.g., personal_adhkar array). Uses structuredClone when available,
    50→ *  falls back to JSON round-trip. */
    51→function cloneAdhkarSettings(settings: Record<string, unknown> | null): Record<string, unknown> {
    52→  if (!settings) return {};
    53→  if (typeof structuredClone === 'function') return structuredClone(settings);
    54→  return JSON.parse(JSON.stringify(settings));
    55→}

    57→/** Initialize adhkar state — load settings from storage into state. */
    58→export function initAdhkarState(): void {
    59→  const saved = storage.get<AdhkarSettings>('adhkar_settings');
    60→  if (saved) {
    61→    state.adhkarSettings = saved;
    62→  } else {
    63→    state.adhkarSettings = getDefaultAdhkarSettings();
    64→  }
    65→}
    66→
    67→function getDefaultAdhkarSettings(): AdhkarSettings {
    68→  const settings: AdhkarSettings = {
    69→    adhkar_enabled: false,
    70→    adhkar_sound: false,
    71→    _resetDate: new Date().toDateString(),
    72→    personal_adhkar: [],
    73→  };
    74→  for (const cat of ADHKAR_DATA.categories) {
    75→    (settings as Record<string, unknown>)[cat.id] = {
    76→      enabled: true,
    77→      time: cat.defaultTime || '',
    78→      duration: cat.defaultDuration ?? 1,
    79→    };
    80→    for (const item of cat.items) {
    81→      settings[`item_${item.id}`] = 0;
    82→    }
    83→  }
    84→  settings.personal_adhkar = [];
    85→  return settings;
    86→}

    90→/** Load adhkar settings from storage or set defaults; reset daily counters. */
    91→export function loadAdhkarSettings(): void {
    92→  const saved = storage.get<AdhkarSettings>('adhkar_settings');
    93→  const defaults = getDefaultAdhkarSettings();
    94→  if (saved) {
    95→    for (const key of Object.keys(defaults)) {
    96→      if (saved[key] === undefined) saved[key] = defaults[key];
    97→    }
    98→    state.adhkarSettings = saved as Record<string, unknown>;
    99→  } else {
   100→    state.adhkarSettings = defaults as unknown as Record<string, unknown>;
   101→  }
   102→  // Sync toggle UI with loaded state
   103→  if (dom.adhkarEnabledToggle) {
   104→    dom.adhkarEnabledToggle.classList.toggle('on', !!(state.adhkarSettings as AdhkarSettings).adhkar_enabled);
   105→  }
   106→  if (dom.adhkarSoundToggle) {
   107→    dom.adhkarSoundToggle.classList.toggle('on', !!(state.adhkarSettings as AdhkarSettings).adhkar_sound);
   108→  }
   109→
   110→  const today = new Date().toDateString();
   111→  if ((state.adhkarSettings as AdhkarSettings)._resetDate !== today) {
   112→    for (const key of Object.keys(state.adhkarSettings as AdhkarSettings)) {
   113→      if (key.startsWith('item_')) (state.adhkarSettings as Record<string, unknown>)[key] = 0;
   114→    }
   115→    (state.adhkarSettings as AdhkarSettings)._resetDate = today;
   116→    saveAdhkarSettings();
   117→  }
   118→}

   122→}

   164→function renderAdhkarCategory(categoryId: string): void {
   165→  if (!dom.adhkarContent) return;
   166→  if (categoryId === 'personal') {
   167→    renderPersonalAdhkar();
   168→    return;
   169→  }

   171→  if (!cat) return;
   172→  const settings = state.adhkarSettings as AdhkarSettings;
   173→  const catSettings = (settings[cat.id] as Partial<AdhkarCategorySettings>) || {};
   174→  const enabled = !!catSettings.enabled;
   175→  const notifTime = catSettings.time || cat.defaultTime || '';
   176→  const notifDur = catSettings.duration ?? cat.defaultDuration ?? 1;
   177→  let html = `<div class="adhkar-category-title">${cat.icon} ${cat.name}</div>
   178→    <div class="adhkar-category-options" style="display:flex;gap:10px;align-items:center;flex-wrap:wrap;margin-bottom:12px;padding:8px 12px;background:var(--controls-bg);border-radius:8px;">
   179→      <label style="display:flex;align-items:center;gap:4px;font-size:13px;cursor:pointer;">
   180→        <div class="adhkar-cat-toggle toggle-switch${enabled ? ' on' : ''}" data-category="${cat.id}" role="switch" style="transform:scale(0.85);"></div>
   181→        ${__('adhkar_time')}
   182→      </label>
   183→      <label style="display:flex;align-items:center;gap:4px;font-size:13px;">
   184→        ⏰ <input type="time" class="adhkar-cat-time" data-category="${cat.id}" value="${notifTime}" style="border:1px solid var(--border-soft);border-radius:6px;padding:2px 6px;font-size:12px;font-family:inherit;background:var(--select-bg);color:var(--text-primary);width:80px;">

   186→      <label style="display:flex;align-items:center;gap:4px;font-size:13px;">
   187→        🔔 <input type="number" class="adhkar-cat-duration" data-category="${cat.id}" value="${notifDur}" min="1" max="60" style="border:1px solid var(--border-soft);border-radius:6px;padding:2px 6px;font-size:12px;font-family:inherit;background:var(--select-bg);color:var(--text-primary);width:50px;"> ${__('minutes')}
   188→      </label>
   189→      <span style="font-size:11px;color:var(--text-muted);">${enabled ? __('notification_active') : __('notification_paused')}</span>
   190→    </div>`;
   191→  for (const item of cat.items) {
   192→    const counter = (settings[`item_${item.id}`] as number) || 0;
   193→    const remaining = Math.max(0, item.count - counter);
   194→    const completed = counter >= item.count;
   195→    const pct = Math.min(100, (counter / item.count) * 100);
   196→    html += `<div class="adhkar-item${completed ? ' completed' : ''}" data-item-id="${item.id}" data-category="${categoryId}">
   197→      <div class="adhkar-item-text">${item.text}</div>
   198→      <div class="adhkar-progress-bar"><div class="adhkar-progress-fill" style="width:${pct}%"></div></div>
   199→      <div class="adhkar-item-meta">
   200→        <span class="adhkar-item-count">🔄 ${item.count} ${__('adhkar_times')} — ${__('adhkar_remaining')} ${remaining}</span>
   201→        <span class="adhkar-item-reference">📚 ${item.reference}</span>
   202→        <div class="adhkar-counter">
   203→          <button class="adhkar-counter-btn${completed ? ' completed' : ''}" data-action="increment" data-item-id="${item.id}" data-category="${categoryId}">✓</button>
   204→          <span class="adhkar-counter-text">${counter}</span>
   205→        </div>
   206→      </div>
   207→    </div>`;
   208→  }
   209→  html += `<button class="adhkar-add-btn" data-action="reset" data-category="${categoryId}">${__('adhkar_reset')}</button>`;
   210→  dom.adhkarContent.innerHTML = html;
   211→
   212→  const contentEl = dom.adhkarContent as HTMLElement & { _delegationBound?: boolean };
   213→  if (!contentEl._delegationBound) {
   214→    contentEl._delegationBound = true;
   215→    contentEl.addEventListener('click', (e: MouseEvent) => {
   216→      const target = e.target as HTMLElement;
   217→      const actionBtn = target.closest('[data-action]') as HTMLElement | null;
   218→      if (actionBtn) {
   219→        const action = actionBtn.dataset.action;
   220→        if (action === 'increment') {
   221→          handleAdhkarCounter(actionBtn.dataset.itemId as string, actionBtn.dataset.category as string);
   222→          return;
   223→        }
   224→        if (action === 'reset') {
   225→          resetAdhkarCounters(actionBtn.dataset.category as string);
   226→          return;
   227→        }
   228→      }
   229→      const toggle = target.closest('.adhkar-cat-toggle') as HTMLElement | null;
   230→      if (toggle) {
   231→        const catId = toggle.dataset.category as string;
   232→        const newSettings = cloneAdhkarSettings(state.adhkarSettings);
   233→        const s = newSettings as Record<string, unknown>;
   234→        if (!s[catId]) s[catId] = {} as AdhkarCategorySettings;
   235→        (s[catId] as Partial<AdhkarCategorySettings>).enabled = toggle.classList.toggle('on');
   236→        state.adhkarSettings = newSettings;
   237→        saveAdhkarSettings();
   238→        renderAdhkarCategory(catId);
   239→        return;
   240→      }
   241→    });
   242→    contentEl.addEventListener('change', (e: Event) => {
   243→      const target = e.target as HTMLElement;
   244→      if (target.classList.contains('adhkar-cat-time')) {
   245→        const catId = (target as HTMLElement).dataset.category as string;
   246→        const newSettings = cloneAdhkarSettings(state.adhkarSettings);
   247→        const s = newSettings as Record<string, unknown>;
   248→        if (!s[catId]) s[catId] = {} as AdhkarCategorySettings;
   249→        (s[catId] as Partial<AdhkarCategorySettings>).time = (target as HTMLInputElement).value;
   250→        state.adhkarSettings = newSettings;
   251→        saveAdhkarSettings();
   252→        return;
   253→      }
   254→      if (target.classList.contains('adhkar-cat-duration')) {
   255→        const catId = (target as HTMLElement).dataset.category as string;
   256→        const newSettings = cloneAdhkarSettings(state.adhkarSettings);
   257→        const s = newSettings as Record<string, unknown>;
   258→        if (!s[catId]) s[catId] = {} as AdhkarCategorySettings;
   259→        (s[catId] as Partial<AdhkarCategorySettings>).duration = parseInt((target as HTMLInputElement).value, 10) || 1;
   260→        state.adhkarSettings = newSettings;
   261→        saveAdhkarSettings();
   262→      }
   263→    });
   264→  }
   265→}

   267→function updateAdhkarItemDOM(itemId: string, categoryId: string): void {
   268→  const itemEl = dom.adhkarContent?.querySelector(`.adhkar-item[data-item-id="${itemId}"]`) as HTMLElement | null;
   269→  if (!itemEl) return;
   270→  const cat = ADHKAR_DATA.categories.find((c: { id: string }) => c.id === categoryId);
   271→  if (!cat) return;
   272→  const item = cat.items.find((i: { id: string }) => i.id === itemId);
   273→  if (!item) return;
   274→  const key = `item_${item.id}`;
   275→  const settings = state.adhkarSettings as AdhkarSettings;
   276→  const counter = (settings[key] as number) || 0;
   277→  const remaining = Math.max(0, item.count - counter);
   278→  const completed = counter >= item.count;
   279→  const pct = Math.min(100, (counter / item.count) * 100);
   280→  itemEl.classList.toggle('completed', completed);
   281→  const fill = itemEl.querySelector('.adhkar-progress-fill') as HTMLElement | null;
   282→  if (fill) fill.style.width = pct + '%';
   283→  const countSpan = itemEl.querySelector('.adhkar-item-count') as HTMLElement | null;
   284→  if (countSpan)
   285→    countSpan.textContent = `🔄 ${item.count} ${__('adhkar_times')} — ${__('adhkar_remaining')} ${remaining}`;
   286→  const counterText = itemEl.querySelector('.adhkar-counter-text') as HTMLElement | null;
   287→  if (counterText) counterText.textContent = String(counter);
   288→  const counterBtn = itemEl.querySelector('.adhkar-counter-btn') as HTMLElement | null;
   289→  if (counterBtn) counterBtn.classList.toggle('completed', completed);
   290→}

   292→function handleAdhkarCounter(itemId: string, categoryId: string): void {
   293→  const cat = ADHKAR_DATA.categories.find((c: { id: string }) => c.id === categoryId);
   294→  if (!cat) return;
   295→  const item = cat.items.find((i: { id: string }) => i.id === itemId);
   296→  if (!item) return;
   297→  const key = `item_${item.id}`;
   298→  const current = ((state.adhkarSettings as Record<string, unknown>)[key] as number) || 0;
   299→  // Create new object to trigger Proxy notification
   300→  const newSettings = cloneAdhkarSettings(state.adhkarSettings);
   301→  if (current >= item.count) {
   302→    newSettings[key] = 0;
   303→  } else {
   304→    newSettings[key] = current + 1;
   305→  }
   306→  state.adhkarSettings = newSettings;
   307→  saveAdhkarSettings();
   308→  updateAdhkarItemDOM(itemId, categoryId);
   309→}

   311→function resetAdhkarCounters(categoryId: string): void {
   312→  const cat = ADHKAR_DATA.categories.find((c: { id: string }) => c.id === categoryId);
   313→  if (!cat) return;
   314→  // Create new object to trigger Proxy notification
   315→  const newSettings = cloneAdhkarSettings(state.adhkarSettings);
   316→  for (const item of cat.items) {
   317→    newSettings[`item_${item.id}`] = 0;
   318→    updateAdhkarItemDOM(item.id, categoryId);
   319→  }
   320→  state.adhkarSettings = newSettings;
   321→  saveAdhkarSettings();
   322→  showToast(__('adhkar_reset'), 'success');
   323→}

   426→function savePersonalAdhkar(): void {
   427→  const text = dom.adhkarAddText?.value.trim();
   428→  if (!text) {
   429→    showToast(__('adhkar_enter_text'), 'error');
   430→    return;
   431→  }
   432→  const count = parseInt(dom.adhkarAddCount?.value ?? '1', 10) || 1;
   433→  const time = dom.adhkarAddTime?.value || null;
   434→  const duration = parseInt(dom.adhkarAddDuration?.value ?? '1', 10) || 1;
   435→  const editId = dom.adhkarAddOverlay?.dataset.editId || '';
   436→  const settings = state.adhkarSettings as AdhkarSettings;
   437→  if (!settings.personal_adhkar) settings.personal_adhkar = [];
   438→
   439→  if (editId) {
   440→    const p = settings.personal_adhkar.find((x: PersonalAdhkarEntry) => x.id === editId);
   441→    if (p) {
   442→      p.text = text;
   443→      p.count = count;
   444→      p.time = time;
   445→      p.duration = duration;
   446→    }

   534→let _adhkarAudioCtx: AudioContext | null = null;
   535→function playNotificationSound(): void {
   536→  try {
   537→    if (!_adhkarAudioCtx) _adhkarAudioCtx = new (window.AudioContext || window.webkitAudioContext)();
   538→    const ctx = _adhkarAudioCtx;
   539→    const notes = [523.25, 659.25, 783.99];
   540→    notes.forEach((freq: number, i: number) => {
   541→      const osc = ctx.createOscillator();
   542→      const gain = ctx.createGain();
   543→      osc.type = 'sine';
   544→      osc.frequency.value = freq;
   545→      gain.gain.setValueAtTime(0.2, ctx.currentTime + i * 0.15);
   546→      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + i * 0.15 + 0.4);
   547→      osc.connect(gain);
   548→      gain.connect(ctx.destination);
   549→      osc.start(ctx.currentTime + i * 0.15);
   550→      osc.stop(ctx.currentTime + i * 0.15 + 0.4);
   551→    });
   552→  } catch (e: unknown) {
   553→    console.warn('Notification sound failed:', e);
   554→  }
   555→}

   587→  // this function is called from setInterval (every 30s check) without a
   588→  // user gesture. Modern browsers require a user gesture for requestPermission()

   593→function renderNotifAdhkarText(cat: AdhkarNotifContext): void {
   594→  if (!dom.adhkarNotifText) return;
   595→
   596→  if (cat.id === 'personal') {
   597→    dom.adhkarNotifText.textContent = cat.name || '';
   598→    if (dom.adhkarNotifProgress) dom.adhkarNotifProgress.textContent = '';
   599→    if (dom.adhkarNotifShareBtn)
   600→      dom.adhkarNotifShareBtn.onclick = () => {
   601→        copyToClipboard(cat.name || '');
   602→        showToast(__('copied'), 'success');
   603→      };
   604→    return;
   605→  }
   606→
   607→  const category = ADHKAR_DATA.categories.find((c: { id: string }) => c.id === cat.id);
   608→  if (!category) return;
   609→
   610→  let text = '';
   611→  let count = 0;
   612→  let total = 0;
   613→  const settings = state.adhkarSettings as AdhkarSettings;
   614→  for (const item of category.items) {
   615→    total++;
   616→    const key = `item_${item.id}`;
   617→    const counter = (settings[key] as number) || 0;
   618→    if (counter < item.count) {
   619→      if (text) text += '\n\n';
   620→      text += item.text;
   621→      count++;
   622→    }
   623→  }
   624→
   625→  if (!text) {
   626→    dom.adhkarNotifText.textContent = __('adhkar_done');
   627→    if (dom.adhkarNotifProgress) dom.adhkarNotifProgress.textContent = '';
   628→    if (dom.adhkarNotifShareBtn) dom.adhkarNotifShareBtn.style.display = 'none';
   629→  } else {
   630→    dom.adhkarNotifText.textContent = text;
   631→    if (dom.adhkarNotifProgress) dom.adhkarNotifProgress.textContent = `📖 ${count}/${total}`;
   632→    if (dom.adhkarNotifShareBtn) {
   633→      dom.adhkarNotifShareBtn.style.display = 'inline-block';
   634→      dom.adhkarNotifShareBtn.onclick = () => {
   635→        copyToClipboard(text);
   636→        showToast(__('copied'), 'success');
   637→      };
   638→    }
   639→  }
   640→}

   653→/** Check all adhkar categories and personal adhkar for pending notifications. */
   654→export function checkAdhkarNotifications(): void {
   655→  const settings = state.adhkarSettings as AdhkarSettings | null;
   656→  if (!settings?.adhkar_enabled) return;
   657→  const now = new Date();
   658→  const curMin = now.getHours() * 60 + now.getMinutes();
   659→  const today = now.toDateString();
   660→
   661→  for (const cat of ADHKAR_DATA.categories) {
   662→    const catSettings = settings[cat.id] as Partial<AdhkarCategorySettings> | undefined;
   663→    if (!catSettings?.enabled) continue;
   664→    const notifTime = catSettings.time || cat.defaultTime;
   665→    if (!notifTime) continue;
   666→    const [h, m] = notifTime.split(':').map(Number);
   667→    const catMin = h * 60 + m;
   668→    const fireKey = cat.id + '_' + today;
   669→    if (curMin >= catMin && state.lastAdhkarFired !== fireKey) {
   670→      state.lastAdhkarFired = fireKey;
   671→      const notifDuration = catSettings.duration ?? cat.defaultDuration ?? 1;
   672→      showAdhkarNotification({ id: cat.id, icon: cat.icon, name: cat.name }, notifDuration);
   673→      return;
   674→    }
   675→  }
   676→
   677→  for (const p of settings.personal_adhkar || []) {
   678→    if (!p.time) continue;
   679→    const [h, m] = p.time.split(':').map(Number);
   680→    const pMin = h * 60 + m;
   681→    const fireKey = 'personal_' + p.id + '_' + today;
   682→    if (curMin >= pMin && state.lastAdhkarFired !== fireKey) {
   683→      state.lastAdhkarFired = fireKey;
   684→      showAdhkarNotification({
   685→        id: 'personal',
   686→        _personalId: p.id,
   687→        icon: '📝',
   688→        name: p.text,
   689→        duration: p.duration || 1,
   690→      });
   691→      return;
   692→    }
   693→  }
   694→}
