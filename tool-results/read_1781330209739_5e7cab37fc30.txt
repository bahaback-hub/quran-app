    40→/** Get a background image based on the current time of day. */
    41→function getAutoBackground(): (typeof NATURE_BACKGROUNDS)[number] {
    42→  const hour = new Date().getHours();
    43→  if (hour >= 4 && hour < 7) return NATURE_BACKGROUNDS[0]; // dawn
    44→  if (hour >= 7 && hour < 12) return NATURE_BACKGROUNDS[1]; // morning clouds
    45→  if (hour >= 12 && hour < 16) return NATURE_BACKGROUNDS[2]; // afternoon mountains
    46→  if (hour >= 16 && hour < 19) return NATURE_BACKGROUNDS[3]; // sunset
    47→  return NATURE_BACKGROUNDS[4]; // night sky
    48→}
    49→
    50→/** Get a specific nature background by its mood key. */
    51→function getNatureBgByMood(mood: string): (typeof NATURE_BACKGROUNDS)[number] | null {
    52→  return NATURE_BACKGROUNDS.find((bg) => bg.mood === mood) || null;
    53→}

    55→/** Get a random nature background (cycles per ayah). */
    56→function getRandomNatureBg(): (typeof NATURE_BACKGROUNDS)[number] {
    57→  const idx = Math.floor(Math.random() * NATURE_BACKGROUNDS.length);
    58→  return NATURE_BACKGROUNDS[idx];
    59→}

   494→function buildAyahHtml(text: string, surahNum: number, ayahNum: number): string {
   495→  let txt = text;
   496→  let offsetAdj = 0;
   497→  if (surahNum !== 1 && ayahNum === 1) {
   498→    const stripped = txt.replace(
   499→      /^ب[\u064B-\u065F\u0670]*س[\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*[هة][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*[نث][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*[يى][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*/u,
   500→      ''
   501→    );
   502→    offsetAdj = txt.length - stripped.length;
   503→    txt = stripped;
   504→  }
   505→  if (!_presTajweedEnabled) return escapeHtml(txt);
   506→  const annotations = getAyahAnnotations(surahNum, ayahNum);
   507→  if (annotations.length === 0) return escapeHtml(txt);
   508→  const adjusted =
   509→    offsetAdj > 0
   510→      ? annotations.map((ann: TajweedAnnotation) => ({
   511→          rule: ann.rule,
   512→          start: ann.start - offsetAdj,
   513→          end: ann.end - offsetAdj,
   514→        }))
   515→      : annotations;
   516→  const colorMap = buildColorMap(adjusted);
   517→  const words = txt.split(/\s+/).filter((w) => w.length > 0);
   518→  let outputPos = 0;
   519→  return words
   520→    .map((word: string, idx: number) => {
   521→      const wordHtml = tajweedColorWord(word, outputPos, colorMap);
   522→      outputPos += word.length + (idx < words.length - 1 ? 1 : 0); // +1 for space between words
   523→      return wordHtml;
   524→    })
   525→    .join(' ');
   526→}

   599→/** Star scene — twinkling stars with occasional shooting stars. */
   600→function renderStarsScene(canvas: HTMLCanvasElement): () => void {
   601→  const ctx = canvas.getContext('2d')!;
   602→  let stars: StarData[] = [];
   603→  let shootingStars: ShootingStar[] = [];
   604→  let time = 0;
   605→
   606→  function init(): void {
   607→    const w = canvas.width;
   608→    const h = canvas.height;
   609→    stars = [];
   610→    for (let i = 0; i < 250; i++) {
   611→      stars.push({
   612→        x: Math.random() * w,
   613→        y: Math.random() * h,
   614→        size: Math.random() * 2.5 + 0.5,
   615→        speed: Math.random() * 0.02 + 0.005,
   616→        brightness: Math.random(),
   617→        phase: Math.random() * Math.PI * 2,
   618→      });
   619→    }
   620→  }
   621→  init();
   622→
   623→  function frame(): void {
   624→    const w = canvas.width;
   625→    const h = canvas.height;
   626→    time += 0.016;
   627→
   628→    // Dark sky gradient
   629→    const grad = ctx.createLinearGradient(0, 0, 0, h);
   630→    grad.addColorStop(0, '#05050f');
   631→    grad.addColorStop(0.5, '#0a0a2e');
   632→    grad.addColorStop(1, '#101030');
   633→    ctx.fillStyle = grad;
   634→    ctx.fillRect(0, 0, w, h);
   635→
   636→    // Milky way band
   637→    const mwGrad = ctx.createLinearGradient(0, h * 0.2, 0, h * 0.6);
   638→    mwGrad.addColorStop(0, 'rgba(60, 50, 80, 0)');
   639→    mwGrad.addColorStop(0.3, 'rgba(80, 60, 120, 0.06)');
   640→    mwGrad.addColorStop(0.5, 'rgba(100, 80, 140, 0.08)');
   641→    mwGrad.addColorStop(0.7, 'rgba(80, 60, 120, 0.06)');
   642→    mwGrad.addColorStop(1, 'rgba(60, 50, 80, 0)');
   643→    ctx.fillStyle = mwGrad;
   644→    ctx.fillRect(0, 0, w, h);
   645→
   646→    // Draw twinkling stars
   647→    for (const star of stars) {
   648→      const twinkle = 0.5 + 0.5 * Math.sin(time * star.speed * 60 + star.phase);
   649→      const alpha = star.brightness * twinkle;
   650→      ctx.beginPath();
   651→      ctx.arc(star.x, star.y, star.size * (0.7 + 0.3 * twinkle), 0, Math.PI * 2);
   652→      ctx.fillStyle = `rgba(255, 255, 240, ${alpha})`;
   653→      ctx.fill();
   654→      // Glow for bigger stars
   655→      if (star.size > 1.8) {
   656→        ctx.beginPath();
   657→        ctx.arc(star.x, star.y, star.size * 3, 0, Math.PI * 2);
   658→        ctx.fillStyle = `rgba(200, 200, 255, ${alpha * 0.1})`;
   659→        ctx.fill();
   660→      }
   661→    }
   662→
   663→    // Shooting stars
   664→    if (Math.random() < 0.003) {
   665→      shootingStars.push({
   666→        x: Math.random() * w * 0.8,
   667→        y: Math.random() * h * 0.3,
   668→        vx: 4 + Math.random() * 4,
   669→        vy: 2 + Math.random() * 2,
   670→        life: 0,
   671→        maxLife: 30 + Math.random() * 30,
   672→        length: 40 + Math.random() * 60,
   673→      });
   674→    }
   675→    for (let i = shootingStars.length - 1; i >= 0; i--) {
   676→      const ss = shootingStars[i];
   677→      ss.x += ss.vx;
   678→      ss.y += ss.vy;
   679→      ss.life++;
   680→      const alpha = 1 - ss.life / ss.maxLife;
   681→      if (alpha <= 0) { shootingStars.splice(i, 1); continue; }
   682→      ctx.beginPath();
   683→      ctx.moveTo(ss.x, ss.y);
   684→      ctx.lineTo(ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6);
   685→      const ssGrad = ctx.createLinearGradient(
   686→        ss.x, ss.y,
   687→        ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6
   688→      );
   689→      ssGrad.addColorStop(0, `rgba(255, 255, 255, ${alpha})`);
   690→      ssGrad.addColorStop(1, `rgba(255, 255, 255, 0)`);
   691→      ctx.strokeStyle = ssGrad;
   692→      ctx.lineWidth = 2;
   693→      ctx.stroke();
   694→    }
   695→
   696→    _sceneAnimationId = requestAnimationFrame(frame);
   697→  }
   698→  frame();
   699→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   700→}

   704→  const ctx = canvas.getContext('2d')!;

   706→
   707→  const waveLayers: WaveLayer[] = [

   713→    { amplitude: 8, frequency: 0.03, speed: 1.6, yBase: 0.78, color: 'rgba(35, 150, 220, 0.85)' },

   717→  function frame(): void {
   718→    const w = canvas.width;
   719→    const h = canvas.height;

   732→    // Moon glow
   733→    const moonX = w * 0.75;
   734→    const moonY = h * 0.15;
   735→    const moonGlow = ctx.createRadialGradient(moonX, moonY, 0, moonX, moonY, 120);
   736→    moonGlow.addColorStop(0, 'rgba(255, 255, 200, 0.3)');

   766→      const fx = (i / 30) * w + Math.sin(time * 0.5 + i) * 20;
   767→      const fy = h * 0.44 + Math.sin(fx * 0.008 + time * 0.4) * 25 - 5;
   768→      const sparkle = 0.3 + 0.7 * Math.abs(Math.sin(time * 2 + i * 0.5));

   770→      ctx.arc(fx, fy, 1.5, 0, Math.PI * 2);
   771→      ctx.fillStyle = `rgba(255, 255, 255, ${sparkle * 0.5})`;
   772→      ctx.fill();
   773→    }
   774→

   776→  }
   777→  frame();
   778→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   779→}

   782→function renderAuroraScene(canvas: HTMLCanvasElement): () => void {

   784→  let time = 0;

   786→  function frame(): void {

   794→    skyGrad.addColorStop(0.4, '#050d1a');

   803→      const sy = (Math.sin(i * 269.5 + 183.3) * 0.5 + 0.5) * h * 0.6;
   804→      const twinkle = 0.3 + 0.7 * Math.abs(Math.sin(time * 0.5 + i));
   805→      ctx.beginPath();
   806→      ctx.arc(sx, sy, 1, 0, Math.PI * 2);
   807→      ctx.fillStyle = `rgba(255, 255, 240, ${twinkle * 0.6})`;
   808→      ctx.fill();
   809→    }
   810→

   813→      { r: 50, g: 255, b: 100 },  // green
   814→      { r: 80, g: 200, b: 255 },  // blue
   815→      { r: 150, g: 100, b: 255 }, // purple
   816→      { r: 50, g: 255, b: 180 },  // teal
   817→    ];

   819→    for (let band = 0; band < 4; band++) {
   820→      const color = auroraColors[band];
   821→      const yCenter = h * (0.15 + band * 0.1);
   822→      const bandWidth = h * 0.15;
   823→
   824→      ctx.beginPath();
   825→      ctx.moveTo(0, yCenter);
   826→
   827→      for (let x = 0; x <= w; x += 5) {

   829→        const wave2 = Math.sin(x * 0.006 + time * 0.5 + band * 0.8) * 25;
   830→        const wave3 = Math.sin(x * 0.001 + time * 0.15 + band * 2.0) * 60;
   831→        const y = yCenter + wave1 + wave2 + wave3;
   832→        ctx.lineTo(x, y);
   833→      }
   834→
   835→      for (let x = w; x >= 0; x -= 5) {
   836→        const wave1 = Math.sin(x * 0.003 + time * 0.3 + band * 1.5) * 40;
   837→        const wave2 = Math.sin(x * 0.006 + time * 0.5 + band * 0.8) * 25;
   838→        const wave3 = Math.sin(x * 0.001 + time * 0.15 + band * 2.0) * 60;

   884→/** Golden particles scene — floating dust motes. */
   885→function renderParticlesScene(canvas: HTMLCanvasElement): () => void {
   886→  const ctx = canvas.getContext('2d')!;
   887→  let time = 0;
   888→  let particles: Particle[] = [];
   889→
   890→  function init(): void {
   891→    const w = canvas.width;
   892→    const h = canvas.height;
   893→    particles = [];
   894→    for (let i = 0; i < 120; i++) {
   895→      particles.push({
   896→        x: Math.random() * w,
   897→        y: Math.random() * h,
   898→        vx: (Math.random() - 0.5) * 0.3,
   899→        vy: -Math.random() * 0.5 - 0.1,
   900→        size: Math.random() * 3 + 1,
   901→        opacity: Math.random() * 0.5 + 0.2,
   902→        phase: Math.random() * Math.PI * 2,
   903→      });
   904→    }
   905→  }
   906→  init();
   907→
   908→  function frame(): void {
   909→    const w = canvas.width;
   910→    const h = canvas.height;
   911→    time += 0.016;
   912→
   913→    // Dark warm background
   914→    const bgGrad = ctx.createRadialGradient(w * 0.5, h * 0.4, 0, w * 0.5, h * 0.4, Math.max(w, h) * 0.7);
   915→    bgGrad.addColorStop(0, '#1a1510');
   916→    bgGrad.addColorStop(0.5, '#0f0d08');
   917→    bgGrad.addColorStop(1, '#080604');
   918→    ctx.fillStyle = bgGrad;
   919→    ctx.fillRect(0, 0, w, h);
   920→
   921→    // Warm light source glow
   922→    const lightGlow = ctx.createRadialGradient(w * 0.5, h * 0.3, 0, w * 0.5, h * 0.3, w * 0.4);
   923→    lightGlow.addColorStop(0, 'rgba(255, 200, 100, 0.08)');
   924→    lightGlow.addColorStop(0.5, 'rgba(255, 180, 80, 0.03)');
   925→    lightGlow.addColorStop(1, 'rgba(255, 150, 50, 0)');
   926→    ctx.fillStyle = lightGlow;
   927→    ctx.fillRect(0, 0, w, h);
   928→
   929→    // Draw and update particles
   930→    for (const p of particles) {
   931→      p.x += p.vx + Math.sin(time + p.phase) * 0.2;
   932→      p.y += p.vy;
   933→
   934→      // Wrap around
   935→      if (p.y < -10) { p.y = h + 10; p.x = Math.random() * w; }
   936→      if (p.x < -10) p.x = w + 10;
   937→      if (p.x > w + 10) p.x = -10;
   938→
   939→      const pulse = 0.6 + 0.4 * Math.sin(time * 1.5 + p.phase);
   940→      const alpha = p.opacity * pulse;
   941→
   942→      // Glow
   943→      const glowGrad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size * 4);
   944→      glowGrad.addColorStop(0, `rgba(255, 210, 100, ${alpha * 0.5})`);
   945→      glowGrad.addColorStop(1, 'rgba(255, 180, 50, 0)');
   946→      ctx.fillStyle = glowGrad;
   947→      ctx.fillRect(p.x - p.size * 4, p.y - p.size * 4, p.size * 8, p.size * 8);
   948→
   949→      // Core
   950→      ctx.beginPath();
   951→      ctx.arc(p.x, p.y, p.size * pulse, 0, Math.PI * 2);
   952→      ctx.fillStyle = `rgba(255, 220, 120, ${alpha})`;
   953→      ctx.fill();
   954→    }
   955→
   956→    _sceneAnimationId = requestAnimationFrame(frame);
   957→  }
   958→  frame();
   959→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   960→}

   962→/** Rain scene — falling rain with lightning. */
   963→function renderRainScene(canvas: HTMLCanvasElement): () => void {
   964→  const ctx = canvas.getContext('2d')!;
   965→  let time = 0;
   966→  let drops: RainDrop[] = [];
   967→  let lightningAlpha = 0;
   968→  let nextLightning = 200 + Math.random() * 400;
   969→
   970→  function init(): void {
   971→    const w = canvas.width;
   972→    const h = canvas.height;
   973→    drops = [];
   974→    for (let i = 0; i < 200; i++) {
   975→      drops.push({
   976→        x: Math.random() * w,
   977→        y: Math.random() * h,
   978→        speed: 8 + Math.random() * 8,
   979→        length: 15 + Math.random() * 20,
   980→        opacity: 0.1 + Math.random() * 0.3,
   981→      });
   982→    }
   983→  }
   984→  init();
   985→
   986→  function frame(): void {
   987→    const w = canvas.width;
   988→    const h = canvas.height;
   989→    time += 0.016;
   990→
   991→    // Stormy sky
   992→    const skyGrad = ctx.createLinearGradient(0, 0, 0, h);
   993→    skyGrad.addColorStop(0, '#0a0c12');
   994→    skyGrad.addColorStop(0.3, '#111520');
   995→    skyGrad.addColorStop(0.6, '#1a1e2e');
   996→    skyGrad.addColorStop(1, '#0d1018');
   997→    ctx.fillStyle = skyGrad;
   998→    ctx.fillRect(0, 0, w, h);
   999→
  1000→    // Lightning flash
  1001→    if (time * 60 > nextLightning) {
  1002→      lightningAlpha = 0.6;
  1003→      nextLightning = time * 60 + 200 + Math.random() * 500;
  1004→    }
  1005→    if (lightningAlpha > 0) {
  1006→      ctx.fillStyle = `rgba(180, 190, 255, ${lightningAlpha})`;
  1007→      ctx.fillRect(0, 0, w, h);
  1008→      lightningAlpha *= 0.85;
  1009→      if (lightningAlpha < 0.01) lightningAlpha = 0;
  1010→    }
  1011→
  1012→    // Cloud layer
  1013→    for (let i = 0; i < 5; i++) {
  1014→      const cx = (i / 5) * w + Math.sin(time * 0.1 + i * 3) * 40;
  1015→      const cy = h * 0.05 + Math.sin(time * 0.05 + i) * 20;
  1016→      const cw = 200 + Math.sin(i * 7.3) * 80;
  1017→      const ch = 60 + Math.sin(i * 4.1) * 20;
  1018→      const cloudGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, cw * 0.5);
  1019→      cloudGrad.addColorStop(0, 'rgba(30, 35, 50, 0.6)');
  1020→      cloudGrad.addColorStop(1, 'rgba(20, 25, 35, 0)');
  1021→      ctx.fillStyle = cloudGrad;
  1022→      ctx.fillRect(cx - cw, cy - ch, cw * 2, ch * 2);
  1023→    }
  1024→
  1025→    // Rain drops
  1026→    ctx.lineCap = 'round';
  1027→    for (const d of drops) {
  1028→      d.y += d.speed;
  1029→      d.x -= 1; // Slight wind
  1030→
  1031→      if (d.y > h) {
  1032→        d.y = -d.length;
  1033→        d.x = Math.random() * w;
  1034→      }
  1035→      if (d.x < -20) d.x = w + 20;
  1036→
  1037→      ctx.beginPath();
  1038→      ctx.moveTo(d.x, d.y);
  1039→      ctx.lineTo(d.x - 1.5, d.y + d.length);
  1040→      ctx.strokeStyle = `rgba(150, 170, 220, ${d.opacity})`;
  1041→      ctx.lineWidth = 1;
  1042→      ctx.stroke();
  1043→    }
  1044→
  1045→    // Splash effects at bottom
  1046→    for (let i = 0; i < 15; i++) {
  1047→      const sx = (Math.sin(time * 3 + i * 17.3) * 0.5 + 0.5) * w;
  1048→      const sy = h - 10 - Math.random() * 5;
  1049→      const splashAlpha = 0.2 + 0.2 * Math.abs(Math.sin(time * 5 + i * 3.7));
  1050→      ctx.beginPath();
  1051→      ctx.arc(sx, sy, 2, 0, Math.PI * 2);
  1052→      ctx.fillStyle = `rgba(150, 170, 220, ${splashAlpha})`;
  1053→      ctx.fill();
  1054→    }
  1055→
  1056→    _sceneAnimationId = requestAnimationFrame(frame);
  1057→  }
  1058→  frame();
  1059→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
  1060→}

  1363→function handleKeyDown(e: KeyboardEvent): void {
  1364→  if (
  1365→    (e.target as HTMLElement).tagName === 'INPUT' ||
  1366→    (e.target as HTMLElement).tagName === 'TEXTAREA' ||
  1367→    (e.target as HTMLElement).tagName === 'SELECT'
  1368→  )
  1369→    return;
  1370→  // Any key press shows controls
  1371→  showControls();
  1372→  switch (e.key) {
  1373→    case 'Escape':
  1374→      // If in fullscreen, only exit fullscreen — don't close presentation
  1375→      const fsEl = document.fullscreenElement || (document as unknown as { webkitFullscreenElement?: Element }).webkitFullscreenElement;
  1376→      if (fsEl) {
  1377→        e.preventDefault();
  1378→        if (document.exitFullscreen) {
  1379→          document.exitFullscreen().catch(() => {});
  1380→        } else if ((document as unknown as { webkitExitFullscreen?: () => Promise<void> }).webkitExitFullscreen) {
  1381→          (document as unknown as { webkitExitFullscreen: () => Promise<void> }).webkitExitFullscreen().catch(() => {});
  1382→        }
  1383→      } else {
  1384→        closePresentation();
  1385→      }
  1386→      break;
  1387→    case 'ArrowRight':
  1388→    case 'ArrowDown':
  1389→      e.preventDefault();
  1390→      navigateAyah(1);
  1391→      break;
  1392→    case 'ArrowLeft':
  1393→    case 'ArrowUp':
  1394→      e.preventDefault();
  1395→      navigateAyah(-1);
  1396→      break;
  1397→    case ' ':
  1398→      e.preventDefault();
  1399→      togglePlayPause();
  1400→      updatePlayPauseBtn();
  1401→      updatePresPlayPauseBtn();
  1402→      break;
  1403→  }
  1404→}
