    41→/** Get a background image based on the current time of day. */
    42→function getAutoBackground(): (typeof NATURE_BACKGROUNDS)[number] {
    43→  const hour = new Date().getHours();
    44→  if (hour >= 4 && hour < 7) return NATURE_BACKGROUNDS[0]; // dawn
    45→  if (hour >= 7 && hour < 12) return NATURE_BACKGROUNDS[1]; // morning clouds
    46→  if (hour >= 12 && hour < 16) return NATURE_BACKGROUNDS[2]; // afternoon mountains
    47→  if (hour >= 16 && hour < 19) return NATURE_BACKGROUNDS[3]; // sunset
    48→  return NATURE_BACKGROUNDS[4]; // night sky
    49→}
    50→
    51→/** Get a specific nature background by its mood key. */
    52→function getNatureBgByMood(mood: string): (typeof NATURE_BACKGROUNDS)[number] | null {
    53→  return NATURE_BACKGROUNDS.find((bg) => bg.mood === mood) || null;
    54→}

    59→  return NATURE_BACKGROUNDS[idx];

   498→function buildAyahHtml(text: string, surahNum: number, ayahNum: number): string {
   499→  let txt = text;
   500→  let offsetAdj = 0;
   501→  if (surahNum !== 1 && ayahNum === 1) {
   502→    const stripped = txt.replace(
   503→      /^ب[\u064B-\u065F\u0670]*س[\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*[هة][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*[نث][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*[يى][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*/u,
   504→      ''
   505→    );
   506→    offsetAdj = txt.length - stripped.length;
   507→    txt = stripped;
   508→  }
   509→  if (!_presTajweedEnabled) return escapeHtml(txt);
   510→  const annotations = getAyahAnnotations(surahNum, ayahNum);
   511→  if (annotations.length === 0) return escapeHtml(txt);
   512→  const adjusted =
   513→    offsetAdj > 0
   514→      ? annotations.map((ann: TajweedAnnotation) => ({
   515→          rule: ann.rule,
   516→          start: ann.start - offsetAdj,
   517→          end: ann.end - offsetAdj,
   518→        }))
   519→      : annotations;
   520→  const colorMap = buildColorMap(adjusted);
   521→  const words = txt.split(/\s+/).filter((w) => w.length > 0);
   522→  let outputPos = 0;
   523→  return words
   524→    .map((word: string, idx: number) => {
   525→      const wordHtml = tajweedColorWord(word, outputPos, colorMap);
   526→      outputPos += word.length + (idx < words.length - 1 ? 1 : 0); // +1 for space between words
   527→      return wordHtml;
   528→    })
   529→    .join(' ');
   530→}

   608→/** Star scene — twinkling stars with occasional shooting stars. */
   609→function renderStarsScene(canvas: HTMLCanvasElement): () => void {
   610→  const ctx = canvas.getContext('2d')!;
   611→  let stars: StarData[] = [];
   612→  let shootingStars: ShootingStar[] = [];
   613→  let time = 0;
   614→
   615→  function init(): void {
   616→    const w = canvas.width;
   617→    const h = canvas.height;
   618→    stars = [];
   619→    for (let i = 0; i < 250; i++) {
   620→      stars.push({
   621→        x: Math.random() * w,
   622→        y: Math.random() * h,
   623→        size: Math.random() * 2.5 + 0.5,
   624→        speed: Math.random() * 0.02 + 0.005,
   625→        brightness: Math.random(),
   626→        phase: Math.random() * Math.PI * 2,
   627→      });
   628→    }
   629→  }
   630→  init();
   631→
   632→  function frame(): void {
   633→    const w = canvas.width;
   634→    const h = canvas.height;
   635→    time += 0.016;
   636→
   637→    // Dark sky gradient
   638→    const grad = ctx.createLinearGradient(0, 0, 0, h);
   639→    grad.addColorStop(0, '#05050f');
   640→    grad.addColorStop(0.5, '#0a0a2e');
   641→    grad.addColorStop(1, '#101030');
   642→    ctx.fillStyle = grad;
   643→    ctx.fillRect(0, 0, w, h);
   644→
   645→    // Milky way band
   646→    const mwGrad = ctx.createLinearGradient(0, h * 0.2, 0, h * 0.6);
   647→    mwGrad.addColorStop(0, 'rgba(60, 50, 80, 0)');
   648→    mwGrad.addColorStop(0.3, 'rgba(80, 60, 120, 0.06)');
   649→    mwGrad.addColorStop(0.5, 'rgba(100, 80, 140, 0.08)');
   650→    mwGrad.addColorStop(0.7, 'rgba(80, 60, 120, 0.06)');
   651→    mwGrad.addColorStop(1, 'rgba(60, 50, 80, 0)');
   652→    ctx.fillStyle = mwGrad;
   653→    ctx.fillRect(0, 0, w, h);
   654→
   655→    // Draw twinkling stars
   656→    for (const star of stars) {
   657→      const twinkle = 0.5 + 0.5 * Math.sin(time * star.speed * 60 + star.phase);
   658→      const alpha = star.brightness * twinkle;
   659→      ctx.beginPath();
   660→      ctx.arc(star.x, star.y, star.size * (0.7 + 0.3 * twinkle), 0, Math.PI * 2);
   661→      ctx.fillStyle = `rgba(255, 255, 240, ${alpha})`;
   662→      ctx.fill();
   663→      // Glow for bigger stars
   664→      if (star.size > 1.8) {
   665→        ctx.beginPath();
   666→        ctx.arc(star.x, star.y, star.size * 3, 0, Math.PI * 2);
   667→        ctx.fillStyle = `rgba(200, 200, 255, ${alpha * 0.1})`;
   668→        ctx.fill();
   669→      }
   670→    }
   671→
   672→    // Shooting stars
   673→    if (Math.random() < 0.003) {
   674→      shootingStars.push({
   675→        x: Math.random() * w * 0.8,
   676→        y: Math.random() * h * 0.3,
   677→        vx: 4 + Math.random() * 4,
   678→        vy: 2 + Math.random() * 2,
   679→        life: 0,
   680→        maxLife: 30 + Math.random() * 30,
   681→        length: 40 + Math.random() * 60,
   682→      });
   683→    }
   684→    for (let i = shootingStars.length - 1; i >= 0; i--) {
   685→      const ss = shootingStars[i];
   686→      ss.x += ss.vx;
   687→      ss.y += ss.vy;
   688→      ss.life++;
   689→      const alpha = 1 - ss.life / ss.maxLife;
   690→      if (alpha <= 0) { shootingStars.splice(i, 1); continue; }
   691→      ctx.beginPath();
   692→      ctx.moveTo(ss.x, ss.y);
   693→      ctx.lineTo(ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6);
   694→      const ssGrad = ctx.createLinearGradient(
   695→        ss.x, ss.y,
   696→        ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6
   697→      );
   698→      ssGrad.addColorStop(0, `rgba(255, 255, 255, ${alpha})`);
   699→      ssGrad.addColorStop(1, `rgba(255, 255, 255, 0)`);
   700→      ctx.strokeStyle = ssGrad;
   701→      ctx.lineWidth = 2;
   702→      ctx.stroke();
   703→    }
   704→
   705→    _sceneAnimationId = requestAnimationFrame(frame);
   706→  }
   707→  frame();
   708→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   709→}

   893→/** Golden particles scene — floating dust motes. */
   894→function renderParticlesScene(canvas: HTMLCanvasElement): () => void {
   895→  const ctx = canvas.getContext('2d')!;
   896→  let time = 0;
   897→  let particles: Particle[] = [];
   898→
   899→  function init(): void {
   900→    const w = canvas.width;
   901→    const h = canvas.height;
   902→    particles = [];
   903→    for (let i = 0; i < 120; i++) {
   904→      particles.push({
   905→        x: Math.random() * w,
   906→        y: Math.random() * h,
   907→        vx: (Math.random() - 0.5) * 0.3,
   908→        vy: -Math.random() * 0.5 - 0.1,
   909→        size: Math.random() * 3 + 1,
   910→        opacity: Math.random() * 0.5 + 0.2,
   911→        phase: Math.random() * Math.PI * 2,
   912→      });
   913→    }
   914→  }
   915→  init();
   916→
   917→  function frame(): void {
   918→    const w = canvas.width;
   919→    const h = canvas.height;
   920→    time += 0.016;
   921→
   922→    // Dark warm background
   923→    const bgGrad = ctx.createRadialGradient(w * 0.5, h * 0.4, 0, w * 0.5, h * 0.4, Math.max(w, h) * 0.7);
   924→    bgGrad.addColorStop(0, '#1a1510');
   925→    bgGrad.addColorStop(0.5, '#0f0d08');
   926→    bgGrad.addColorStop(1, '#080604');
   927→    ctx.fillStyle = bgGrad;
   928→    ctx.fillRect(0, 0, w, h);
   929→
   930→    // Warm light source glow
   931→    const lightGlow = ctx.createRadialGradient(w * 0.5, h * 0.3, 0, w * 0.5, h * 0.3, w * 0.4);
   932→    lightGlow.addColorStop(0, 'rgba(255, 200, 100, 0.08)');
   933→    lightGlow.addColorStop(0.5, 'rgba(255, 180, 80, 0.03)');
   934→    lightGlow.addColorStop(1, 'rgba(255, 150, 50, 0)');
   935→    ctx.fillStyle = lightGlow;
   936→    ctx.fillRect(0, 0, w, h);
   937→
   938→    // Draw and update particles
   939→    for (const p of particles) {
   940→      p.x += p.vx + Math.sin(time + p.phase) * 0.2;
   941→      p.y += p.vy;
   942→
   943→      // Wrap around
   944→      if (p.y < -10) { p.y = h + 10; p.x = Math.random() * w; }
   945→      if (p.x < -10) p.x = w + 10;
   946→      if (p.x > w + 10) p.x = -10;
   947→
   948→      const pulse = 0.6 + 0.4 * Math.sin(time * 1.5 + p.phase);
   949→      const alpha = p.opacity * pulse;
   950→
   951→      // Glow
   952→      const glowGrad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size * 4);
   953→      glowGrad.addColorStop(0, `rgba(255, 210, 100, ${alpha * 0.5})`);
   954→      glowGrad.addColorStop(1, 'rgba(255, 180, 50, 0)');
   955→      ctx.fillStyle = glowGrad;
   956→      ctx.fillRect(p.x - p.size * 4, p.y - p.size * 4, p.size * 8, p.size * 8);
   957→
   958→      // Core
   959→      ctx.beginPath();
   960→      ctx.arc(p.x, p.y, p.size * pulse, 0, Math.PI * 2);
   961→      ctx.fillStyle = `rgba(255, 220, 120, ${alpha})`;
   962→      ctx.fill();
   963→    }
   964→
   965→    _sceneAnimationId = requestAnimationFrame(frame);
   966→  }
   967→  frame();
   968→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   969→}

   971→/** Rain scene — falling rain with lightning. */
   972→function renderRainScene(canvas: HTMLCanvasElement): () => void {
   973→  const ctx = canvas.getContext('2d')!;
   974→  let time = 0;
   975→  let drops: RainDrop[] = [];
   976→  let lightningAlpha = 0;
   977→  let nextLightning = 200 + Math.random() * 400;
   978→
   979→  function init(): void {
   980→    const w = canvas.width;
   981→    const h = canvas.height;
   982→    drops = [];
   983→    for (let i = 0; i < 200; i++) {
   984→      drops.push({
   985→        x: Math.random() * w,
   986→        y: Math.random() * h,
   987→        speed: 8 + Math.random() * 8,
   988→        length: 15 + Math.random() * 20,
   989→        opacity: 0.1 + Math.random() * 0.3,
   990→      });
   991→    }
   992→  }
   993→  init();
   994→
   995→  function frame(): void {
   996→    const w = canvas.width;
   997→    const h = canvas.height;
   998→    time += 0.016;
   999→
  1000→    // Stormy sky
  1001→    const skyGrad = ctx.createLinearGradient(0, 0, 0, h);
  1002→    skyGrad.addColorStop(0, '#0a0c12');
  1003→    skyGrad.addColorStop(0.3, '#111520');
  1004→    skyGrad.addColorStop(0.6, '#1a1e2e');
  1005→    skyGrad.addColorStop(1, '#0d1018');
  1006→    ctx.fillStyle = skyGrad;
  1007→    ctx.fillRect(0, 0, w, h);
  1008→
  1009→    // Lightning flash
  1010→    if (time * 60 > nextLightning) {
  1011→      lightningAlpha = 0.6;
  1012→      nextLightning = time * 60 + 200 + Math.random() * 500;
  1013→    }
  1014→    if (lightningAlpha > 0) {
  1015→      ctx.fillStyle = `rgba(180, 190, 255, ${lightningAlpha})`;
  1016→      ctx.fillRect(0, 0, w, h);
  1017→      lightningAlpha *= 0.85;
  1018→      if (lightningAlpha < 0.01) lightningAlpha = 0;
  1019→    }
  1020→
  1021→    // Cloud layer
  1022→    for (let i = 0; i < 5; i++) {
  1023→      const cx = (i / 5) * w + Math.sin(time * 0.1 + i * 3) * 40;
  1024→      const cy = h * 0.05 + Math.sin(time * 0.05 + i) * 20;
  1025→      const cw = 200 + Math.sin(i * 7.3) * 80;
  1026→      const ch = 60 + Math.sin(i * 4.1) * 20;
  1027→      const cloudGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, cw * 0.5);
  1028→      cloudGrad.addColorStop(0, 'rgba(30, 35, 50, 0.6)');
  1029→      cloudGrad.addColorStop(1, 'rgba(20, 25, 35, 0)');
  1030→      ctx.fillStyle = cloudGrad;
  1031→      ctx.fillRect(cx - cw, cy - ch, cw * 2, ch * 2);
  1032→    }
  1033→
  1034→    // Rain drops
  1035→    ctx.lineCap = 'round';
  1036→    for (const d of drops) {
  1037→      d.y += d.speed;
  1038→      d.x -= 1; // Slight wind
  1039→
  1040→      if (d.y > h) {
  1041→        d.y = -d.length;
  1042→        d.x = Math.random() * w;
  1043→      }
  1044→      if (d.x < -20) d.x = w + 20;
  1045→
  1046→      ctx.beginPath();
  1047→      ctx.moveTo(d.x, d.y);
  1048→      ctx.lineTo(d.x - 1.5, d.y + d.length);
  1049→      ctx.strokeStyle = `rgba(150, 170, 220, ${d.opacity})`;
  1050→      ctx.lineWidth = 1;
  1051→      ctx.stroke();
  1052→    }
  1053→
  1054→    // Splash effects at bottom
  1055→    for (let i = 0; i < 15; i++) {
  1056→      const sx = (Math.sin(time * 3 + i * 17.3) * 0.5 + 0.5) * w;
  1057→      const sy = h - 10 - Math.random() * 5;
  1058→      const splashAlpha = 0.2 + 0.2 * Math.abs(Math.sin(time * 5 + i * 3.7));
  1059→      ctx.beginPath();
  1060→      ctx.arc(sx, sy, 2, 0, Math.PI * 2);
  1061→      ctx.fillStyle = `rgba(150, 170, 220, ${splashAlpha})`;
  1062→      ctx.fill();
  1063→    }
  1064→
  1065→    _sceneAnimationId = requestAnimationFrame(frame);
  1066→  }
  1067→  frame();
  1068→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
  1069→}

  1421→function handleKeyDown(e: KeyboardEvent): void {
  1422→  if (
  1423→    (e.target as HTMLElement).tagName === 'INPUT' ||
  1424→    (e.target as HTMLElement).tagName === 'TEXTAREA' ||
  1425→    (e.target as HTMLElement).tagName === 'SELECT'
  1426→  )
  1427→    return;
  1428→  // Any key press shows controls
  1429→  showControls();
  1430→  switch (e.key) {
  1431→    case 'Escape':
  1432→      // If in fullscreen, only exit fullscreen — don't close presentation
  1433→      const fsEl = document.fullscreenElement || (document as unknown as { webkitFullscreenElement?: Element }).webkitFullscreenElement;
  1434→      if (fsEl) {
  1435→        e.preventDefault();
  1436→        if (document.exitFullscreen) {
  1437→          document.exitFullscreen().catch(() => {});
  1438→        } else if ((document as unknown as { webkitExitFullscreen?: () => Promise<void> }).webkitExitFullscreen) {
  1439→          (document as unknown as { webkitExitFullscreen: () => Promise<void> }).webkitExitFullscreen().catch(() => {});
  1440→        }
  1441→      } else {
  1442→        closePresentation();
  1443→      }
  1444→      break;
  1445→    case 'ArrowRight':
  1446→    case 'ArrowDown':
  1447→      e.preventDefault();
  1448→      navigateAyah(1);

  1450→    case 'ArrowLeft':
  1451→    case 'ArrowUp':

  1454→      break;
  1455→    case ' ':
  1456→      e.preventDefault();
  1457→      togglePlayPause();
  1458→      updatePlayPauseBtn();
  1459→      updatePresPlayPauseBtn();
  1460→      break;
  1461→  }

  1492→export function initPresentation(): void {
  1493→  console.log('[Presentation] initPresentation() called');
  1494→  injectStyles();
  1495→
  1496→  // Verify the presentation overlay exists in the DOM
  1497→  const overlay = document.getElementById('presentationOverlay');
  1498→  if (!overlay) {
  1499→    console.error('[Presentation] ERROR: presentationOverlay element not found in DOM!');
  1500→    return;
  1501→  }
  1502→  console.log('[Presentation] Overlay element found, binding event handlers...');
  1503→
  1504→  if (dom.presentationCloseBtn) dom.presentationCloseBtn.addEventListener('click', closePresentation);
  1505→  if (dom.presentationPrevBtn) dom.presentationPrevBtn.addEventListener('click', () => navigateAyah(-1));
  1506→  if (dom.presentationNextBtn) dom.presentationNextBtn.addEventListener('click', () => navigateAyah(1));
  1507→  dom.presentationOverlay?.addEventListener('click', (e: MouseEvent) => {
  1508→    // Don't close on background click when in fullscreen
  1509→    const inFs = !!(document.fullscreenElement || (document as unknown as { webkitFullscreenElement?: Element }).webkitFullscreenElement);
  1510→    if (e.target === dom.presentationOverlay && !inFs) closePresentation();
  1511→  });
  1512→
  1513→  // Play/Pause button
  1514→  if (dom.presPlayPauseBtn) {
  1515→    dom.presPlayPauseBtn.addEventListener('click', () => {
  1516→      togglePlayPause();
  1517→      updatePlayPauseBtn();
  1518→      updatePresPlayPauseBtn();
  1519→      showControls(); // Reset auto-hide timer
  1520→    });
  1521→  }
  1522→
  1523→  // Tajweed toggle button (presentation-local)
  1524→  if (dom.presTajweedBtn) {
  1525→    dom.presTajweedBtn.addEventListener('click', () => {
  1526→      togglePresTajweed();
  1527→      showControls(); // Reset auto-hide timer
  1528→    });
  1529→  }
  1530→
  1531→  // Fullscreen toggle button
  1532→  if (dom.presFullscreenBtn) {
  1533→    dom.presFullscreenBtn.addEventListener('click', () => {
  1534→      togglePresFullscreen();
  1535→      showControls(); // Reset auto-hide timer
  1536→    });
  1537→  }
  1538→
  1539→  // Mouse/touch movement shows controls
  1540→  dom.presentationOverlay?.addEventListener('mousemove', handleOverlayMouseMove);
  1541→  dom.presentationOverlay?.addEventListener('touchstart', handleOverlayMouseMove, { passive: true });
  1542→
  1543→  // Fullscreen change event
  1544→  document.addEventListener('fullscreenchange', handleFullscreenChange);
  1545→  document.addEventListener('webkitfullscreenchange', handleFullscreenChange);
  1546→
  1547→  console.log('[Presentation] initPresentation() completed successfully');
  1548→}
