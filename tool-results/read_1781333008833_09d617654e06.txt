    41→    41→/** Get a background image based on the current time of day. */
    42→    42→function getAutoBackground(): (typeof NATURE_BACKGROUNDS)[number] {
    43→    43→  const hour = new Date().getHours();
    44→    44→  if (hour >= 4 && hour < 7) return NATURE_BACKGROUNDS[0]; // dawn
    45→    45→  if (hour >= 7 && hour < 12) return NATURE_BACKGROUNDS[1]; // morning clouds
    46→    46→  if (hour >= 12 && hour < 16) return NATURE_BACKGROUNDS[2]; // afternoon mountains
    47→    47→  if (hour >= 16 && hour < 19) return NATURE_BACKGROUNDS[3]; // sunset
    48→    48→  return NATURE_BACKGROUNDS[4]; // night sky
    49→    49→}

   495→   495→function buildAyahHtml(text: string, surahNum: number, ayahNum: number): string {
   496→   496→  let txt = text;
   497→   497→  let offsetAdj = 0;
   498→   498→  if (surahNum !== 1 && ayahNum === 1) {
   499→   499→    const stripped = txt.replace(
   500→   500→      /^ب[\u064B-\u065F\u0670]*س[\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*[هة][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*[نث][\u064B-\u065F\u0670]*\s*[إأآٱ][\u064B-\u065F\u0670]*ل[\u064B-\u065F\u0670]*ر[\u064B-\u065F\u0670]*[حخ][\u064B-\u065F\u0670]*[يى][\u064B-\u065F\u0670]*م[\u064B-\u065F\u0670]*\s*/u,
   501→   501→      ''
   502→   502→    );
   503→   503→    offsetAdj = txt.length - stripped.length;
   504→   504→    txt = stripped;
   505→   505→  }
   506→   506→  if (!_presTajweedEnabled) return escapeHtml(txt);
   507→   507→  const annotations = getAyahAnnotations(surahNum, ayahNum);
   508→   508→  if (annotations.length === 0) return escapeHtml(txt);
   509→   509→  const adjusted =
   510→   510→    offsetAdj > 0
   511→   511→      ? annotations.map((ann: TajweedAnnotation) => ({
   512→   512→          rule: ann.rule,
   513→   513→          start: ann.start - offsetAdj,
   514→   514→          end: ann.end - offsetAdj,
   515→   515→        }))
   516→   516→      : annotations;
   517→   517→  const colorMap = buildColorMap(adjusted);
   518→   518→  const words = txt.split(/\s+/).filter((w) => w.length > 0);
   519→   519→  let outputPos = 0;
   520→   520→  return words
   521→   521→    .map((word: string, idx: number) => {
   522→   522→      const wordHtml = tajweedColorWord(word, outputPos, colorMap);
   523→   523→      outputPos += word.length + (idx < words.length - 1 ? 1 : 0); // +1 for space between words
   524→   524→      return wordHtml;
   525→   525→    })
   526→   526→    .join(' ');
   527→   527→}

   600→   600→/** Star scene — twinkling stars with occasional shooting stars. */
   601→   601→function renderStarsScene(canvas: HTMLCanvasElement): () => void {
   602→   602→  const ctx = canvas.getContext('2d')!;
   603→   603→  let stars: StarData[] = [];
   604→   604→  let shootingStars: ShootingStar[] = [];
   605→   605→  let time = 0;
   606→   606→
   607→   607→  function init(): void {
   608→   608→    const w = canvas.width;
   609→   609→    const h = canvas.height;
   610→   610→    stars = [];
   611→   611→    for (let i = 0; i < 250; i++) {
   612→   612→      stars.push({
   613→   613→        x: Math.random() * w,
   614→   614→        y: Math.random() * h,
   615→   615→        size: Math.random() * 2.5 + 0.5,
   616→   616→        speed: Math.random() * 0.02 + 0.005,
   617→   617→        brightness: Math.random(),
   618→   618→        phase: Math.random() * Math.PI * 2,
   619→   619→      });
   620→   620→    }
   621→   621→  }
   622→   622→  init();
   623→   623→
   624→   624→  function frame(): void {
   625→   625→    const w = canvas.width;
   626→   626→    const h = canvas.height;
   627→   627→    time += 0.016;
   628→   628→
   629→   629→    // Dark sky gradient
   630→   630→    const grad = ctx.createLinearGradient(0, 0, 0, h);
   631→   631→    grad.addColorStop(0, '#05050f');
   632→   632→    grad.addColorStop(0.5, '#0a0a2e');
   633→   633→    grad.addColorStop(1, '#101030');
   634→   634→    ctx.fillStyle = grad;
   635→   635→    ctx.fillRect(0, 0, w, h);
   636→   636→
   637→   637→    // Milky way band
   638→   638→    const mwGrad = ctx.createLinearGradient(0, h * 0.2, 0, h * 0.6);
   639→   639→    mwGrad.addColorStop(0, 'rgba(60, 50, 80, 0)');
   640→   640→    mwGrad.addColorStop(0.3, 'rgba(80, 60, 120, 0.06)');
   641→   641→    mwGrad.addColorStop(0.5, 'rgba(100, 80, 140, 0.08)');
   642→   642→    mwGrad.addColorStop(0.7, 'rgba(80, 60, 120, 0.06)');
   643→   643→    mwGrad.addColorStop(1, 'rgba(60, 50, 80, 0)');
   644→   644→    ctx.fillStyle = mwGrad;
   645→   645→    ctx.fillRect(0, 0, w, h);
   646→   646→
   647→   647→    // Draw twinkling stars
   648→   648→    for (const star of stars) {
   649→   649→      const twinkle = 0.5 + 0.5 * Math.sin(time * star.speed * 60 + star.phase);
   650→   650→      const alpha = star.brightness * twinkle;
   651→   651→      ctx.beginPath();
   652→   652→      ctx.arc(star.x, star.y, star.size * (0.7 + 0.3 * twinkle), 0, Math.PI * 2);
   653→   653→      ctx.fillStyle = `rgba(255, 255, 240, ${alpha})`;
   654→   654→      ctx.fill();
   655→   655→      // Glow for bigger stars
   656→   656→      if (star.size > 1.8) {
   657→   657→        ctx.beginPath();
   658→   658→        ctx.arc(star.x, star.y, star.size * 3, 0, Math.PI * 2);
   659→   659→        ctx.fillStyle = `rgba(200, 200, 255, ${alpha * 0.1})`;
   660→   660→        ctx.fill();
   661→   661→      }
   662→   662→    }
   663→   663→
   664→   664→    // Shooting stars
   665→   665→    if (Math.random() < 0.003) {
   666→   666→      shootingStars.push({
   667→   667→        x: Math.random() * w * 0.8,
   668→   668→        y: Math.random() * h * 0.3,
   669→   669→        vx: 4 + Math.random() * 4,
   670→   670→        vy: 2 + Math.random() * 2,
   671→   671→        life: 0,
   672→   672→        maxLife: 30 + Math.random() * 30,
   673→   673→        length: 40 + Math.random() * 60,
   674→   674→      });
   675→   675→    }
   676→   676→    for (let i = shootingStars.length - 1; i >= 0; i--) {
   677→   677→      const ss = shootingStars[i];
   678→   678→      ss.x += ss.vx;
   679→   679→      ss.y += ss.vy;
   680→   680→      ss.life++;
   681→   681→      const alpha = 1 - ss.life / ss.maxLife;
   682→   682→      if (alpha <= 0) { shootingStars.splice(i, 1); continue; }
   683→   683→      ctx.beginPath();
   684→   684→      ctx.moveTo(ss.x, ss.y);
   685→   685→      ctx.lineTo(ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6);
   686→   686→      const ssGrad = ctx.createLinearGradient(
   687→   687→        ss.x, ss.y,
   688→   688→        ss.x - ss.vx * ss.length / 6, ss.y - ss.vy * ss.length / 6
   689→   689→      );
   690→   690→      ssGrad.addColorStop(0, `rgba(255, 255, 255, ${alpha})`);
   691→   691→      ssGrad.addColorStop(1, `rgba(255, 255, 255, 0)`);
   692→   692→      ctx.strokeStyle = ssGrad;
   693→   693→      ctx.lineWidth = 2;
   694→   694→      ctx.stroke();
   695→   695→    }
   696→   696→
   697→   697→    _sceneAnimationId = requestAnimationFrame(frame);
   698→   698→  }
   699→   699→  frame();
   700→   700→  return () => { if (_sceneAnimationId !== null) { cancelAnimationFrame(_sceneAnimationId); _sceneAnimationId = null; } };
   701→   701→}

   886→   886→function renderParticlesScene(canvas: HTMLCanvasElement): () => void {
   887→   887→  const ctx = canvas.getContext('2d')!;
   888→   888→  let time = 0;
   889→   889→  let particles: Particle[] = [];
   890→   890→
   891→   891→  function init(): void {
   892→   892→    const w = canvas.width;
   893→   893→    const h = canvas.height;
   894→   894→    particles = [];
   895→   895→    for (let i = 0; i < 120; i++) {

   898→   898→        y: Math.random() * h,
   899→   899→        vx: (Math.random() - 0.5) * 0.3,
   900→   900→        vy: -Math.random() * 0.5 - 0.1,
   901→   901→        size: Math.random() * 3 + 1,
   902→   902→        opacity: Math.random() * 0.5 + 0.2,
   903→   903→        phase: Math.random() * Math.PI * 2,
   904→   904→      });
   905→   905→    }
   906→   906→  }
   907→   907→  init();
   908→   908→
   909→   909→  function frame(): void {
   910→   910→    const w = canvas.width;
   911→   911→    const h = canvas.height;
   912→   912→    time += 0.016;
   913→   913→
   914→   914→    // Dark warm background
   915→   915→    const bgGrad = ctx.createRadialGradient(w * 0.5, h * 0.4, 0, w * 0.5, h * 0.4, Math.max(w, h) * 0.7);
   916→   916→    bgGrad.addColorStop(0, '#1a1510');
   917→   917→    bgGrad.addColorStop(0.5, '#0f0d08');

   921→   921→

   923→   923→    const lightGlow = ctx.createRadialGradient(w * 0.5, h * 0.3, 0, w * 0.5, h * 0.3, w * 0.4);
   924→   924→    lightGlow.addColorStop(0, 'rgba(255, 200, 100, 0.08)');
   925→   925→    lightGlow.addColorStop(0.5, 'rgba(255, 180, 80, 0.03)');

   929→   929→
   930→   930→    // Draw and update particles
   931→   931→    for (const p of particles) {
   932→   932→      p.x += p.vx + Math.sin(time + p.phase) * 0.2;
   933→   933→      p.y += p.vy;
   934→   934→
   935→   935→      // Wrap around
   936→   936→      if (p.y < -10) { p.y = h + 10; p.x = Math.random() * w; }
   937→   937→      if (p.x < -10) p.x = w + 10;
   938→   938→      if (p.x > w + 10) p.x = -10;
   939→   939→

   941→   941→      const alpha = p.opacity * pulse;
   942→   942→
   943→   943→      // Glow
   944→   944→      const glowGrad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size * 4);
   945→   945→      glowGrad.addColorStop(0, `rgba(255, 210, 100, ${alpha * 0.5})`);
   946→   946→      glowGrad.addColorStop(1, 'rgba(255, 180, 50, 0)');
   947→   947→      ctx.fillStyle = glowGrad;
   948→   948→      ctx.fillRect(p.x - p.size * 4, p.y - p.size * 4, p.size * 8, p.size * 8);
   949→   949→
   950→   950→      // Core
   951→   951→      ctx.beginPath();
   952→   952→      ctx.arc(p.x, p.y, p.size * pulse, 0, Math.PI * 2);
   953→   953→      ctx.fillStyle = `rgba(255, 220, 120, ${alpha})`;
   954→   954→      ctx.fill();
   955→   955→    }
   956→   956→
   957→   957→    _sceneAnimationId = requestAnimationFrame(frame);
   958→   958→  }
   959→   959→  frame();

   961→   961→}

  1388→  1388→function handleKeyDown(e: KeyboardEvent): void {
  1389→  1389→  if (
  1390→  1390→    (e.target as HTMLElement).tagName === 'INPUT' ||
  1391→  1391→    (e.target as HTMLElement).tagName === 'TEXTAREA' ||
  1392→  1392→    (e.target as HTMLElement).tagName === 'SELECT'
  1393→  1393→  )
  1394→  1394→    return;
  1395→  1395→  // Any key press shows controls
  1396→  1396→  showControls();
  1397→  1397→  switch (e.key) {
  1398→  1398→    case 'Escape':
  1399→  1399→      // If in fullscreen, only exit fullscreen — don't close presentation
  1400→  1400→      const fsEl = document.fullscreenElement || (document as unknown as { webkitFullscreenElement?: Element }).webkitFullscreenElement;
  1401→  1401→      if (fsEl) {
  1402→  1402→        e.preventDefault();
  1403→  1403→        if (document.exitFullscreen) {
  1404→  1404→          document.exitFullscreen().catch(() => {});
  1405→  1405→        } else if ((document as unknown as { webkitExitFullscreen?: () => Promise<void> }).webkitExitFullscreen) {
  1406→  1406→          (document as unknown as { webkitExitFullscreen: () => Promise<void> }).webkitExitFullscreen().catch(() => {});
  1407→  1407→        }
  1408→  1408→      } else {
  1409→  1409→        closePresentation();
  1410→  1410→      }
  1411→  1411→      break;
  1412→  1412→    case 'ArrowRight':
  1413→  1413→    case 'ArrowDown':
  1414→  1414→      e.preventDefault();
  1415→  1415→      navigateAyah(1);
  1416→  1416→      break;
  1417→  1417→    case 'ArrowLeft':
  1418→  1418→    case 'ArrowUp':
  1419→  1419→      e.preventDefault();
  1420→  1420→      navigateAyah(-1);
  1421→  1421→      break;
  1422→  1422→    case ' ':
  1423→  1423→      e.preventDefault();
  1424→  1424→      togglePlayPause();
  1425→  1425→      updatePlayPauseBtn();
  1426→  1426→      updatePresPlayPauseBtn();
  1427→  1427→      break;
  1428→  1428→  }
  1429→  1429→}
