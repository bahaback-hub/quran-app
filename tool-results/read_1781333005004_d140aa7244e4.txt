   110→export async function loadPageData(pageNum: number): Promise<PageLayoutData | null> {
   111→  const key = `qcf4-${pageNum}`;
   112→  if (layoutCache.has(key)) return layoutCache.get(key)!;
   113→  const padded = String(pageNum).padStart(3, '0');
   114→  try {
   115→    const res = await fetch(`${PAGE_BASE}${padded}.json`);
   116→    if (!res.ok) return null;
   117→    const data = (await res.json()) as PageLayoutData;
   118→    // Evict oldest entry if cache is full (LRU policy)
   119→    if (layoutCache.size >= LAYOUT_CACHE_MAX) {
   120→      const firstKey = layoutCache.keys().next().value;
   121→      if (firstKey !== undefined) layoutCache.delete(firstKey);
   122→    }
   123→    layoutCache.set(key, data);
   124→    return data;
   125→  } catch (e: unknown) {
   126→    console.warn('[Mushaf] Failed to load page data:', e);
   127→    return null;
   128→  }

   131→/**
   132→ * Build the woff2 font URL for a given QCF4 font name.
   133→ * BSML font uses a different naming convention than Hafs fonts.
   134→ */
   135→function getFontUrl(fontName: string): string {
   136→  if (fontName === BSML_FONT) {
   137→    return `${FONT_BASE}${BSML_FONT}.woff2`;
   138→  }
   139→  return `${FONT_BASE}${fontName}_W.woff2`;

   142→/**
   143→ * Fetch font binary data using XMLHttpRequest instead of fetch().
   144→ *
   145→ * CRITICAL: In Capacitor with CapacitorHttp enabled, the native HTTP plugin
   146→ * intercepts ALL fetch() calls and can corrupt binary data (woff2 fonts).
   147→ * XMLHttpRequest is NOT intercepted by CapacitorHttp, so it returns
   148→ * correct binary data. This is the key fix for mushaf mode in Android APK.
   149→ */
   150→function fetchFontViaXHR(url: string): Promise<ArrayBuffer> {
   151→  return new Promise((resolve, reject) => {
   152→    const xhr = new XMLHttpRequest();
   153→    xhr.open('GET', url, true);
   154→    xhr.responseType = 'arraybuffer';
   155→    xhr.onload = () => {
   156→      if (xhr.status === 200) {
   157→        resolve(xhr.response as ArrayBuffer);
   158→      } else {
   159→        reject(new Error(`XHR HTTP ${xhr.status}`));
   160→      }
   161→    };
   162→    xhr.onerror = () => reject(new Error('XHR network error'));
   163→    xhr.ontimeout = () => reject(new Error('XHR timeout'));
   164→    xhr.timeout = 15000; // 15 second timeout
   165→    xhr.send();
   166→  });
   167→}

   169→/**
   170→ * Fetch font binary data. In Capacitor, uses XHR to avoid CapacitorHttp
   171→ * interception. In browsers, uses standard fetch().
   172→ */
   173→async function fetchFontBinary(url: string): Promise<ArrayBuffer> {
   174→  if (_isCapacitor) {
   175→    // Use XHR in Capacitor to avoid CapacitorHttp interception
   176→    console.log(`[Mushaf] Using XHR for font binary (bypassing CapacitorHttp): ${url}`);
   177→    return fetchFontViaXHR(url);
   178→  }
   179→  // Use standard fetch in browser
   180→  const res = await fetch(url);
   181→  if (!res.ok) throw new Error(`HTTP ${res.status}`);
   182→  return res.arrayBuffer();
   183→}

   185→/**
   186→ * Verify a font is truly usable on Canvas by drawing a PUA character
   187→ * and checking if the pixels are non-blank.
   188→ *
   189→ * IMPORTANT: In Android WebView (Capacitor), Canvas pixel verification is unreliable
   190→ * because WebView processes fonts differently. We use document.fonts.check()
   191→ * as a fallback verification method in that case.
   192→ */
   193→function verifyFontOnCanvas(fontName: string): boolean {
   194→  try {
   195→    // In Capacitor/Android WebView, skip pixel-level Canvas verification
   196→    // because it's unreliable — trust document.fonts API instead
   197→    if (_isCapacitor) {
   198→      const fontsCheck = document.fonts.check(`40px "${fontName}"`);
   199→      console.log(`[Mushaf] Capacitor: document.fonts.check for "${fontName}": ${fontsCheck}`);
   200→      return fontsCheck;
   201→    }
   202→
   203→    const testSize = 40;
   204→    const canvas = document.createElement('canvas');
   205→    canvas.width = testSize * 2;
   206→    canvas.height = testSize * 2;
   207→    const ctx = canvas.getContext('2d');
   208→    if (!ctx) return false;
   209→
   210→    ctx.fillStyle = '#FFFFFF';
   211→    ctx.fillRect(0, 0, canvas.width, canvas.height);
   212→
   213→    ctx.font = `${testSize}px "${fontName}"`;
   214→    ctx.fillStyle = '#000000';
   215→    ctx.textAlign = 'center';
   216→    ctx.textBaseline = 'middle';
   217→    ctx.fillText('\uF100', canvas.width / 2, canvas.height / 2);
   218→
   219→    const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
   220→    const data = imageData.data;
   221→    for (let i = 0; i < data.length; i += 4) {
   222→      if (data[i] < 250 || data[i + 1] < 250 || data[i + 2] < 250) {
   223→        return true;
   224→      }
   225→    }
   226→
   227→    return false;
   228→  } catch {
   229→    return false;
   230→  }
   231→}

   233→/**
   234→ * Create a hidden DOM element that uses the font. This forces the browser
   235→ * to download and parse the font binary, making it available for Canvas.
   236→ * Some browsers (especially Android WebView) won't make fonts available
   237→ * to Canvas unless they're used in the DOM first.
   238→ */
   239→function createFontPreloadElement(fontName: string): void {
   240→  const id = `qcf-preload-${fontName}`;
   241→  if (document.getElementById(id)) return;
   242→
   243→  const span = document.createElement('span');
   244→  span.id = id;
   245→  span.setAttribute('aria-hidden', 'true');
   246→  span.style.cssText = `
   247→    position: absolute;
   248→    left: -9999px;
   249→    top: -9999px;
   250→    visibility: hidden;
   251→    font-family: "${fontName}";
   252→    font-size: 40px;
   253→    pointer-events: none;
   254→  `;
   255→  // Use multiple PUA characters to ensure the browser loads the font
   256→  span.textContent = '\uF100\uF101\uF102\uF103\uF104\uF105';
   257→  document.body.appendChild(span);
   258→}

   260→/**
   261→ * Load a QCF4 font using the most reliable method available.
   262→ *
   263→ * Strategy order (most to least reliable):
   264→ * 1. XHR fetch as ArrayBuffer → FontFace from binary data
   265→ *    (Bypasses CapacitorHttp which corrupts binary data)
   266→ * 2. FontFace API with URL (browser handles download)
   267→ * 3. CSS @font-face injection with DOM preload element
   268→ *
   269→ * In Capacitor: We skip pixel-level Canvas verification and trust
   270→ * document.fonts API instead.
   271→ */
   272→async function ensureFontLoaded(fontName: string): Promise<void> {
   273→  if (loadedFonts.has(fontName)) return;
   274→
   275→  // Deduplicate concurrent loads
   276→  if (loadingFonts.has(fontName)) {
   277→    const result = await loadingFonts.get(fontName);
   278→    if (result) return;
   279→    loadingFonts.delete(fontName);
   280→  }
   281→
   282→  const loadPromise = _doLoadFont(fontName);
   283→  loadingFonts.set(fontName, loadPromise);
   284→
   285→  try {
   286→    const success = await loadPromise;
   287→    if (success) {
   288→      loadedFonts.add(fontName);
   289→    }
   290→  } finally {
   291→    loadingFonts.delete(fontName);
   292→  }
   293→}

   295→/**
   296→ * Internal font loading implementation.
   297→ * Returns true if the font was loaded and verified.
   298→ */
   299→async function _doLoadFont(fontName: string): Promise<boolean> {
   300→  const fontUrl = getFontUrl(fontName);
   301→  const waitMs = _isCapacitor ? 1500 : 200;
   302→
   303→  for (let attempt = 1; attempt <= (_isCapacitor ? 2 : 3); attempt++) {
   304→    console.log(`[Mushaf] Loading font "${fontName}" (attempt ${attempt})...`);
   305→
   306→    // --- Strategy 1: XHR/Fetch ArrayBuffer → FontFace (most reliable) ---
   307→    try {
   308→      console.log(`[Mushaf] Strategy 1: ArrayBuffer for "${fontName}" from ${fontUrl}`);
   309→      const buffer = await fetchFontBinary(fontUrl);
   310→      console.log(`[Mushaf] Font "${fontName}" downloaded: ${buffer.byteLength} bytes`);
   311→
   312→      // Validate: woff2 files should be at least 10KB
   313→      if (buffer.byteLength < 10000) {
   314→        console.warn(`[Mushaf] Font "${fontName}" too small (${buffer.byteLength} bytes), likely corrupted`);
   315→        throw new Error(`Font data too small: ${buffer.byteLength} bytes`);
   316→      }
   317→
   318→      const fontFace = new FontFace(fontName, buffer, { display: 'block' });
   319→      const loaded = await fontFace.load();
   320→      document.fonts.add(loaded);
   321→
   322→      await document.fonts.ready;
   323→      createFontPreloadElement(fontName);
   324→      await new Promise<void>((r) => setTimeout(r, waitMs));
   325→
   326→      const verified = verifyFontOnCanvas(fontName);
   327→      console.log(`[Mushaf] Strategy 1 verify for "${fontName}": ${verified}`);
   328→      if (verified) {
   329→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via ArrayBuffer`);
   330→        return true;
   331→      }
   332→
   333→      // In Capacitor: trust document.fonts and proceed even if Canvas verify fails
   334→      if (_isCapacitor) {
   335→        const fontsCheck = document.fonts.check(`40px "${fontName}"`);
   336→        if (fontsCheck) {
   337→          console.log(`[Mushaf] Capacitor: Trusting document.fonts.check for "${fontName}"`);
   338→          return true;
   339→        }
   340→        // Even if fonts.check fails, if we added the font, proceed anyway
   341→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" despite verification failure`);
   342→        return true;
   343→      }
   344→    } catch (e) {
   345→      console.warn(`[Mushaf] ArrayBuffer strategy failed for "${fontName}":`, e);
   346→    }
   347→
   348→    // --- Strategy 2: FontFace with URL (browser handles download) ---
   349→    try {
   350→      console.log(`[Mushaf] Strategy 2: FontFace URL for "${fontName}"`);
   351→      const fontFace = new FontFace(fontName, `url('${fontUrl}')`, { display: 'block' });
   352→      const loaded = await fontFace.load();
   353→      document.fonts.add(loaded);
   354→
   355→      await document.fonts.ready;
   356→      createFontPreloadElement(fontName);
   357→      await new Promise<void>((r) => setTimeout(r, waitMs));
   358→
   359→      const verified = verifyFontOnCanvas(fontName);
   360→      console.log(`[Mushaf] Strategy 2 verify for "${fontName}": ${verified}`);
   361→      if (verified) {
   362→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via FontFace URL`);
   363→        return true;
   364→      }
   365→
   366→      if (_isCapacitor) {
   367→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" (strategy 2) despite verification failure`);
   368→        return true;
   369→      }
   370→    } catch (e) {
   371→      console.warn(`[Mushaf] FontFace URL strategy failed for "${fontName}":`, e);
   372→    }
   373→
   374→    // --- Strategy 3: CSS @font-face + DOM preload ---
   375→    try {
   376→      console.log(`[Mushaf] Strategy 3: CSS @font-face for "${fontName}"`);
   377→      const styleId = fontName === BSML_FONT ? 'qcf-basml' : `qcf-${fontName}`;
   378→      if (!document.getElementById(styleId)) {
   379→        const style = document.createElement('style');
   380→        style.id = styleId;
   381→        style.textContent = `
   382→          @font-face {
   383→            font-family: '${fontName}';
   384→            src: url('${fontUrl}') format('woff2');
   385→            font-display: block;
   386→          }`;
   387→        document.head.appendChild(style);
   388→      }
   389→
   390→      createFontPreloadElement(fontName);
   391→
   392→      try {
   393→        await document.fonts.load(`40px "${fontName}"`, '\uF100');
   394→      } catch {
   395→        /* some browsers don't support the text param */
   396→      }
   397→
   398→      await document.fonts.ready;
   399→      const cssWaitMs = _isCapacitor ? 1500 : 500;
   400→      await new Promise<void>((r) => setTimeout(r, cssWaitMs));
   401→
   402→      const verified = verifyFontOnCanvas(fontName);
   403→      console.log(`[Mushaf] Strategy 3 verify for "${fontName}": ${verified}`);
   404→      if (verified) {
   405→        console.log(`[Mushaf] Font "${fontName}" loaded successfully via CSS @font-face`);
   406→        return true;
   407→      }
   408→
   409→      if (_isCapacitor) {
   410→        console.warn(`[Mushaf] Capacitor: Proceeding with font "${fontName}" (strategy 3) despite verification failure`);
   411→        return true;
   412→      }
   413→    } catch (e) {
   414→      console.warn(`[Mushaf] CSS @font-face strategy failed for "${fontName}":`, e);
   415→    }
   416→
   417→    if (attempt < (_isCapacitor ? 2 : 3)) {
   418→      const delay = _isCapacitor ? 1000 : 500 * attempt;
   419→      console.warn(`[Mushaf] All strategies failed for "${fontName}" (attempt ${attempt}), retrying in ${delay}ms...`);
   420→      await new Promise<void>((r) => setTimeout(r, delay));
   421→    }
   422→  }
   423→
   424→  // In Capacitor: proceed anyway — better to show something than blank
   425→  if (_isCapacitor) {
   426→    console.warn(`[Mushaf] Capacitor: Proceeding despite font "${fontName}" loading failure — will retry render later`);
   427→    return true;
   428→  }
   429→
   430→  console.error(`[Mushaf] FAILED to load font "${fontName}" after all attempts.`);
   431→  return false;
   432→}

   447→/**
   448→ * Render a mushaf page onto a canvas element.
   449→ *
   450→ * For Capacitor/Android WebView, includes special handling:
   451→ * - Uses XHR for font binary data (bypasses CapacitorHttp)
   452→ * - Skips strict Canvas pixel verification
   453→ * - Schedules a re-render after a delay for WebView font processing
   454→ * - Adds an overall timeout to prevent hanging
   455→ */
   456→export async function renderPage(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   457→  console.log(`[Mushaf] renderPage(${pageNum}) starting... Capacitor=${_isCapacitor}`);
   458→
   459→  // Add overall timeout for Capacitor to prevent hanging
   460→  if (_isCapacitor) {
   461→    const timeoutMs = 30000;
   462→    return Promise.race([
   463→      _renderPageInternal(pageNum, targetCanvas),
   464→      new Promise<RenderPageResult>((resolve) =>
   465→        setTimeout(() => {
   466→          console.error(`[Mushaf] renderPage(${pageNum}) timed out after ${timeoutMs}ms`);
   467→          _renderPageWithCurrentFonts(pageNum, targetCanvas).then(resolve);
   468→        }, timeoutMs)
   469→      )
   470→    ]);
   471→  }
   472→
   473→  return _renderPageInternal(pageNum, targetCanvas);
   474→}

   476→/**
   477→ * Attempt to render a page using whatever fonts are currently available,
   478→ * even if font loading failed. Better to show partial text than blank.
   479→ */
   480→async function _renderPageWithCurrentFonts(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   481→  try {
   482→    const data = await loadPageData(pageNum);
   483→    if (!data) return { canvas: null, layout: null };
   484→
   485→    const pageFont = data.font || getPageFont(pageNum, null);
   486→    const canvas = targetCanvas || document.createElement('canvas');
   487→    canvas.width = CANVAS_W;
   488→    canvas.height = CANVAS_H;
   489→    const ctx = canvas.getContext('2d');
   490→    if (!ctx) return { canvas: null, layout: data };
   491→
   492→    const colors = getColors();
   493→    ctx.fillStyle = colors.bg;
   494→    ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   495→    renderPageContent(ctx, data, pageFont, colors);
   496→    drawPageFrame(ctx, colors);
   497→    drawPageNumber(ctx, pageNum, colors);
   498→    return { canvas, layout: data };
   499→  } catch {
   500→    return { canvas: null, layout: null };
   501→  }
   502→}

   504→async function _renderPageInternal(pageNum: number, targetCanvas?: HTMLCanvasElement | null): Promise<RenderPageResult> {
   505→  const data = await loadPageData(pageNum);
   506→  if (!data) {
   507→    console.error(`[Mushaf] Failed to load page data for page ${pageNum}`);
   508→    return { canvas: null, layout: null };
   509→  }
   510→  console.log(`[Mushaf] Page ${pageNum} data loaded: ${data.lines?.length || 0} lines`);
   511→
   512→  const pageFont = data.font || getPageFont(pageNum, null);
   513→
   514→  // Collect ALL unique font names used on this page
   515→  const pageFonts = new Set<string>();
   516→  pageFonts.add(pageFont);
   517→  pageFonts.add(BSML_FONT);
   518→  for (const line of data.lines ?? []) {
   519→    for (const w of line.words ?? []) {
   520→      if (w.font) pageFonts.add(w.font);
   521→    }
   522→  }
   523→
   524→  console.log(`[Mushaf] Page ${pageNum} needs fonts: ${[...pageFonts].join(', ')}`);
   525→
   526→  // Load all fonts in parallel with individual timeouts
   527→  const fontPromises = [...pageFonts].map((fn) =>
   528→    Promise.race([
   529→      ensureFontLoaded(fn),
   530→      new Promise<void>((resolve) => setTimeout(resolve, _isCapacitor ? 8000 : 5000))
   531→    ])
   532→  );
   533→  await Promise.all(fontPromises);
   534→
   535→  // Ensure Reem Kufi is loaded for bismillah rendering
   536→  try {
   537→    await document.fonts.load('500 40px "Reem Kufi"');
   538→  } catch {
   539→    /* ignore */
   540→  }
   541→
   542→  // Wait for all fonts to be ready
   543→  try {
   544→    await document.fonts.ready;
   545→  } catch {
   546→    /* ignore */
   547→  }
   548→
   549→  // In Capacitor: extra wait for WebView font processing
   550→  if (_isCapacitor) {
   551→    await new Promise<void>((r) => setTimeout(r, 800));
   552→    await document.fonts.ready;
   553→  } else {
   554→    // Non-Capacitor: verify font on Canvas
   555→    if (!loadedFonts.has(pageFont) || !verifyFontOnCanvas(pageFont)) {
   556→      console.warn(`[Mushaf] Font "${pageFont}" not verified on Canvas, attempting final reload...`);
   557→      loadedFonts.delete(pageFont);
   558→      await ensureFontLoaded(pageFont);
   559→    }
   560→  }
   561→
   562→  const canvas = targetCanvas || document.createElement('canvas');
   563→  canvas.width = CANVAS_W;
   564→  canvas.height = CANVAS_H;
   565→  const ctx = canvas.getContext('2d');
   566→
   567→  const colors = getColors();
   568→
   569→  ctx!.fillStyle = colors.bg;
   570→  ctx!.fillRect(0, 0, CANVAS_W, CANVAS_H);
   571→
   572→  renderPageContent(ctx!, data, pageFont, colors);
   573→
   574→  drawPageFrame(ctx!, colors);
   575→
   576→  drawPageNumber(ctx!, pageNum, colors);
   577→
   578→  console.log(`[Mushaf] Page ${pageNum} rendered successfully`);
   579→
   580→  // In Capacitor: schedule MULTIPLE re-renders because Android WebView
   581→  // sometimes needs several render passes for fonts to take effect on Canvas.
   582→  if (_isCapacitor) {
   583→    const renderData = { data, pageFont, colors, pageNum };
   584→    [2500, 5000, 8000].forEach((delay) => {
   585→      setTimeout(() => {
   586→        console.log(`[Mushaf] Capacitor: Re-rendering page ${renderData.pageNum} after ${delay}ms delay`);
   587→        try {
   588→          const c = targetCanvas || document.querySelector('.mushaf-page-canvas') as HTMLCanvasElement;
   589→          if (c && c.width === CANVAS_W) {
   590→            const ctx2 = c.getContext('2d');
   591→            if (ctx2) {
   592→              ctx2.fillStyle = renderData.colors.bg;
   593→              ctx2.fillRect(0, 0, CANVAS_W, CANVAS_H);
   594→              renderPageContent(ctx2, renderData.data, renderData.pageFont, renderData.colors);
   595→              drawPageFrame(ctx2, renderData.colors);
   596→              drawPageNumber(ctx2, renderData.pageNum, renderData.colors);
   597→              console.log(`[Mushaf] Capacitor: Re-render of page ${renderData.pageNum} complete at ${delay}ms`);
   598→            }
   599→          }
   600→        } catch (e) {
   601→          console.warn('[Mushaf] Capacitor re-render failed:', e);
   602→        }
   603→      }, delay);
   604→    });
   605→  }
   606→
   607→  return { canvas, layout: data };
   608→}

   610→function measureLine(ctx: CanvasRenderingContext2D, words: PageWord[], pageFont: string, fontSize: number): number[] {
   611→  return words.map((w) => {
   612→    const fn = w.font || pageFont;
   613→    ctx.font = `${fontSize}px "${fn}"`;
   614→    return ctx.measureText(w.char).width;
   615→  });
   616→}

   618→export function getLineY(lineIndex: number, lineCount: number, imgHeight: number): number {
   619→  if (lineIndex <= 0) return 0;
   620→  if (lineIndex >= lineCount) return imgHeight;
   621→  const usableHeight = CANVAS_H - TOP_OFFSET - BOTTOM_OFFSET - PAD_V;
   622→  const stdLineHeight = usableHeight / STD_LINES;
   623→  const isShortPage = lineCount < STD_LINES;
   624→  const lineSpacing = isShortPage ? (usableHeight - stdLineHeight) / Math.max(1, lineCount - 1) : stdLineHeight;
   625→  const y = TOP_OFFSET + lineIndex * lineSpacing + (isShortPage ? 0 : stdLineHeight / 2);
   626→  return (y / CANVAS_H) * imgHeight;
   627→}

   629→/** Pre-compute per-word tajweed coloring data for the entire page. */
   630→function computePageTajweed(data: PageLayoutData): { wordIdx: number; lineIdx: number; color: string | null }[] | null {
   631→  if (!state.tajweedEnabled) return null;
   632→
   633→  const verseKeyWords = new Map<string, { text: string; lineIdx: number; wordIdx: number }[]>();
   634→  for (let li = 0; li < data.lines.length; li++) {
   635→    const line = data.lines[li];
   636→    if (!line?.words) continue;
   637→    for (let wi = 0; wi < line.words.length; wi++) {
   638→      const w = line.words[wi];
   639→      if (w.verse_key && w.text && w.type === 'word') {
   640→        if (!verseKeyWords.has(w.verse_key)) verseKeyWords.set(w.verse_key, []);
   641→        verseKeyWords.get(w.verse_key)!.push({ text: w.text, lineIdx: li, wordIdx: wi });
   642→      }
   643→    }
   644→  }
   645→
   646→  const result: { wordIdx: number; lineIdx: number; color: string | null }[] = [];
   647→
   648→  for (const [vk, words] of verseKeyWords) {
   649→    const parts = vk.split(':');
   650→    const surah = parseInt(parts[0], 10);
   651→    const ayah = parseInt(parts[1], 10);
   652→
   653→    const annotations: TajweedAnnotation[] = getAyahAnnotations(surah, ayah);
   654→    if (annotations.length === 0) {
   655→      for (const w of words) result.push({ wordIdx: w.wordIdx, lineIdx: w.lineIdx, color: null });
   656→      continue;
   657→    }
   658→
   659→    const ayahText = words.map((w) => w.text).join(' ');
   660→    const colorMap = buildColorMap(annotations);
   661→    if (!colorMap || colorMap.size === 0) {
   662→      for (const w of words) result.push({ wordIdx: w.wordIdx, lineIdx: w.lineIdx, color: null });
   663→      continue;
   664→    }
   665→
   666→    let outputPos = 0;
   667→    for (let wi = 0; wi < words.length; wi++) {
   668→      const wordText = words[wi].text;
   669→      let wordColor: string | null = null;
   670→      for (let ci = 0; ci < wordText.length; ci++) {
   671→        const c = colorMap.get(outputPos + ci);
   672→        if (c) {
   673→          wordColor = c;
   674→          break;
   675→        }
   676→      }
   677→      result.push({ wordIdx: words[wi].wordIdx, lineIdx: words[wi].lineIdx, color: wordColor });
   678→      outputPos += wordText.length + 1;
   679→    }
   680→  }
   681→
   682→  return result;
   683→}

   685→function renderPageContent(
   686→  ctx: CanvasRenderingContext2D,
   687→  data: PageLayoutData,
   688→  pageFont: string,
   689→  colors: PageColors
   690→): void {
   691→  const lines = data.lines;
   692→  if (!lines || lines.length === 0) return;
   693→
   694→  const lineCount = lines.length;
   695→  const usableHeight = CANVAS_H - TOP_OFFSET - BOTTOM_OFFSET - PAD_V;
   696→  const availableW = CANVAS_W - PAD_H * 2;
   697→
   698→  const stdLineHeight = usableHeight / STD_LINES;
   699→  const baseFontSize = Math.max(26, Math.min(55, stdLineHeight * 0.85));
   700→  const pageFontSize = baseFontSize;
   701→
   702→  const isShortPage = lineCount < 15;
   703→  const lineSpacing = isShortPage ? (usableHeight - stdLineHeight) / Math.max(1, lineCount - 1) : stdLineHeight;
   704→
   705→  const lineWidths: (LineWidths | null)[] = [];
   706→  for (let i = 0; i < lineCount; i++) {
   707→    const line = lines[i];
   708→    if (!line?.words || line.words.length === 0) {
   709→      lineWidths.push(null);
   710→      continue;
   711→    }
   712→    const widths = measureLine(ctx, line.words, pageFont, pageFontSize);
   713→    const totalW = widths.reduce((a: number, b: number) => a + b, 0);
   714→    const gap = line.words.length > 1 ? (availableW - totalW) / (line.words.length - 1) : 0;
   715→    lineWidths.push({ widths, gap: Math.max(0, gap) });
   716→  }
   717→
   718→  const tajweedColors = computePageTajweed(data);
   719→  const tajweedLookup = new Map<string, string | null>();
   720→  if (tajweedColors) {
   721→    for (const entry of tajweedColors) {
   722→      tajweedLookup.set(`${entry.lineIdx}:${entry.wordIdx}`, entry.color);
   723→    }
   724→  }
   725→
   726→  const bsmlColor = isNightMode() ? '#d4af37' : '#8b6914';
   727→  const headerColor = isNightMode() ? '#c4a87c' : '#7a5c3a';
   728→  const bsmlFont = 'Reem Kufi';
   729→  const bsmlText = 'بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ';
   730→
   731→  ctx.textBaseline = 'middle';
   732→
   733→  for (let i = 0; i < lineCount; i++) {
   734→    const line = lines[i];
   735→    if (!line?.words || line.words.length === 0) continue;
   736→
   737→    const y = TOP_OFFSET + i * lineSpacing + (isShortPage ? 0 : stdLineHeight / 2);
   738→    const words = line.words;
   739→    const lw = lineWidths[i];
   740→    if (!lw) continue;
   741→    const { widths, gap } = lw;
   742→
   743→    const lineType = words[0]?.type;
   744→    const isBismillahLine = lineType === 'bismillah';
   745→    const isHeaderLine = lineType === 'surah_header';
   746→
   747→    if (isBismillahLine) {
   748→      const bsmlFontSize = Math.round(pageFontSize * 0.95);
   749→      ctx.font = `500 ${bsmlFontSize}px "${bsmlFont}", "Scheherazade New", serif`;
   750→      ctx.textAlign = 'center';
   751→      ctx.fillStyle = bsmlColor;
   752→      ctx.fillText(bsmlText, CANVAS_W / 2, y);
   753→      continue;
   754→    }
   755→
   756→    let x = CANVAS_W - PAD_H;
   757→
   758→    for (let j = 0; j < words.length; j++) {
   759→      const w = words[j];
   760→      const fn = w.font || pageFont;
   761→      ctx.font = `${pageFontSize}px "${fn}"`;
   762→      ctx.textAlign = 'right';
   763→
   764→      if (isHeaderLine) {
   765→        ctx.fillStyle = headerColor;
   766→      } else {
   767→        const wordColor = tajweedLookup ? tajweedLookup.get(`${i}:${j}`) : null;
   768→        ctx.fillStyle = wordColor || colors.txt;
   769→      }
   770→
   771→      ctx.fillText(w.char, x, y);
   772→      x -= widths[j] + gap;
   773→    }
   774→  }
   775→}

   777→function drawPageNumber(ctx: CanvasRenderingContext2D, pageNum: number, _colors: PageColors): void {
   778→  const cx = CANVAS_W / 2;
   779→  const cy = CANVAS_H - 18;
   780→  const numStr = pageNum.toLocaleString('ar-SA');
   781→  ctx.font = 'bold 16px "Amiri", serif';
   782→  ctx.textAlign = 'center';
   783→  ctx.textBaseline = 'middle';
   784→  ctx.fillStyle = isNightMode() ? '#6a6e7e' : '#8b6f5a';
   785→  ctx.fillText(numStr, cx, cy);
   786→}

   788→function drawPageFrame(ctx: CanvasRenderingContext2D, colors: PageColors): void {
   789→  const m = 8;
   790→  ctx.strokeStyle = colors.frame;
   791→  ctx.lineWidth = 2;
   792→  ctx.strokeRect(m, m, CANVAS_W - m * 2, CANVAS_H - m * 2);
   793→
   794→  ctx.strokeStyle = colors.frameInner;
   795→  ctx.lineWidth = 1;
   796→  const gap = 6;
   797→  ctx.strokeRect(m + gap, m + gap, CANVAS_W - (m + gap) * 2, CANVAS_H - (m + gap) * 2);
   798→}
